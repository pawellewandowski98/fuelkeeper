# FuelKeeper Plan 2: Pool Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the in-process hot path (allocator, coalescer, shard manager, pool service) and the leased background jobs (reaper with oracle, watcher, minter, janitor) on top of the Plan 1 store contract, with a mock treasury and mock oracle, and prove G1/G2 with chaos and randomized multi-instance invariant tests.

**Architecture:** Each instance leases whole shards and serves reservations from an in-memory FIFO; the coalescer turns many concurrent reserve requests into one idempotent `store.Reserve` per flush and only ever answers with outpoints the store reported as applied. Correctness never depends on the lease or the FIFO: every store write is a CAS. Background jobs run on one instance at a time under `store.Leaser` and only perform CAS transitions, so double runs are harmless.

**Tech Stack:** Go 1.26, `pkg/store` (Plan 1), `github.com/bsv-blockchain/go-sdk` (`chainhash`, `script` for the puzzle scripts), `log/slog`, `github.com/prometheus/client_golang` (metrics registration only; HTTP exposure is Plan 3), testify.

**Spec:** `docs/superpowers/specs/2026-09-29-fuelkeeper-design.md` — "Sharded ownership (hot path)", "Background jobs", "Chain oracle", "Guarantees".

**Model tiering (user preference):** implementers on `sonnet` for every task (complete code is given); spec reviews on `sonnet`; code-quality reviews of Tasks 3–7 and 11 (coalescer, shard manager, service, reaper, invariant test) on the most capable model, the rest on `sonnet`.

// Package fuel holds the fuel-output primitives shared by the pool, treasury and
// clients: hash-puzzle scripts, preimage derivation and fee arithmetic.
package fuel

import (
	"crypto/rand"
	"crypto/sha256"
	"encoding/binary"
)

// PreimageSize is the puzzle preimage length in bytes.
const PreimageSize = 32

const (
	opSHA256 = 0xa8
	opEQUAL  = 0x87
	push32   = 0x20
)

// Seed is a per-shard secret from which every output's preimage is derived.
type Seed [32]byte

// NewSeed returns a cryptographically random seed.
func NewSeed() Seed {
	var s Seed
	if _, err := rand.Read(s[:]); err != nil {
		panic("fuel: crypto/rand failed: " + err.Error())
	}
	return s
}

// Preimage derives the preimage for output vout: SHA256(seed || big-endian vout).
func Preimage(seed Seed, vout uint32) [PreimageSize]byte {
	var buf [36]byte
	copy(buf[:32], seed[:])
	binary.BigEndian.PutUint32(buf[32:], vout)
	return sha256.Sum256(buf[:])
}

// PuzzleHash is the value committed in the locking script.
func PuzzleHash(preimage [PreimageSize]byte) [32]byte { return sha256.Sum256(preimage[:]) }

// PuzzleLock returns the locking script OP_SHA256 <hash> OP_EQUAL (35 bytes).
func PuzzleLock(hash [32]byte) []byte {
	out := make([]byte, 0, 35)
	out = append(out, opSHA256, push32)
	out = append(out, hash[:]...)
	return append(out, opEQUAL)
}

// PuzzleUnlock returns the unlocking script: a single 32-byte push (33 bytes).
func PuzzleUnlock(preimage [PreimageSize]byte) []byte {
	out := make([]byte, 0, 33)
	out = append(out, push32)
	return append(out, preimage[:]...)
}

// ParsePuzzleLock extracts the hash from a PuzzleLock script.
func ParsePuzzleLock(lock []byte) ([32]byte, bool) {
	var h [32]byte
	if len(lock) != 35 || lock[0] != opSHA256 || lock[1] != push32 || lock[34] != opEQUAL {
		return h, false
	}
	copy(h[:], lock[2:34])
	return h, true
}

// VerifyPuzzle reports whether unlock satisfies lock.
func VerifyPuzzle(lock, unlock []byte) bool {
	h, ok := ParsePuzzleLock(lock)
	if !ok || len(unlock) != 33 || unlock[0] != push32 {
		return false
	}
	var p [PreimageSize]byte
	copy(p[:], unlock[1:])
	return PuzzleHash(p) == h
}
```

const (
	PuzzleInputSize = 74  // 41 + 33-byte unlocking script
	P2PKHInputSize  = 148 // 41 + 107-byte unlocking script

// and the wallet toolboxes use (the node floors, so ceil always satisfies it).
func FeeSats(sizeBytes int, satPerKB uint64) uint64 {
	if sizeBytes <= 0 {
		return 0
	}
	n := uint64(sizeBytes) * satPerKB
	return (n + 999) / 1000
}

// Package oracle answers "is this outpoint spent?" and "what happened to this tx?"
// from the chain/mempool. The reaper and watcher never guess: an Unknown answer
// leaves rows quarantined.
package oracle

import (
	"context"
	"errors"

	"github.com/bsv-blockchain/go-sdk/chainhash"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

// ErrUnavailable means the oracle could not answer right now; retry later.
var ErrUnavailable = errors.New("oracle: unavailable")

// Status of one outpoint.
type Status int

const (
	StatusUnknown     Status = iota // no verdict; leave the row alone
	StatusUnspent                   // seen in UTXO set / mempool as unspent
	StatusSpent                     // spent; SpendingTxID set when known
	StatusConflicting               // double-spend attempt observed; treat as not safe to requeue
)

func (s Status) String() string {
	switch s {
	case StatusUnspent:
		return "unspent"
	case StatusSpent:
		return "spent"
	case StatusConflicting:
		return "conflicting"
	}
	return "unknown"
}

// Result is the verdict for one outpoint.
type Result struct {
	Outpoint     store.Outpoint
	Status       Status
	SpendingTxID *chainhash.Hash
}

// TxStatus is the verdict for one transaction id.
type TxStatus struct {
	Known        bool   // the oracle has seen this txid
	Mined        bool   // included in a block
	Rejected     bool   // terminally rejected (invalid or lost a double-spend)
	Status       string // backend status string, for logs/metrics
	BlockHeight  uint32
	MerklePath   []byte // BUMP bytes when Mined and available
	CompetingTxs []string
}

// Oracle is implemented by teranode/arcade/woc clients (Plan 5), Mock and None.
type Oracle interface {
	// Outpoints returns one Result per input outpoint, in input order.
	Outpoints(ctx context.Context, ops []store.Outpoint) ([]Result, error)
	// Tx returns the status of a transaction.
	Tx(ctx context.Context, txid chainhash.Hash) (TxStatus, error)
}
```

// None answers Unknown for everything. Quarantined rows then stay quarantined

type None struct{}

func (None) Outpoints(_ context.Context, ops []store.Outpoint) ([]Result, error) {
	out := make([]Result, len(ops))
	for i, op := range ops {

	}
	return out, nil
}

package oracle

import (
	"context"
	"sync"

	"github.com/bsv-blockchain/go-sdk/chainhash"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

// Mock is a controllable oracle for tests.
type Mock struct {
	mu    sync.Mutex
	ops   map[store.Outpoint]Result
	txs   map[chainhash.Hash]TxStatus
	down  bool
	calls int
}

var _ Oracle = (*Mock)(nil)

// NewMock returns an empty mock: everything is Unknown until set.
func NewMock() *Mock {
	return &Mock{ops: map[store.Outpoint]Result{}, txs: map[chainhash.Hash]TxStatus{}}
}

// SetUnspent marks op unspent.
func (m *Mock) SetUnspent(op store.Outpoint) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.ops[op] = Result{Outpoint: op, Status: StatusUnspent}
}

// SetSpent marks op spent by txid.
func (m *Mock) SetSpent(op store.Outpoint, txid chainhash.Hash) {
	m.mu.Lock()
	defer m.mu.Unlock()
	h := txid
	m.ops[op] = Result{Outpoint: op, Status: StatusSpent, SpendingTxID: &h}
}

// SetConflicting marks op as having a double-spend conflict.
func (m *Mock) SetConflicting(op store.Outpoint) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.ops[op] = Result{Outpoint: op, Status: StatusConflicting}
}

// Forget removes any verdict for op (back to Unknown).
func (m *Mock) Forget(op store.Outpoint) {
	m.mu.Lock()
	defer m.mu.Unlock()
	delete(m.ops, op)
}

// SetTx sets the verdict for txid.
func (m *Mock) SetTx(txid chainhash.Hash, st TxStatus) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.txs[txid] = st
}

// SetDown makes every call fail with ErrUnavailable.
func (m *Mock) SetDown(down bool) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.down = down
}

// Calls returns how many Outpoints/Tx calls were made.
func (m *Mock) Calls() int {
	m.mu.Lock()
	defer m.mu.Unlock()
	return m.calls
}

func (m *Mock) Outpoints(_ context.Context, ops []store.Outpoint) ([]Result, error) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.calls++
	if m.down {
		return nil, ErrUnavailable
	}
	out := make([]Result, len(ops))
	for i, op := range ops {
		if r, ok := m.ops[op]; ok {
			out[i] = r
		} else {
			out[i] = Result{Outpoint: op, Status: StatusUnknown}
		}
	}
	return out, nil
}

func (m *Mock) Tx(_ context.Context, txid chainhash.Hash) (TxStatus, error) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.calls++
	if m.down {
		return TxStatus{}, ErrUnavailable
	}
	return m.txs[txid], nil
}
```

// Package treasury defines how FuelKeeper obtains new fuel: a Treasury builds a
// split transaction (without broadcasting), the minter records the shard, and
// only then the treasury sends it. That ordering is load-bearing: fuel must never
// exist on-chain without a shard record.
package treasury

import (
	"context"
	"errors"

	"github.com/bsv-blockchain/go-sdk/chainhash"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

var (
	// ErrInvalidRequest is returned for a malformed MintRequest.
	ErrInvalidRequest = errors.New("treasury: invalid mint request")
	// ErrUnsupportedLockMode is returned when the treasury cannot mint the requested lock mode.
	ErrUnsupportedLockMode = errors.New("treasury: unsupported lock mode")
	// ErrInsufficientFunds means the treasury wallet cannot fund a split right now.
	ErrInsufficientFunds = errors.New("treasury: insufficient funds")
	// ErrRejected means the network terminally rejected the split transaction.
	ErrRejected = errors.New("treasury: split rejected")
	// ErrUnknownTx is returned by Send for a txid this treasury did not build.
	ErrUnknownTx = errors.New("treasury: unknown transaction")
)

// MintRequest asks for one split transaction.
type MintRequest struct {
	Pool         string
	Denomination uint64
	Count        int
	LockMode     store.LockMode
}

// Validate checks the request.
func (r MintRequest) Validate() error {
	if err := store.ValidatePool(r.Pool); err != nil {
		return errors.Join(ErrInvalidRequest, err)
	}
	if r.Denomination == 0 || r.Count <= 0 {
		return ErrInvalidRequest
	}
	return nil
}

// BuiltShard is a signed, not-yet-broadcast split: the shard record (status
// unmined, BEEF = split tx + ancestry) and its rows (state available, with
// unlocking data).
type BuiltShard struct {
	Shard store.Shard
	Rows  []store.Row
	RawTx []byte
}

// Treasury builds and sends split transactions.
type Treasury interface {
	// Build creates and signs a split without broadcasting it.
	Build(ctx context.Context, req MintRequest) (*BuiltShard, error)
	// Send broadcasts a previously built split. It is idempotent; a terminal
	// network rejection returns ErrRejected.
	Send(ctx context.Context, txid chainhash.Hash) error
}
```

// Treasury is the mock.
type Treasury struct {
	mu        sync.Mutex
	built     map[chainhash.Hash]*treasury.BuiltShard
	sent      map[chainhash.Hash]int // send count per txid
	failBuild error
	failSend  error
	nBuilt    int
}

var _ treasury.Treasury = (*Treasury)(nil)

// New returns an empty mock treasury.
func New() *Treasury {
	return &Treasury{built: map[chainhash.Hash]*treasury.BuiltShard{}, sent: map[chainhash.Hash]int{}}
}

// Sends returns how many times txid was sent.
func (t *Treasury) Sends(txid chainhash.Hash) int {
	t.mu.Lock()
	defer t.mu.Unlock()
	return t.sent[txid]
}

// FailBuild makes Build return err (nil clears it).
func (t *Treasury) FailBuild(err error) {
	t.mu.Lock()
	defer t.mu.Unlock()
	t.failBuild = err
}

// FailSend makes Send return err (nil clears it).
func (t *Treasury) FailSend(err error) {
	t.mu.Lock()
	defer t.mu.Unlock()
	t.failSend = err
}

// Sent reports whether txid was sent at least once.
func (t *Treasury) Sent(txid chainhash.Hash) bool {
	t.mu.Lock()
	defer t.mu.Unlock()
	return t.sent[txid] > 0
}

// Built returns the number of successful builds.
func (t *Treasury) Built() int {
	t.mu.Lock()
	defer t.mu.Unlock()
	return t.nBuilt
}

func (t *Treasury) Build(_ context.Context, req treasury.MintRequest) (*treasury.BuiltShard, error) {
	if err := req.Validate(); err != nil {
		return nil, err
	}
	if req.LockMode != store.LockPuzzle {
		return nil, treasury.ErrUnsupportedLockMode
	}
	t.mu.Lock()
	defer t.mu.Unlock()
	if t.failBuild != nil {
		return nil, t.failBuild
	}
	var rnd [32]byte
	_, _ = rand.Read(rnd[:])
	sum := sha256.Sum256(rnd[:])
	txid, _ := chainhash.NewHash(sum[:])
	seed := fuel.NewSeed()
	rows := make([]store.Row, req.Count)
	for i := range rows {
		pre := fuel.Preimage(seed, uint32(i)) //nolint:gosec // count is small
		rows[i] = store.Row{
			Outpoint:        store.Outpoint{TxID: *txid, Vout: uint32(i)}, //nolint:gosec // count is small
			Pool:            req.Pool,
			Satoshis:        req.Denomination,
			LockingScript:   fuel.PuzzleLock(fuel.PuzzleHash(pre)),
			UnlockingScript: fuel.PuzzleUnlock(pre),
		}
	}
	raw := append([]byte("mock-rawtx-"), txid[:]...)
	b := &treasury.BuiltShard{
		Shard: store.Shard{
			Pool:         req.Pool,
			TxID:         *txid,
			Size:         req.Count,
			Denomination: req.Denomination,
			LockMode:     store.LockPuzzle,
			Status:       store.ShardUnmined,
			BEEF:         append([]byte("mock-beef-"), txid[:]...),
		},
		Rows:  rows,
		RawTx: raw,
	}
	t.built[*txid] = b
	t.nBuilt++
	return b, nil
}

func (t *Treasury) Send(_ context.Context, txid chainhash.Hash) error {
	t.mu.Lock()
	defer t.mu.Unlock()
	if _, ok := t.built[txid]; !ok {
		return treasury.ErrUnknownTx
	}
	if t.failSend != nil {
		return t.failSend
	}
	t.sent[txid]++
	return nil
}

package fuel

import (
	"errors"
	"fmt"

	"github.com/bsv-blockchain/go-sdk/chainhash"
	"github.com/bsv-blockchain/go-sdk/transaction"
)

// ErrTxNotInBEEF is returned when the BEEF does not contain the requested tx.
var ErrTxNotInBEEF = errors.New("fuel: transaction not in BEEF")

// RawTxFromBEEF extracts the raw bytes of txid from a BEEF blob.
func RawTxFromBEEF(beef []byte, txid chainhash.Hash) ([]byte, error) {
	b, err := transaction.NewBeefFromBytes(beef)
	if err != nil {
		return nil, fmt.Errorf("fuel: parse beef: %w", err)
	}
	tx := b.FindTransaction(txid.String())
	if tx == nil {
		return nil, ErrTxNotInBEEF
	}
	return tx.Bytes(), nil
}

// MinedBEEF builds the compact BEEF v2 for a mined split: the raw split tx plus
// its BUMP. unminedBEEF is the shard's stored BEEF (split + ancestry); merklePath
// is the BUMP bytes reported by the oracle.
func MinedBEEF(unminedBEEF []byte, txid chainhash.Hash, merklePath []byte) ([]byte, error) {
	raw, err := RawTxFromBEEF(unminedBEEF, txid)
	if err != nil {
		return nil, err
	}
	mp, err := transaction.NewMerklePathFromBinary(merklePath)
	if err != nil {
		return nil, fmt.Errorf("fuel: parse merkle path: %w", err)
	}
	if _, err := mp.ComputeRoot(&txid); err != nil {
		return nil, fmt.Errorf("fuel: merkle path does not include %s: %w", txid, err)
	}
	out := transaction.NewBeefV2()
	idx := out.MergeBump(mp)
	if _, err := out.MergeRawTx(raw, &idx); err != nil {
		return nil, fmt.Errorf("fuel: merge raw tx: %w", err)
	}
	bytes, err := out.Bytes()
	if err != nil {
		return nil, fmt.Errorf("fuel: serialise beef: %w", err)
	}
	return bytes, nil
}

// ServePolicy says which shard statuses an instance may serve from.
type ServePolicy string

const (
	ServeMinedOnly ServePolicy = "mined_only"
	ServeAny       ServePolicy = "any"
)

// ReleasePolicy says whether a client Release is trusted.
type ReleasePolicy string

const (
	ReleaseTrust  ReleasePolicy = "trust"  // reserved → available immediately
	ReleaseVerify ReleasePolicy = "verify" // reserved → quarantined; the reaper asks the oracle
)

// SpendPolicy says whether a client Spend is a hint or a fact.
type SpendPolicy string

const (
	SpendVerify SpendPolicy = "verify" // reserved → quarantined(hint); oracle proves spent
	SpendTrust  SpendPolicy = "trust"  // reserved → spent
)

// PoolConfig describes one pool (one denomination).
type PoolConfig struct {
	Name            string         `yaml:"name"`
	Denomination    uint64         `yaml:"denomination"`
	LockMode        store.LockMode `yaml:"-"`
	LockModeName    string         `yaml:"lock_mode"` // "puzzle" | "p2pkh"; parsed into LockMode by Validate
	SplitSize       int            `yaml:"split_size"`
	TargetAvailable int64          `yaml:"target_available"`
	LowWater        int64          `yaml:"low_water"`
	ServePolicy     ServePolicy    `yaml:"serve_policy"`
	DefaultTTL      time.Duration  `yaml:"default_ttl"`
	MinTTL          time.Duration  `yaml:"min_ttl"`
	MaxTTL          time.Duration  `yaml:"max_ttl"`
	MaxCount        int            `yaml:"max_count"`
	ReleasePolicy   ReleasePolicy  `yaml:"release_policy"`
	SpendPolicy     SpendPolicy    `yaml:"spend_policy"`
}

func DefaultPoolConfig(name string, denomination uint64) PoolConfig {
	return PoolConfig{
		Name:            name,
		Denomination:    denomination,
		LockMode:        store.LockPuzzle,
		LockModeName:    "puzzle",
		SplitSize:       500,
		TargetAvailable: 100_000,
		LowWater:        60_000,
		ServePolicy:     ServeMinedOnly,
		DefaultTTL:      2 * time.Minute,
		MinTTL:          30 * time.Second,
		MaxTTL:          10 * time.Minute,
		MaxCount:        100,
		ReleasePolicy:   ReleaseTrust,
		SpendPolicy:     SpendVerify,
	}
}

// Validate checks the pool config and resolves LockModeName.
func (c *PoolConfig) Validate() error {
	if err := store.ValidatePool(c.Name); err != nil {
		return err
	}
	if c.Denomination == 0 {
		return fmt.Errorf("pool %s: denomination must be > 0", c.Name)
	}
	switch c.LockModeName {
	case "", "puzzle":
		c.LockMode = store.LockPuzzle
	case "p2pkh":
		c.LockMode = store.LockP2PKH
	default:
		return fmt.Errorf("pool %s: unknown lock_mode %q", c.Name, c.LockModeName)
	}
	if c.SplitSize <= 0 || c.SplitSize > 10_000 {
		return fmt.Errorf("pool %s: split_size must be 1..10000", c.Name)
	}
	if c.TargetAvailable <= 0 || c.LowWater < 0 || c.LowWater > c.TargetAvailable {
		return fmt.Errorf("pool %s: need 0 <= low_water <= target_available and target_available > 0", c.Name)
	}
	switch c.ServePolicy {
	case ServeMinedOnly, ServeAny:
	default:
		return fmt.Errorf("pool %s: serve_policy must be mined_only or any", c.Name)
	}
	if c.MinTTL <= 0 || c.MaxTTL < c.MinTTL || c.DefaultTTL < c.MinTTL || c.DefaultTTL > c.MaxTTL {
		return fmt.Errorf("pool %s: need 0 < min_ttl <= default_ttl <= max_ttl", c.Name)
	}
	if c.MaxCount <= 0 {
		return fmt.Errorf("pool %s: max_count must be > 0", c.Name)
	}
	switch c.ReleasePolicy {
	case ReleaseTrust, ReleaseVerify:
	default:
		return fmt.Errorf("pool %s: release_policy must be trust or verify", c.Name)
	}
	switch c.SpendPolicy {
	case SpendVerify, SpendTrust:
	default:
		return fmt.Errorf("pool %s: spend_policy must be verify or trust", c.Name)
	}
	return nil
}

func (c PoolConfig) ServeStatuses() []store.ShardStatus {
	if c.ServePolicy == ServeAny {
		return []store.ShardStatus{store.ShardMined, store.ShardUnmined}
	}
	return []store.ShardStatus{store.ShardMined}

}

```go
package pool

import (
	"bytes"
	"sort"
	"sync"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

// OwnedShard is the allocator's view of a shard this instance leases.
type OwnedShard struct {
	TxID      chainhash.Hash
	Status    store.ShardStatus
	Version   int64
	CreatedAt time.Time
}

type shardQueue struct {
	head OwnedShard
	fifo []store.Row // front at index 0
}

// Allocator is the per-pool in-memory FIFO of available rows across owned
// shards. Rows are popped oldest-shard-first. A row is "known" from Add until
// Forget (or RemoveShard); known rows are never re-added, which makes rescans
// safe while rows are in flight to the coalescer.
type Allocator struct {
	mu     sync.Mutex
	order  []*shardQueue // oldest first
	byTx   map[chainhash.Hash]*shardQueue
	known  map[store.Outpoint]struct{}
	avail  int
}

// NewAllocator returns an empty allocator.
func NewAllocator() *Allocator {
	return &Allocator{byTx: map[chainhash.Hash]*shardQueue{}, known: map[store.Outpoint]struct{}{}}
}

func lessOwned(a, b OwnedShard) bool {
	if !a.CreatedAt.Equal(b.CreatedAt) {
		return a.CreatedAt.Before(b.CreatedAt)
	}
	return bytes.Compare(a.TxID[:], b.TxID[:]) < 0
}

// AddShard registers a shard; false if already present.
func (a *Allocator) AddShard(sh OwnedShard) bool {
	a.mu.Lock()
	defer a.mu.Unlock()
	if _, ok := a.byTx[sh.TxID]; ok {
		return false
	}
	q := &shardQueue{head: sh}
	a.byTx[sh.TxID] = q
	a.order = append(a.order, q)
	sort.SliceStable(a.order, func(i, j int) bool { return lessOwned(a.order[i].head, a.order[j].head) })
	return true
}

// RemoveShard drops a shard and its FIFO, forgetting those rows. Returns how
// many queued rows were dropped. Rows of that shard currently in flight stay
// known until Forget.
func (a *Allocator) RemoveShard(txid chainhash.Hash) int {
	a.mu.Lock()
	defer a.mu.Unlock()
	q, ok := a.byTx[txid]
	if !ok {
		return 0
	}
	for _, r := range q.fifo {
		delete(a.known, r.Outpoint)
	}
	n := len(q.fifo)
	a.avail -= n
	delete(a.byTx, txid)
	for i, o := range a.order {
		if o == q {
			a.order = append(a.order[:i], a.order[i+1:]...)
			break
		}
	}
	return n
}

// Has reports whether the shard is registered.
func (a *Allocator) Has(txid chainhash.Hash) bool {
	a.mu.Lock()
	defer a.mu.Unlock()
	_, ok := a.byTx[txid]
	return ok
}

// UpdateHead records a new status/version for an owned shard.
func (a *Allocator) UpdateHead(txid chainhash.Hash, status store.ShardStatus, version int64) {
	a.mu.Lock()
	defer a.mu.Unlock()
	if q, ok := a.byTx[txid]; ok {
		q.head.Status, q.head.Version = status, version
	}
}

// Add appends rows to the shard's FIFO, skipping known outpoints and rows of
// other shards. Returns how many were added.
func (a *Allocator) Add(txid chainhash.Hash, rows []store.Row) int {
	a.mu.Lock()
	defer a.mu.Unlock()
	q, ok := a.byTx[txid]
	if !ok {
		return 0
	}
	n := 0
	for _, r := range rows {
		if r.TxID != txid {
			continue
		}
		if _, dup := a.known[r.Outpoint]; dup {
			continue
		}
		a.known[r.Outpoint] = struct{}{}
		q.fifo = append(q.fifo, r)
		n++
	}
	a.avail += n
	return n
}

// Pop removes up to n rows, oldest shard first. Popped rows remain known.
func (a *Allocator) Pop(n int) []store.Row {
	if n <= 0 {
		return nil
	}
	a.mu.Lock()
	defer a.mu.Unlock()
	out := make([]store.Row, 0, n)
	for _, q := range a.order {
		if len(out) == n {
			break
		}
		take := n - len(out)
		if take > len(q.fifo) {
			take = len(q.fifo)
		}
		out = append(out, q.fifo[:take]...)
		q.fifo = append([]store.Row(nil), q.fifo[take:]...)
	}
	a.avail -= len(out)
	return out
}

// Return puts rows that never reached the store back at the FRONT of their
// shard's FIFO. Rows whose shard is gone are forgotten instead.
func (a *Allocator) Return(rows []store.Row) {
	a.mu.Lock()
	defer a.mu.Unlock()
	for i := len(rows) - 1; i >= 0; i-- { // reverse so original order is preserved at the front
		r := rows[i]
		q, ok := a.byTx[r.TxID]
		if !ok {
			delete(a.known, r.Outpoint)
			continue
		}
		if _, known := a.known[r.Outpoint]; !known {
			a.known[r.Outpoint] = struct{}{}
		}
		q.fifo = append([]store.Row{r}, q.fifo...)
		a.avail++
	}
}

// Forget removes outpoints from the known set (after a CAS result, applied or
// missed). They are not in any FIFO at that point.
func (a *Allocator) Forget(ops []store.Outpoint) {
	a.mu.Lock()
	defer a.mu.Unlock()
	for _, op := range ops {
		delete(a.known, op)
	}
}

// Available returns the number of queued rows across all shards.
func (a *Allocator) Available() int {
	a.mu.Lock()
	defer a.mu.Unlock()
	return a.avail
}

// ShardAvailable returns the queued rows of one shard.
func (a *Allocator) ShardAvailable(txid chainhash.Hash) int {
	a.mu.Lock()
	defer a.mu.Unlock()
	if q, ok := a.byTx[txid]; ok {
		return len(q.fifo)
	}
	return 0
}

// Shards lists owned shards oldest first.
func (a *Allocator) Shards() []OwnedShard {
	a.mu.Lock()
	defer a.mu.Unlock()
	out := make([]OwnedShard, len(a.order))
	for i, q := range a.order {
		out[i] = q.head
	}
	return out
}
```

package storetest

import (
	"context"
	"errors"
	"sync"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

// ErrInjected is the error Faulty returns when a fault fires.
var ErrInjected = errors.New("storetest: injected failure")

// Faulty wraps a Store and injects failures into Reserve. Every other method is
// delegated unchanged.
//
//   - FailAfterApplying(n): the next Reserve applies only the first n ops, then
//     returns ErrInjected wrapped in store.ErrUnavailable (partial apply + lost
//     response).
//   - InDoubtEvery(k): every k-th Reserve applies fully but reports
//     store.ErrUnavailable (commit succeeded, response lost).
//   - BeforeReserve / AfterReserve hooks run around every Reserve (for pausing).
type Faulty struct {
	store.Store
	mu            sync.Mutex
	failAfter     int  // -1 = off
	inDoubtEvery  int  // 0 = off
	reserveCalls  int
	BeforeReserve func(ops []store.ReserveOp)
	AfterReserve  func(applied []store.ReserveOp, err error)
}

// NewFaulty wraps s.
func NewFaulty(s store.Store) *Faulty { return &Faulty{Store: s, failAfter: -1} }

// FailAfterApplying arms a one-shot partial failure.
func (f *Faulty) FailAfterApplying(n int) {
	f.mu.Lock()
	defer f.mu.Unlock()
	f.failAfter = n
}

// InDoubtEvery makes every k-th Reserve report ErrUnavailable after committing.
func (f *Faulty) InDoubtEvery(k int) {
	f.mu.Lock()
	defer f.mu.Unlock()
	f.inDoubtEvery = k
}

// ReserveCalls returns how many Reserve calls reached the wrapper.
func (f *Faulty) ReserveCalls() int {
	f.mu.Lock()
	defer f.mu.Unlock()
	return f.reserveCalls
}

func (f *Faulty) Reserve(ctx context.Context, pool string, ops []store.ReserveOp) ([]store.ReserveOp, error) {
	if f.BeforeReserve != nil {
		f.BeforeReserve(ops)
	}
	f.mu.Lock()
	f.reserveCalls++
	calls := f.reserveCalls
	failAfter := f.failAfter
	f.failAfter = -1
	inDoubt := f.inDoubtEvery > 0 && calls%f.inDoubtEvery == 0
	f.mu.Unlock()

	var applied []store.ReserveOp
	var err error
	switch {
	case failAfter >= 0:
		if failAfter > 0 {
			applied, err = f.Store.Reserve(ctx, pool, ops[:min(failAfter, len(ops))])
		}
		if err == nil {
			err = errors.Join(store.ErrUnavailable, ErrInjected)
		}
		applied = nil // the caller never learns what applied
	default:
		applied, err = f.Store.Reserve(ctx, pool, ops)
		if err == nil && inDoubt {
			applied, err = nil, errors.Join(store.ErrUnavailable, ErrInjected)
		}
	}
	if f.AfterReserve != nil {
		f.AfterReserve(applied, err)
	}
	return applied, err
}

// Compile-time checks that Faulty still satisfies the interface through embedding.
var _ store.Store = (*Faulty)(nil)

// Keep chainhash and time referenced for future hooks without unused-import churn.
var (
	_ = chainhash.Hash{}
	_ = time.Second
)

package pool

import (
	"context"
	"errors"
	"fmt"
	"sync"
	"sync/atomic"
	"time"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

// CoalescerConfig tunes group commit.
type CoalescerConfig struct {
	Slots          int           // concurrent flushes (store Caps.MaxConcurrentFlushes)
	MaxBatch       int           // ops per flush (<= store Caps.MaxBatch)
	Window         time.Duration // optional accumulation window when a slot is free (0 = none)
	MaxInflightOps int           // backpressure threshold; 0 = 4*Slots*MaxBatch
	Retries        int           // retries after the first attempt on ErrUnavailable; default 2
	RetryBackoff   time.Duration // first backoff; doubles; default 20ms
	FlushTimeout   time.Duration // per-attempt store timeout; default 5s
	CompensateTime time.Duration // budget for compensating releases; default 2s
}

func (c CoalescerConfig) withDefaults() CoalescerConfig {
	if c.Slots <= 0 {
		c.Slots = 1
	}
	if c.MaxBatch <= 0 {
		c.MaxBatch = 256
	}
	if c.MaxInflightOps <= 0 {
		c.MaxInflightOps = 4 * c.Slots * c.MaxBatch
	}
	if c.Retries < 0 {
		c.Retries = 0
	} else if c.Retries == 0 {
		c.Retries = 2
	}
	if c.RetryBackoff <= 0 {
		c.RetryBackoff = 20 * time.Millisecond
	}
	if c.FlushTimeout <= 0 {
		c.FlushTimeout = 5 * time.Second
	}
	if c.CompensateTime <= 0 {
		c.CompensateTime = 2 * time.Second
	}
	return c
}

// Result is what Submit returns.
type Result struct {
	Applied   []store.ReserveOp // committed under the request's token; safe to hand out
	Missed    []store.Outpoint  // CAS failed (not available / other token); caller forgets and picks others
	Unflushed bool              // the request never reached the store; caller may Return its rows
	Err       error             // ErrBadRequest, ErrBackpressure, ctx error, or ErrUnavailable
}

const (
	reqPending   int32 = 0
	reqDelivered int32 = 1
	reqAbandoned int32 = 2
)

type request struct {
	ops    []store.ReserveOp
	state  atomic.Int32
	done   chan struct{}
	result Result
}

// Coalescer groups concurrent reserve requests into batched, idempotent
// store.Reserve calls. Start it with Run; submit with Submit.
type Coalescer struct {
	st   store.Store
	pool string
	cfg  CoalescerConfig

	mu         sync.Mutex
	pending    []*request
	pendingOps int
	inflight   atomic.Int64 // ops in pending + in flight
	kick       chan struct{}
	slots      chan struct{}
	runCtx     context.Context //nolint:containedctx // dispatcher context for detached flush contexts
	stopped    atomic.Bool
}

// NewCoalescer creates a coalescer for one pool.
func NewCoalescer(st store.Store, pool string, cfg CoalescerConfig) *Coalescer {
	cfg = cfg.withDefaults()
	return &Coalescer{
		st:    st,
		pool:  pool,
		cfg:   cfg,
		kick:  make(chan struct{}, 1),
		slots: make(chan struct{}, cfg.Slots),
	}
}

// Submit enqueues ops under one token (all ops must share it) and waits for the
// flush that carries them, or for ctx.
func (c *Coalescer) Submit(ctx context.Context, ops []store.ReserveOp) Result {
	if len(ops) == 0 || len(ops) > c.cfg.MaxBatch {
		return Result{Unflushed: true, Err: fmt.Errorf("%w: reserve batch must have 1..%d ops", ErrBadRequest, c.cfg.MaxBatch)}
	}
	if err := store.ValidateReserveOps(ops, c.cfg.MaxBatch); err != nil {
		return Result{Unflushed: true, Err: fmt.Errorf("%w: %w", ErrBadRequest, err)}
	}
	if c.stopped.Load() {
		return Result{Unflushed: true, Err: ErrUnavailable}
	}
	req := &request{ops: ops, done: make(chan struct{})}
	if !c.enqueue(req) {
		return Result{Unflushed: true, Err: ErrBackpressure}
	}
	select {
	case <-req.done:
		return req.result
	case <-ctx.Done():
	}
	if c.withdraw(req) {
		return Result{Unflushed: true, Err: ctx.Err()}
	}
	// Already taken by a flush. Whoever wins the state CAS handles applied rows.
	if req.state.CompareAndSwap(reqPending, reqAbandoned) {
		return Result{Err: ctx.Err()}
	}
	<-req.done // delivered concurrently: the result is ours to clean up
	c.releaseApplied(req.result.Applied)
	return Result{Err: ctx.Err()}
}

func (c *Coalescer) enqueue(req *request) bool {
	n := int64(len(req.ops))
	if c.inflight.Add(n) > int64(c.cfg.MaxInflightOps) {
		c.inflight.Add(-n)
		return false
	}
	c.mu.Lock()
	c.pending = append(c.pending, req)
	c.pendingOps += len(req.ops)
	c.mu.Unlock()
	select {
	case c.kick <- struct{}{}:
	default:
	}
	return true
}

// withdraw removes a request that has not been taken by a flush yet.
func (c *Coalescer) withdraw(req *request) bool {
	c.mu.Lock()
	defer c.mu.Unlock()
	for i, p := range c.pending {
		if p == req {
			c.pending = append(c.pending[:i], c.pending[i+1:]...)
			c.pendingOps -= len(req.ops)
			c.inflight.Add(-int64(len(req.ops)))
			return true
		}
	}
	return false
}

// takeBatch pops whole requests until MaxBatch ops are collected.
func (c *Coalescer) takeBatch() []*request {
	c.mu.Lock()
	defer c.mu.Unlock()
	var batch []*request
	n := 0
	for len(c.pending) > 0 {
		next := c.pending[0]
		if n > 0 && n+len(next.ops) > c.cfg.MaxBatch {
			break
		}
		batch = append(batch, next)
		n += len(next.ops)
		c.pending = c.pending[1:]
		c.pendingOps -= len(next.ops)
	}
	return batch
}

func (c *Coalescer) hasPending() bool {
	c.mu.Lock()
	defer c.mu.Unlock()
	return len(c.pending) > 0
}

// Run dispatches flushes until ctx is done, then fails queued requests as
// Unflushed. In-flight flushes settle on their own detached contexts.
func (c *Coalescer) Run(ctx context.Context) error {
	c.runCtx = ctx
	for {
		select {
		case <-ctx.Done():
			c.drain(ctx.Err())
			return nil
		case <-c.kick:
		}
		for c.hasPending() {
			select {
			case c.slots <- struct{}{}:
			case <-ctx.Done():
				c.drain(ctx.Err())
				return nil
			}
			if c.cfg.Window > 0 {
				timer := time.NewTimer(c.cfg.Window)
				select {
				case <-timer.C:
				case <-ctx.Done():
					timer.Stop()
				}
			}
			batch := c.takeBatch()
			if len(batch) == 0 {
				<-c.slots
				continue
			}
			go c.flush(batch)
		}
	}
}

func (c *Coalescer) drain(err error) {
	c.stopped.Store(true)
	c.mu.Lock()
	pending := c.pending
	c.pending = nil
	c.pendingOps = 0
	c.mu.Unlock()
	for _, req := range pending {
		c.inflight.Add(-int64(len(req.ops)))
		c.deliver(req, Result{Unflushed: true, Err: err})
	}
}

func (c *Coalescer) flush(batch []*request) {
	defer func() {
		<-c.slots
		select {
		case c.kick <- struct{}{}:
		default:
		}
	}()
	total := 0
	for _, r := range batch {
		total += len(r.ops)
	}
	ops := make([]store.ReserveOp, 0, total)
	for _, r := range batch {
		ops = append(ops, r.ops...)
	}
	defer c.inflight.Add(-int64(total))

	base := context.WithoutCancel(c.runCtx)
	var applied []store.ReserveOp
	var err error
	backoff := c.cfg.RetryBackoff
	for attempt := 0; attempt <= c.cfg.Retries; attempt++ {
		if attempt > 0 {
			time.Sleep(backoff)
			backoff *= 2
		}
		fctx, cancel := context.WithTimeout(base, c.cfg.FlushTimeout)
		applied, err = c.st.Reserve(fctx, c.pool, ops)
		cancel()
		if err == nil {
			break
		}
		if !errors.Is(err, store.ErrUnavailable) && !errors.Is(err, context.DeadlineExceeded) {
			break // not retryable
		}
	}
	if err != nil {
		// Outcome unknown: compensate by token. Release on a never-applied row is a no-op.
		for _, r := range batch {
			c.releaseApplied(r.ops)
			c.deliver(r, Result{Err: fmt.Errorf("%w: %w", ErrUnavailable, err)})
		}
		return
	}

	appliedBy := make(map[store.Outpoint]store.ReserveOp, len(applied))
	for _, op := range applied {
		appliedBy[op.Outpoint] = op
	}
	for _, r := range batch {
		res := Result{}
		for _, op := range r.ops {
			if a, ok := appliedBy[op.Outpoint]; ok && a.Token == op.Token {
				res.Applied = append(res.Applied, a)
			} else {
				res.Missed = append(res.Missed, op.Outpoint)
			}
		}
		c.deliver(r, res)
	}
}

// deliver hands the result to the waiter, or releases applied rows if the
// waiter abandoned the request.
func (c *Coalescer) deliver(req *request, res Result) {
	if req.state.CompareAndSwap(reqPending, reqDelivered) {
		req.result = res
		close(req.done)
		return
	}
	// abandoned: nobody will use these rows
	req.result = res
	close(req.done)
	c.releaseApplied(res.Applied)
}

// releaseApplied best-effort releases ops grouped by token under a detached context.
func (c *Coalescer) releaseApplied(ops []store.ReserveOp) {
	if len(ops) == 0 {
		return
	}
	byToken := map[store.Token][]store.Outpoint{}
	for _, op := range ops {
		byToken[op.Token] = append(byToken[op.Token], op.Outpoint)
	}
	ctx, cancel := context.WithTimeout(context.Background(), c.cfg.CompensateTime)
	defer cancel()
	for tok, outs := range byToken {
		_, _ = c.st.Release(ctx, c.pool, tok, outs) // errors leave rows to the expiry path
	}
}
```

Semantics to preserve when editing: (1) an outpoint is in `Result.Applied` only if the store returned it applied under the request's own token; (2) `Unflushed` is true only when the store was never called for the request; (3) after `Err != nil` from a flush, rows have been compensated best-effort and the caller must `Forget` them (never `Return`).

// NewShardManager builds a manager with a fresh holder id.
func NewShardManager(st store.Store, pc PoolConfig, cfg InstanceConfig, alloc *Allocator, clock Clock, log *slog.Logger) *ShardManager {
	return &ShardManager{
		st:            st,
		pc:            pc,
		cfg:           cfg,
		alloc:         alloc,
		clock:         clock,
		log:           log.With("component", "shardmgr", "pool", pc.Name),
		holder:        NewHolder(cfg.InstanceID),
		serve:         pc.ServeStatuses(),

		recentlyEmpty: map[chainhash.Hash]time.Time{},
		hungry:        make(chan struct{}, 1),

// Holder returns this instance's lease holder id.
func (m *ShardManager) Holder() string { return m.holder }

// Hungry signals the run loop to try adopting (non-blocking).
func (m *ShardManager) Hungry() {
	select {
	case m.hungry <- struct{}{}:
	default:
	}
}

func (m *ShardManager) isHungry() bool {
	m.mu.Lock()
	n := len(m.owned)
	m.mu.Unlock()
	return n < m.cfg.MaxShards && m.alloc.Available() < m.cfg.LowWaterRows
}

// FeedOnce adopts shards while hungry, bounded by MaxShards and adoptAttempts.
func (m *ShardManager) FeedOnce(ctx context.Context) error {
	for i := 0; i < adoptAttempts && m.isHungry(); i++ {
		adopted, err := m.AdoptOnce(ctx)
		if err != nil {
			return err
		}
		if !adopted {
			return nil
		}
	}
	return nil
}

// EnsureRows is the bounded synchronous path Reserve uses when the FIFO is
// empty: it tries to adopt until rows are available, AdoptTimeout passes, or
// nothing is adoptable. Returns true if rows are now available.
func (m *ShardManager) EnsureRows(ctx context.Context) bool {
	if m.alloc.Available() > 0 {
		return true
	}
	ctx, cancel := context.WithTimeout(ctx, m.cfg.AdoptTimeout)
	defer cancel()
	for i := 0; i < adoptAttempts; i++ {
		m.mu.Lock()
		full := len(m.owned) >= m.cfg.MaxShards
		m.mu.Unlock()
		if full {
			return m.alloc.Available() > 0
		}
		adopted, err := m.AdoptOnce(ctx)
		if err != nil || !adopted {
			return m.alloc.Available() > 0
		}
		if m.alloc.Available() > 0 {
			return true
		}
	}
	return m.alloc.Available() > 0
}

func (m *ShardManager) excludeList(now time.Time) []chainhash.Hash {
	m.mu.Lock()
	defer m.mu.Unlock()
	var ex []chainhash.Hash
	for txid, at := range m.recentlyEmpty {
		if now.Sub(at) > emptyExclusion {
			delete(m.recentlyEmpty, txid)
			continue
		}
		ex = append(ex, txid)
	}
	return ex
}

// AdoptOnce tries to acquire one adoptable shard and load its rows. It returns
// false when nothing was adoptable (or the adopted shard turned out empty and
// was released again).
func (m *ShardManager) AdoptOnce(ctx context.Context) (bool, error) {
	now := m.clock.Now()
	until := now.Add(m.cfg.LeaseTTL)
	sh, err := m.st.AcquireShard(ctx, m.pc.Name, m.holder, now, until, m.serve, m.excludeList(now))
	if err != nil {
		return false, err
	}
	if sh == nil {
		return false, nil
	}
	m.alloc.AddShard(OwnedShard{TxID: sh.TxID, Status: sh.Status, Version: sh.Version, CreatedAt: sh.CreatedAt})
	m.mu.Lock()
	m.owned[sh.TxID] = &ownedState{head: store.ShardHead{Status: sh.Status, Version: sh.Version}, leaseUntil: until}
	m.mu.Unlock()
	loaded, err := m.loadRows(ctx, sh.TxID)
	if err != nil {
		return true, err // keep the shard; rescan will retry
	}
	if loaded == 0 {
		m.drop(ctx, sh.TxID, true)
		return false, nil
	}
	m.log.Debug("adopted shard", "txid", sh.TxID.String(), "rows", loaded)
	return true, nil
}

// loadRows pages ListAvailable into the allocator.
func (m *ShardManager) loadRows(ctx context.Context, txid chainhash.Hash) (int, error) {
	total := 0
	var from uint32
	for {
		rows, err := m.st.ListAvailable(ctx, m.pc.Name, txid, from, listPage)
		if err != nil {
			return total, err
		}
		total += m.alloc.Add(txid, rows)
		if len(rows) < listPage {
			return total, nil
		}
		from = rows[len(rows)-1].Vout + 1
	}
}

// drop forgets a shard locally and, when release is true, releases its lease.
func (m *ShardManager) drop(ctx context.Context, txid chainhash.Hash, release bool) {
	m.alloc.RemoveShard(txid)
	m.mu.Lock()
	delete(m.owned, txid)
	if release {
		m.recentlyEmpty[txid] = m.clock.Now()
	}
	m.mu.Unlock()
	if release {
		if err := m.st.ReleaseShard(ctx, m.pc.Name, txid, m.holder); err != nil && !errors.Is(err, store.ErrNotFound) {
			m.log.Warn("release shard", "txid", txid.String(), "err", err)
		}
	}
}

func (m *ShardManager) ownedIDs() []chainhash.Hash {
	m.mu.Lock()
	defer m.mu.Unlock()
	ids := make([]chainhash.Hash, 0, len(m.owned))
	for id := range m.owned {
		ids = append(ids, id)
	}
	return ids
}

// RenewOnce renews every owned lease and drops shards that were lost or are no
// longer servable.
func (m *ShardManager) RenewOnce(ctx context.Context) error {
	now := m.clock.Now()
	until := now.Add(m.cfg.LeaseTTL)
	var firstErr error
	for _, txid := range m.ownedIDs() {
		head, err := m.st.RenewShard(ctx, m.pc.Name, txid, m.holder, until)
		switch {
		case errors.Is(err, store.ErrNotOwner), errors.Is(err, store.ErrNotFound):
			m.log.Info("lost shard lease", "txid", txid.String())
			m.drop(ctx, txid, false)
			continue
		case err != nil:
			if firstErr == nil {
				firstErr = err
			}
			// keep serving; but drop if the lease has been expired for more than one TTL
			m.mu.Lock()
			st := m.owned[txid]
			stale := st != nil && now.Sub(st.leaseUntil) > m.cfg.LeaseTTL
			m.mu.Unlock()
			if stale {
				m.log.Warn("dropping shard with stale lease", "txid", txid.String())
				m.drop(ctx, txid, false)
			}
			continue
		}
		if !m.servable(head.Status) {
			m.log.Info("shard no longer servable", "txid", txid.String(), "status", head.Status.String())
			m.drop(ctx, txid, true)
			continue
		}
		m.mu.Lock()
		if st := m.owned[txid]; st != nil {
			st.leaseUntil = until
			st.head = head
		}
		m.mu.Unlock()
		m.alloc.UpdateHead(txid, head.Status, head.Version)
	}
	return firstErr
}

func (m *ShardManager) servable(s store.ShardStatus) bool {
	for _, want := range m.serve {
		if s == want {
			return true
		}
	}
	return false
}

// RescanOnce reloads available rows of owned shards and drops shards that stay
// empty for emptyRescansMax consecutive scans.
func (m *ShardManager) RescanOnce(ctx context.Context) error {
	var firstErr error
	for _, txid := range m.ownedIDs() {
		added, err := m.loadRows(ctx, txid)
		if err != nil {
			if firstErr == nil {
				firstErr = err
			}
			continue
		}
		m.mu.Lock()
		st := m.owned[txid]
		if st == nil {
			m.mu.Unlock()
			continue
		}
		st.rescanNow = false
		if added == 0 && m.alloc.ShardAvailable(txid) == 0 {
			st.emptyRescans++
		} else {
			st.emptyRescans = 0
		}
		empty := st.emptyRescans >= emptyRescansMax
		m.mu.Unlock()
		if empty {
			m.log.Debug("dropping drained shard", "txid", txid.String())
			m.drop(ctx, txid, true)
		}
	}
	return firstErr
}

// NoteCASMiss flags a shard for an immediate rescan (another holder may be serving it).
func (m *ShardManager) NoteCASMiss(txid chainhash.Hash) {
	m.mu.Lock()
	if st := m.owned[txid]; st != nil {
		st.rescanNow = true
	}
	m.mu.Unlock()
	m.Hungry()
}

// NoteReleased re-adds rows released through this instance when we own their shard.
func (m *ShardManager) NoteReleased(rows []store.Row) {
	byShard := map[chainhash.Hash][]store.Row{}
	for _, r := range rows {
		byShard[r.TxID] = append(byShard[r.TxID], r)
	}
	for txid, rs := range byShard {
		if m.alloc.Has(txid) {
			m.alloc.Add(txid, rs)
		}
	}
}

// ReleaseAll releases every owned shard (graceful shutdown).
func (m *ShardManager) ReleaseAll(ctx context.Context) error {
	var firstErr error
	for _, txid := range m.ownedIDs() {
		m.alloc.RemoveShard(txid)
		m.mu.Lock()
		delete(m.owned, txid)
		m.mu.Unlock()
		if err := m.st.ReleaseShard(ctx, m.pc.Name, txid, m.holder); err != nil && firstErr == nil {
			firstErr = err
		}
	}
	return firstErr
}

// Run drives FeedOnce on hunger, RenewOnce every LeaseTTL/3 and RescanOnce
// every RescanInterval, until ctx is done; then it releases all shards.
func (m *ShardManager) Run(ctx context.Context) error {
	renew := time.NewTicker(m.cfg.LeaseTTL / 3)
	defer renew.Stop()
	rescan := time.NewTicker(m.cfg.RescanInterval)
	defer rescan.Stop()
	m.Hungry()
	for {
		select {
		case <-ctx.Done():
			rctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 5*time.Second)
			defer cancel()
			return m.ReleaseAll(rctx)
		case <-m.hungry:
			if err := m.FeedOnce(ctx); err != nil {
				m.log.Warn("adopt", "err", err)
			}
		case <-renew.C:
			if err := m.RenewOnce(ctx); err != nil {
				m.log.Warn("renew", "err", err)
			}
			if m.isHungry() {
				m.Hungry()
			}
		case <-rescan.C:
			if err := m.RescanOnce(ctx); err != nil {
				m.log.Warn("rescan", "err", err)
			}
			if m.isHungry() {
				m.Hungry()
			}
		}
	}
}

package pool

import (
	"context"
	"errors"
	"fmt"
	"log/slog"
	"sync"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

// ReserveRequest is what a client asks for.
type ReserveRequest struct {
	Count    int
	MinCount int           // 0 = Count
	TTL      time.Duration // 0 = pool default
}

// Output is one reserved fuel output with its unlocking data.
type Output struct {
	store.Outpoint
	Satoshis        uint64
	LockingScript   []byte
	UnlockingScript []byte
}

// ShardRef points a client at the BEEF it needs (GET /v1/beef/{txid}).
type ShardRef struct {
	TxID    chainhash.Hash
	Version int64
}

// Reservation is a successful reserve.
type Reservation struct {
	Token     store.Token
	ExpiresAt time.Time
	TTL       time.Duration
	LockMode  store.LockMode
	Outputs   []Output
	Shards    []ShardRef
}

// Pool serves one configured pool on this instance.
type Pool struct {
	cfg   PoolConfig
	icfg  InstanceConfig
	st    store.Store
	alloc *Allocator
	mgr   *ShardManager
	coal  *Coalescer
	clock Clock
	log   *slog.Logger
}

// NewPool wires allocator, shard manager and coalescer for pc.
func NewPool(st store.Store, pc PoolConfig, ic InstanceConfig, clock Clock, log *slog.Logger) *Pool {
	caps := st.Capabilities()
	maxBatch := ic.MaxBatch
	if maxBatch <= 0 || maxBatch > caps.MaxBatch {
		maxBatch = min(256, caps.MaxBatch)
	}
	alloc := NewAllocator()
	return &Pool{
		cfg:   pc,
		icfg:  ic,
		st:    st,
		alloc: alloc,
		mgr:   NewShardManager(st, pc, ic, alloc, clock, log),
		coal: NewCoalescer(st, pc.Name, CoalescerConfig{
			Slots:        max(1, caps.MaxConcurrentFlushes),
			MaxBatch:     maxBatch,
			Window:       ic.FlushWindow,
			FlushTimeout: ic.FlushTimeout,
		}),
		clock: clock,
		log:   log.With("component", "pool", "pool", pc.Name),
	}
}

// Config returns the pool configuration.
func (p *Pool) Config() PoolConfig { return p.cfg }

// Holder returns the shard-lease holder id of this instance for this pool.
func (p *Pool) Holder() string { return p.mgr.Holder() }

// EnsureRows exposes the bounded adoption path (tests, readiness).
func (p *Pool) EnsureRows(ctx context.Context) bool { return p.mgr.EnsureRows(ctx) }

// Run runs the coalescer and shard manager until ctx is done.
func (p *Pool) Run(ctx context.Context) error {
	var wg sync.WaitGroup
	var mu sync.Mutex
	var first error
	run := func(f func(context.Context) error) {
		defer wg.Done()
		if err := f(ctx); err != nil {
			mu.Lock()
			if first == nil {
				first = err
			}
			mu.Unlock()
		}
	}
	wg.Add(2)
	go run(p.coal.Run)
	go run(p.mgr.Run)
	wg.Wait()
	return first
}

func (p *Pool) validate(req ReserveRequest) (count, minCount int, ttl time.Duration, err error) {
	count = req.Count
	if count <= 0 || count > p.cfg.MaxCount {
		return 0, 0, 0, fmt.Errorf("%w: count must be 1..%d", ErrBadRequest, p.cfg.MaxCount)
	}
	minCount = req.MinCount
	if minCount == 0 {
		minCount = count
	}
	if minCount < 1 || minCount > count {
		return 0, 0, 0, fmt.Errorf("%w: min_count must be 1..count", ErrBadRequest)
	}
	ttl = req.TTL
	if ttl == 0 {
		ttl = p.cfg.DefaultTTL
	}
	if ttl < p.cfg.MinTTL || ttl > p.cfg.MaxTTL {
		return 0, 0, 0, fmt.Errorf("%w: ttl must be %s..%s", ErrBadRequest, p.cfg.MinTTL, p.cfg.MaxTTL)
	}
	return count, minCount, ttl, nil
}

// Reserve implements the spec's request flow.
func (p *Pool) Reserve(ctx context.Context, req ReserveRequest) (*Reservation, error) {
	start := time.Now()
	count, minCount, ttl, err := p.validate(req)
	if err != nil {
		reserveRequests.WithLabelValues(p.cfg.Name, "bad_request").Inc()
		return nil, err
	}
	ctx, cancel := context.WithTimeout(ctx, p.icfg.ReserveTimeout)
	defer cancel()

	token := store.NewToken()
	expiresAt := p.clock.Now().Add(ttl)
	need := count
	var applied []store.ReserveOp
	byOutpoint := make(map[store.Outpoint]store.Row, count)
	var flowErr error

	for round := 0; round < 3 && need > 0; round++ {
		rows := p.alloc.Pop(need)
		if len(rows) == 0 {
			if round > 0 || !p.mgr.EnsureRows(ctx) {
				break
			}
			if rows = p.alloc.Pop(need); len(rows) == 0 {
				break
			}
		}
		for _, r := range rows {
			byOutpoint[r.Outpoint] = r
		}
		ops := make([]store.ReserveOp, len(rows))
		for i, r := range rows {
			ops[i] = store.ReserveOp{Outpoint: r.Outpoint, Token: token, ExpiresAt: expiresAt}
		}
		res := p.coal.Submit(ctx, ops)
		if res.Unflushed {
			p.alloc.Return(rows)
			flowErr = res.Err
			break
		}
		p.alloc.Forget(opsOf(rows))
		if res.Err != nil {
			flowErr = res.Err
			break
		}
		applied = append(applied, res.Applied...)
		need -= len(res.Applied)
		if len(res.Missed) > 0 {
			casMissed.WithLabelValues(p.cfg.Name).Add(float64(len(res.Missed)))
			seen := map[chainhash.Hash]bool{}
			for _, op := range res.Missed {
				if !seen[op.TxID] {
					seen[op.TxID] = true
					p.mgr.NoteCASMiss(op.TxID)
				}
			}
		}
	}
	allocatorAvailable.WithLabelValues(p.cfg.Name).Set(float64(p.alloc.Available()))
	shardsOwned.WithLabelValues(p.cfg.Name).Set(float64(len(p.alloc.Shards())))
	if p.alloc.Available() < p.icfg.LowWaterRows {
		p.mgr.Hungry()
	}

	if len(applied) < minCount {
		if len(applied) > 0 {
			p.releaseBestEffort(token, applied)
		}
		switch {
		case flowErr != nil:
			reserveRequests.WithLabelValues(p.cfg.Name, "error").Inc()
			return nil, mapFlowErr(flowErr)
		case len(applied) == 0:
			reserveRequests.WithLabelValues(p.cfg.Name, "exhausted").Inc()
			return nil, ErrExhausted
		default:
			reserveRequests.WithLabelValues(p.cfg.Name, "insufficient").Inc()
			return nil, ErrInsufficient
		}
	}

	out := &Reservation{Token: token, ExpiresAt: expiresAt, TTL: ttl, LockMode: p.cfg.LockMode, Outputs: make([]Output, 0, len(applied))}
	shardSeen := map[chainhash.Hash]bool{}
	for _, op := range applied {
		r := byOutpoint[op.Outpoint]
		out.Outputs = append(out.Outputs, Output{Outpoint: r.Outpoint, Satoshis: r.Satoshis, LockingScript: r.LockingScript, UnlockingScript: r.UnlockingScript})
		if !shardSeen[op.TxID] {
			shardSeen[op.TxID] = true
			ver := int64(1)
			for _, sh := range p.alloc.Shards() {
				if sh.TxID == op.TxID {
					ver = sh.Version
				}
			}
			out.Shards = append(out.Shards, ShardRef{TxID: op.TxID, Version: ver})
		}
	}
	reserveRequests.WithLabelValues(p.cfg.Name, "ok").Inc()
	reserveOutputs.WithLabelValues(p.cfg.Name).Add(float64(len(out.Outputs)))
	reserveLatency.WithLabelValues(p.cfg.Name).Observe(time.Since(start).Seconds())
	return out, nil
}

func mapFlowErr(err error) error {
	switch {
	case errors.Is(err, ErrBackpressure), errors.Is(err, ErrUnavailable), errors.Is(err, ErrBadRequest):
		return err
	case errors.Is(err, context.DeadlineExceeded), errors.Is(err, context.Canceled):
		return fmt.Errorf("%w: %w", ErrUnavailable, err)
	default:
		return fmt.Errorf("%w: %w", ErrUnavailable, err)
	}
}

func opsOf(rows []store.Row) []store.Outpoint {
	ops := make([]store.Outpoint, len(rows))
	for i, r := range rows {
		ops[i] = r.Outpoint
	}
	return ops
}

func (p *Pool) releaseBestEffort(token store.Token, applied []store.ReserveOp) {
	ops := make([]store.Outpoint, len(applied))
	for i, a := range applied {
		ops[i] = a.Outpoint
	}
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()
	if _, err := p.st.Release(ctx, p.cfg.Name, token, ops); err != nil {
		p.log.Warn("release partial reservation", "err", err)
		return
	}
	p.noteReleased(ctx, ops)
}

// noteReleased re-adds released rows of owned shards to the allocator.
func (p *Pool) noteReleased(ctx context.Context, ops []store.Outpoint) {
	var owned []store.Outpoint
	for _, op := range ops {
		if p.alloc.Has(op.TxID) {
			owned = append(owned, op)
		}
	}
	if len(owned) == 0 {
		return
	}
	rows, err := p.st.GetRows(ctx, p.cfg.Name, owned)
	if err != nil {
		return
	}
	avail := rows[:0]
	for _, r := range rows {
		if r.State == store.StateAvailable {
			avail = append(avail, r)
		}
	}
	p.mgr.NoteReleased(avail)
}

// Release returns reserved rows to the pool (trust) or parks them for the
// reaper to verify (verify). Returns how many rows changed.
func (p *Pool) Release(ctx context.Context, token store.Token, ops []store.Outpoint) (int, error) {
	if token.IsZero() || len(ops) == 0 || len(ops) > p.cfg.MaxCount {
		releaseTotal.WithLabelValues(p.cfg.Name, "bad_request").Inc()
		return 0, fmt.Errorf("%w: token and 1..%d outpoints required", ErrBadRequest, p.cfg.MaxCount)
	}
	var n int
	var err error
	if p.cfg.ReleasePolicy == ReleaseVerify {
		n, err = p.st.Spend(ctx, p.cfg.Name, token, ops, chainhash.Hash{}, false) // zero hint = "no hint"
	} else {
		n, err = p.st.Release(ctx, p.cfg.Name, token, ops)
	}
	if err != nil {
		releaseTotal.WithLabelValues(p.cfg.Name, "error").Inc()
		return 0, mapStoreErr(err)
	}
	releaseTotal.WithLabelValues(p.cfg.Name, "ok").Inc()
	if n > 0 && p.cfg.ReleasePolicy == ReleaseTrust {
		p.noteReleased(ctx, ops)
	}
	return n, nil
}

// Spend records that the client spent rows in txid: a hint that quarantines
// them (verify) or a fact that spends them (trust).
func (p *Pool) Spend(ctx context.Context, token store.Token, ops []store.Outpoint, txid chainhash.Hash) (int, error) {
	if token.IsZero() || len(ops) == 0 || len(ops) > p.cfg.MaxCount || txid == (chainhash.Hash{}) {
		spendTotal.WithLabelValues(p.cfg.Name, "bad_request").Inc()
		return 0, fmt.Errorf("%w: token, txid and 1..%d outpoints required", ErrBadRequest, p.cfg.MaxCount)
	}
	n, err := p.st.Spend(ctx, p.cfg.Name, token, ops, txid, p.cfg.SpendPolicy == SpendTrust)
	if err != nil {
		spendTotal.WithLabelValues(p.cfg.Name, "error").Inc()
		return 0, mapStoreErr(err)
	}
	spendTotal.WithLabelValues(p.cfg.Name, "ok").Inc()
	return n, nil
}

// Stats reads pool counts from the store.
func (p *Pool) Stats(ctx context.Context) (store.Stats, error) {
	st, err := p.st.Stats(ctx, p.cfg.Name)
	if err != nil {
		return st, mapStoreErr(err)
	}
	return st, nil
}

// BEEF returns a shard's BEEF, version and status.
func (p *Pool) BEEF(ctx context.Context, txid chainhash.Hash) ([]byte, int64, store.ShardStatus, error) {
	sh, err := p.st.GetShard(ctx, p.cfg.Name, txid)
	if err != nil {
		return nil, 0, 0, err
	}
	return sh.BEEF, sh.Version, sh.Status, nil
}

func mapStoreErr(err error) error {
	if errors.Is(err, store.ErrInvalidArgument) {
		return fmt.Errorf("%w: %w", ErrBadRequest, err)
	}
	if errors.Is(err, store.ErrUnavailable) {
		return fmt.Errorf("%w: %w", ErrUnavailable, err)
	}
	return err
}

	Interval() time.Duration

	holder string
	ttl    time.Duration
	log    *slog.Logger
	mu     sync.Mutex
	jobs   []Job

// NewRunner creates a runner. leaseTTL must exceed the longest job interval.
func NewRunner(leaser store.Leaser, holder string, leaseTTL time.Duration, log *slog.Logger) *Runner {
	return &Runner{leaser: leaser, holder: holder, ttl: leaseTTL, log: log.With("component", "jobs"), held: map[string]bool{}}
}

// Add registers a job before Run.
func (r *Runner) Add(j Job) {
	r.mu.Lock()
	defer r.mu.Unlock()
	r.jobs = append(r.jobs, j)
}

func leaseName(j Job) string { return "jobs/" + j.Name() }

// Tick acquires or renews the job's lease and, if held, runs it once. It returns
// whether the job ran and the job's error.
func (r *Runner) Tick(ctx context.Context, j Job) (bool, error) {
	until, err := r.leaser.TryAcquireOrRenew(ctx, leaseName(j), r.holder, r.ttl)
	if err != nil {
		return false, err
	}
	held := !until.IsZero()
	r.mu.Lock()
	r.held[j.Name()] = held
	r.mu.Unlock()
	if held {
		jobLeaseHeld.WithLabelValues(j.Name()).Set(1)
	} else {
		jobLeaseHeld.WithLabelValues(j.Name()).Set(0)
		return false, nil
	}
	return true, j.RunOnce(ctx)
}

// ReleaseAll releases every lease this runner holds.
func (r *Runner) ReleaseAll(ctx context.Context) error {
	r.mu.Lock()
	defer r.mu.Unlock()
	var first error
	for _, j := range r.jobs {
		if r.held[j.Name()] {
			if err := r.leaser.ReleaseLease(ctx, leaseName(j), r.holder); err != nil && first == nil {
				first = err
			}
			r.held[j.Name()] = false
		}
	}
	return first
}

// Run ticks every job on its interval until ctx is done, then releases leases.
func (r *Runner) Run(ctx context.Context) error {
	r.mu.Lock()
	jobs := append([]Job(nil), r.jobs...)
	r.mu.Unlock()
	var wg sync.WaitGroup
	for _, j := range jobs {
		wg.Add(1)
		go func(j Job) {
			defer wg.Done()
			t := time.NewTicker(j.Interval())
			defer t.Stop()
			for {
				if _, err := r.Tick(ctx, j); err != nil && ctx.Err() == nil {
					r.log.Warn("job failed", "job", j.Name(), "err", err)
				}
				select {
				case <-ctx.Done():
					return
				case <-t.C:
				}
			}
		}(j)
	}
	wg.Wait()
	rctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 5*time.Second)
	defer cancel()
	return r.ReleaseAll(rctx)
}

import (
	"context"
	"log/slog"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"
	"github.com/prometheus/client_golang/prometheus"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

var (
	reaperQuarantined = promauto.NewCounterVec(prometheus.CounterOpts{
		Name: "fuelkeeper_reaper_quarantined_total", Help: "Rows moved to quarantine by expiry.",
	}, []string{"pool"})
	reaperResolved = promauto.NewCounterVec(prometheus.CounterOpts{
		Name: "fuelkeeper_reaper_resolved_total", Help: "Quarantined rows resolved, by target state.",
	}, []string{"pool", "to"})
	quarantineOldest = promauto.NewGaugeVec(prometheus.GaugeOpts{
		Name: "fuelkeeper_quarantine_oldest_age_seconds", Help: "Age of the oldest quarantined row seen this tick.",
	}, []string{"pool"})
	oracleRequests = promauto.NewCounterVec(prometheus.CounterOpts{
		Name: "fuelkeeper_oracle_requests_total", Help: "Oracle calls by outcome.",
	}, []string{"kind", "result"})
)

// ReaperConfig tunes the expiry/quarantine/oracle flow (spec "Background jobs").
type ReaperConfig struct {
	Interval         time.Duration `yaml:"interval"`
	Grace            time.Duration `yaml:"grace"`              // clock-skew guard before quarantining an expired row
	QuarantineMinAge time.Duration `yaml:"quarantine_min_age"` // wait before the first oracle check
	MinUnspentAge    time.Duration `yaml:"min_unspent_age"`    // unhinted rows: quarantine age before requeue
	HintPatience     time.Duration `yaml:"hint_patience"`      // hinted rows: quarantine age before requeue
	Batch            int           `yaml:"batch"`
	QuarantineBatch  int           `yaml:"quarantine_batch"`
	OracleBatch      int           `yaml:"oracle_batch"`
}

// DefaultReaperConfig returns the spec defaults.
func DefaultReaperConfig() ReaperConfig {
	return ReaperConfig{
		Interval: time.Second, Grace: 10 * time.Second, QuarantineMinAge: 30 * time.Second,
		MinUnspentAge: 30 * time.Second, HintPatience: 10 * time.Minute,
		Batch: 500, QuarantineBatch: 1000, OracleBatch: 200,
	}
}

// Reaper is the expiry job.
type Reaper struct {
	st    store.Store
	orc   oracle.Oracle
	pools []string
	cfg   ReaperConfig
	clock pool.Clock
	log   *slog.Logger
}

// NewReaper creates a reaper for the named pools.
func NewReaper(st store.Store, orc oracle.Oracle, pools []string, cfg ReaperConfig, clock pool.Clock, log *slog.Logger) *Reaper {
	return &Reaper{st: st, orc: orc, pools: pools, cfg: cfg, clock: clock, log: log.With("job", "reaper")}
}

func (r *Reaper) Name() string            { return "reaper" }
func (r *Reaper) Interval() time.Duration { return r.cfg.Interval }

// RunOnce processes every pool. Oracle outages are logged, not returned.
func (r *Reaper) RunOnce(ctx context.Context) error {
	for _, p := range r.pools {
		if err := r.reapPool(ctx, p); err != nil {
			return err
		}
	}
	return nil
}

func hasHint(row store.Row) bool { return row.SpentHint != nil && *row.SpentHint != (chainhash.Hash{}) }

func (r *Reaper) reapPool(ctx context.Context, p string) error {
	now := r.clock.Now()
	n, err := r.st.Quarantine(ctx, p, now.Add(-r.cfg.Grace), r.cfg.QuarantineBatch)
	if err != nil {
		return err
	}
	if n > 0 {
		reaperQuarantined.WithLabelValues(p).Add(float64(n))
	}
	rows, err := r.st.ListQuarantined(ctx, p, now.Add(-r.cfg.QuarantineMinAge), r.cfg.Batch)
	if err != nil {
		return err
	}
	if len(rows) == 0 {
		quarantineOldest.WithLabelValues(p).Set(0)
		return nil
	}
	quarantineOldest.WithLabelValues(p).Set(now.Sub(rows[0].QuarantinedAt).Seconds())

	toSpent := map[chainhash.Hash][]store.Outpoint{}
	var toAvailable []store.Outpoint
	txCache := map[chainhash.Hash]*oracle.TxStatus{}

	for start := 0; start < len(rows); start += r.cfg.OracleBatch {
		end := min(start+r.cfg.OracleBatch, len(rows))
		batch := rows[start:end]
		ops := make([]store.Outpoint, len(batch))
		for i, row := range batch {
			ops[i] = row.Outpoint
		}
		results, err := r.orc.Outpoints(ctx, ops)
		if err != nil {
			oracleRequests.WithLabelValues("outpoints", "error").Inc()
			r.log.Warn("oracle unavailable; leaving rows quarantined", "pool", p, "err", err)
			return nil
		}
		oracleRequests.WithLabelValues("outpoints", "ok").Inc()
		for i, row := range batch {
			res := results[i]
			age := now.Sub(row.QuarantinedAt)
			switch res.Status {
			case oracle.StatusSpent:
				spender := chainhash.Hash{}
				switch {
				case res.SpendingTxID != nil:
					spender = *res.SpendingTxID
				case hasHint(row):
					spender = *row.SpentHint
				}
				toSpent[spender] = append(toSpent[spender], row.Outpoint)
			case oracle.StatusUnspent:
				wait := r.cfg.MinUnspentAge
				if hasHint(row) {
					wait = r.cfg.HintPatience
				}
				if age >= wait {
					toAvailable = append(toAvailable, row.Outpoint)
				}
			case oracle.StatusUnknown:
				if !hasHint(row) {
					continue
				}
				ts, ok := txCache[*row.SpentHint]
				if !ok {
					got, err := r.orc.Tx(ctx, *row.SpentHint)
					if err != nil {
						oracleRequests.WithLabelValues("tx", "error").Inc()
						continue
					}
					oracleRequests.WithLabelValues("tx", "ok").Inc()
					ts = &got
					txCache[*row.SpentHint] = ts
				}
				if ts.Known && !ts.Rejected {
					toSpent[*row.SpentHint] = append(toSpent[*row.SpentHint], row.Outpoint)
				}
			case oracle.StatusConflicting:
				// leave quarantined; an operator decides
			}
		}
	}

	for spender, ops := range toSpent {
		sp := spender
		k, err := r.st.Resolve(ctx, p, ops, store.StateSpent, &sp)
		if err != nil {
			return err
		}
		reaperResolved.WithLabelValues(p, "spent").Add(float64(k))
	}
	if len(toAvailable) > 0 {
		k, err := r.st.Resolve(ctx, p, toAvailable, store.StateAvailable, nil)
		if err != nil {
			return err
		}
		reaperResolved.WithLabelValues(p, "available").Add(float64(k))
	}
	return nil
}
```
If `Resolve` is called with more than the store's `MaxBatch` outpoints, split the slice into `MaxBatch` chunks (`r.st.Capabilities().MaxBatch`); `Batch` (500) is below every backend's limit (1000), so a simple guard `if len(ops) > caps.MaxBatch { chunk }` is enough.

package jobs

import (
	"context"
	"log/slog"
	"time"

	"github.com/prometheus/client_golang/prometheus"
	"github.com/prometheus/client_golang/prometheus/promauto"

var (
	watcherUnmined = promauto.NewGaugeVec(prometheus.GaugeOpts{
		Name: "fuelkeeper_watcher_unmined_shards", Help: "Shards awaiting a merkle proof.",
	}, []string{"pool"})
	watcherOldestUnmined = promauto.NewGaugeVec(prometheus.GaugeOpts{
		Name: "fuelkeeper_watcher_oldest_unmined_age_seconds", Help: "Age of the oldest unmined shard.",
	}, []string{"pool"})
	watcherTransitions = promauto.NewCounterVec(prometheus.CounterOpts{
		Name: "fuelkeeper_watcher_transitions_total", Help: "Shard transitions by outcome.",
	}, []string{"pool", "outcome"})
)

// WatcherConfig tunes proof polling.
type WatcherConfig struct {
	Interval         time.Duration `yaml:"interval"`
	RebroadcastAfter time.Duration `yaml:"rebroadcast_after"` // unknown to the oracle for this long → Send again
	AlertAfter       time.Duration `yaml:"alert_after"`       // unmined for this long → error log
	Limit            int           `yaml:"limit"`
}

// DefaultWatcherConfig returns the spec defaults.
func DefaultWatcherConfig() WatcherConfig {
	return WatcherConfig{Interval: 10 * time.Second, RebroadcastAfter: 5 * time.Minute, AlertAfter: 6 * time.Hour, Limit: 500}
}

// Watcher is the proof job.
type Watcher struct {
	st     store.Store
	orc    oracle.Oracle
	tr     treasury.Treasury
	pools  []string
	cfg    WatcherConfig
	clock  pool.Clock
	log    *slog.Logger
	lastTx map[string]time.Time // txid → last (re)send time seen by this watcher
}

// NewWatcher creates a watcher.
func NewWatcher(st store.Store, orc oracle.Oracle, tr treasury.Treasury, pools []string, cfg WatcherConfig, clock pool.Clock, log *slog.Logger) *Watcher {
	return &Watcher{st: st, orc: orc, tr: tr, pools: pools, cfg: cfg, clock: clock, log: log.With("job", "watcher"), lastTx: map[string]time.Time{}}
}

func (w *Watcher) Name() string            { return "watcher" }
func (w *Watcher) Interval() time.Duration { return w.cfg.Interval }

// RunOnce checks every unmined shard once.
func (w *Watcher) RunOnce(ctx context.Context) error {
	for _, p := range w.pools {
		if err := w.watchPool(ctx, p); err != nil {
			return err
		}
	}
	return nil
}

func (w *Watcher) watchPool(ctx context.Context, p string) error {
	now := w.clock.Now()
	shards, err := w.st.ListShards(ctx, p, store.ShardFilter{Statuses: []store.ShardStatus{store.ShardUnmined}, Limit: w.cfg.Limit})
	if err != nil {
		return err
	}
	watcherUnmined.WithLabelValues(p).Set(float64(len(shards)))
	if len(shards) == 0 {
		watcherOldestUnmined.WithLabelValues(p).Set(0)
		return nil
	}
	watcherOldestUnmined.WithLabelValues(p).Set(now.Sub(shards[0].CreatedAt).Seconds())

	for _, sh := range shards {
		ts, err := w.orc.Tx(ctx, sh.TxID)
		if err != nil {
			oracleRequests.WithLabelValues("tx", "error").Inc()
			w.log.Warn("oracle unavailable", "pool", p, "err", err)
			return nil
		}
		oracleRequests.WithLabelValues("tx", "ok").Inc()
		age := now.Sub(sh.CreatedAt)
		switch {
		case ts.Rejected:
			n, err := w.st.RetireShard(ctx, p, sh.TxID)
			if err != nil {
				return err
			}
			watcherTransitions.WithLabelValues(p, "rejected").Inc()
			w.log.Error("split rejected; shard retired", "pool", p, "txid", sh.TxID.String(), "status", ts.Status, "competing", ts.CompetingTxs, "rows_invalidated", n)
		case ts.Mined && len(ts.MerklePath) > 0:
			full, err := w.st.GetShard(ctx, p, sh.TxID)
			if err != nil {
				return err
			}
			beef, err := w.tr.MinedBEEF(ctx, sh.TxID, full.BEEF, ts.MerklePath)
			if err != nil {
				w.log.Error("build mined beef", "pool", p, "txid", sh.TxID.String(), "err", err)
				continue
			}
			if err := w.st.UpdateShard(ctx, p, sh.TxID, store.ShardMined, beef); err != nil {
				return err
			}
			watcherTransitions.WithLabelValues(p, "mined").Inc()
			w.log.Info("shard mined", "pool", p, "txid", sh.TxID.String(), "height", ts.BlockHeight)
		case !ts.Known:
			last, seen := w.lastTx[sh.TxID.String()]
			if !seen {
				last = sh.CreatedAt
			}
			if now.Sub(last) > w.cfg.RebroadcastAfter {
				if err := w.tr.Send(ctx, sh.TxID); err != nil {
					w.log.Warn("rebroadcast", "pool", p, "txid", sh.TxID.String(), "err", err)
				} else {
					watcherTransitions.WithLabelValues(p, "rebroadcast").Inc()
				}
				w.lastTx[sh.TxID.String()] = now
			}
			if age > w.cfg.AlertAfter {
				w.log.Error("shard unmined for too long", "pool", p, "txid", sh.TxID.String(), "age", age.String())
			}
		default:
			// known, not yet mined: wait
		}
	}
	return nil
}

package jobs

import (
	"context"
	"errors"
	"log/slog"
	"time"

	"github.com/prometheus/client_golang/prometheus"
	"github.com/prometheus/client_golang/prometheus/promauto"

	"github.com/bsv-blockchain/fuelkeeper/pkg/pool"
	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
	"github.com/bsv-blockchain/fuelkeeper/pkg/treasury"
)

var (
	minterSplits = promauto.NewCounterVec(prometheus.CounterOpts{
		Name: "fuelkeeper_minter_splits_total", Help: "Split attempts by outcome.",
	}, []string{"pool", "result"})
	minterLastSuccess = promauto.NewGaugeVec(prometheus.GaugeOpts{
		Name: "fuelkeeper_minter_last_success_timestamp", Help: "Unix time of the last successful split.",
	}, []string{"pool"})
	poolRows = promauto.NewGaugeVec(prometheus.GaugeOpts{
		Name: "fuelkeeper_pool_rows", Help: "Rows by state (sampled by the minter).",
	}, []string{"pool", "state"})
	poolShards = promauto.NewGaugeVec(prometheus.GaugeOpts{
		Name: "fuelkeeper_pool_shards", Help: "Shards by status (sampled by the minter).",
	}, []string{"pool", "status"})
)

// MinterConfig tunes replenishment.
type MinterConfig struct {
	Interval         time.Duration `yaml:"interval"`
	MaxPendingShards int           `yaml:"max_pending_shards"` // unmined shards allowed at once
	MaxPerRound      int           `yaml:"max_per_round"`      // splits per RunOnce
}

// DefaultMinterConfig returns the spec defaults.
func DefaultMinterConfig() MinterConfig {
	return MinterConfig{Interval: 30 * time.Second, MaxPendingShards: 50, MaxPerRound: 20}
}

// Minter is the replenishment job.
type Minter struct {
	st    store.Store
	tr    treasury.Treasury
	pools []pool.PoolConfig
	cfg   MinterConfig
	clock pool.Clock
	log   *slog.Logger
}

// NewMinter creates a minter for the configured pools.
func NewMinter(st store.Store, tr treasury.Treasury, pools []pool.PoolConfig, cfg MinterConfig, clock pool.Clock, log *slog.Logger) *Minter {
	return &Minter{st: st, tr: tr, pools: pools, cfg: cfg, clock: clock, log: log.With("job", "minter")}
}

func (m *Minter) Name() string            { return "minter" }
func (m *Minter) Interval() time.Duration { return m.cfg.Interval }

// RunOnce tops up every pool.
func (m *Minter) RunOnce(ctx context.Context) error {
	for _, pc := range m.pools {
		if err := m.mintPool(ctx, pc); err != nil {
			return err
		}
	}
	return nil
}

func publishStats(p string, st store.Stats) {
	poolRows.WithLabelValues(p, "available").Set(float64(st.Available))
	poolRows.WithLabelValues(p, "reserved").Set(float64(st.Reserved))
	poolRows.WithLabelValues(p, "quarantined").Set(float64(st.Quarantined))
	poolRows.WithLabelValues(p, "spent").Set(float64(st.Spent))
	poolRows.WithLabelValues(p, "invalid").Set(float64(st.Invalid))
	poolShards.WithLabelValues(p, "unmined").Set(float64(st.ShardsUnmined))
	poolShards.WithLabelValues(p, "mined").Set(float64(st.ShardsMined))
	poolShards.WithLabelValues(p, "retired").Set(float64(st.ShardsRetired))
	poolShards.WithLabelValues(p, "invalid").Set(float64(st.ShardsInvalid))
}

func (m *Minter) mintPool(ctx context.Context, pc pool.PoolConfig) error {
	stats, err := m.st.Stats(ctx, pc.Name)
	if err != nil {
		return err
	}
	publishStats(pc.Name, stats)
	unmined, err := m.st.ListShards(ctx, pc.Name, store.ShardFilter{Statuses: []store.ShardStatus{store.ShardUnmined}})
	if err != nil {
		return err
	}
	// Rows in unmined shards are already counted in stats.Available (they are
	// available rows that mined_only instances cannot serve yet), so the deficit
	// is simply target − available.
	deficit := pc.TargetAvailable - stats.Available
	pending := len(unmined)
	for built := 0; deficit > 0 && pending < m.cfg.MaxPendingShards && built < m.cfg.MaxPerRound; built++ {
		b, err := m.tr.Build(ctx, treasury.MintRequest{Pool: pc.Name, Denomination: pc.Denomination, Count: pc.SplitSize, LockMode: pc.LockMode})
		if err != nil {
			minterSplits.WithLabelValues(pc.Name, "build_failed").Inc()
			if errors.Is(err, treasury.ErrInsufficientFunds) {
				m.log.Warn("treasury cannot fund a split", "pool", pc.Name, "deficit", deficit)
				return nil
			}
			m.log.Error("build split", "pool", pc.Name, "err", err)
			return nil
		}
		if err := m.st.PutShard(ctx, b.Shard, b.Rows); err != nil {
			minterSplits.WithLabelValues(pc.Name, "record_failed").Inc()
			return err // never send what we could not record
		}
		if err := m.tr.Send(ctx, b.Shard.TxID); err != nil {
			if errors.Is(err, treasury.ErrRejected) {
				minterSplits.WithLabelValues(pc.Name, "rejected").Inc()
				if _, rerr := m.st.RetireShard(ctx, pc.Name, b.Shard.TxID); rerr != nil {
					return rerr
				}
				m.log.Error("split rejected on send", "pool", pc.Name, "txid", b.Shard.TxID.String(), "err", err)
				continue
			}
			minterSplits.WithLabelValues(pc.Name, "send_failed").Inc()
			m.log.Warn("send split failed; watcher will retry", "pool", pc.Name, "txid", b.Shard.TxID.String(), "err", err)
		} else {
			minterSplits.WithLabelValues(pc.Name, "ok").Inc()
		}
		minterLastSuccess.WithLabelValues(pc.Name).Set(float64(m.clock.Now().Unix()))
		deficit -= int64(pc.SplitSize)
		pending++
	}
	return nil
}

package pool_test

import (
	"context"
	"log/slog"
	"math/rand"
	"sync"
	"testing"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"
	"github.com/stretchr/testify/require"

	"github.com/bsv-blockchain/fuelkeeper/pkg/jobs"
	"github.com/bsv-blockchain/fuelkeeper/pkg/oracle"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
	"github.com/bsv-blockchain/fuelkeeper/pkg/store/memory"
	"github.com/bsv-blockchain/fuelkeeper/pkg/store/storetest"
)

// live tracks, per outpoint, the reservation the "network of clients" believes
// is live: token + expiry. A reservation stops being live when the client
// releases it, spends it, or its expiry passes on the manual clock.
type live struct {
	mu   sync.Mutex
	held map[store.Outpoint]liveRes
}

type liveRes struct {
	tok store.Token
	exp time.Time
}

func (l *live) claim(t *testing.T, now time.Time, tok store.Token, exp time.Time, ops []store.Outpoint) {
	l.mu.Lock()
	defer l.mu.Unlock()
	for _, op := range ops {
		if prev, ok := l.held[op]; ok && prev.tok != tok && now.Before(prev.exp) {
			t.Errorf("G1 violated: %s handed to %s while live under %s until %s (now %s)", op, tok, prev.tok, prev.exp, now)
		}
		l.held[op] = liveRes{tok: tok, exp: exp}
	}
}

func (l *live) drop(ops []store.Outpoint) {
	l.mu.Lock()
	defer l.mu.Unlock()
	for _, op := range ops {
		delete(l.held, op)
	}
}

type instance struct {
	p      *pool.Pool
	cancel context.CancelFunc
	done   chan struct{}
}

func startInstance(t *testing.T, st store.Store, pc pool.PoolConfig, clk *storetest.Clock, id string) *instance {
	t.Helper()
	ic := pool.DefaultInstanceConfig()
	ic.InstanceID = id
	ic.MaxShards = 3
	ic.LowWaterRows = 20
	ic.LeaseTTL = 30 * time.Second
	ic.RescanInterval = 5 * time.Second

	p := pool.NewPool(st, pc, ic, pool.ClockFunc(clk.Now), slog.New(slog.DiscardHandler))
	ctx, cancel := context.WithCancel(context.Background())
	done := make(chan struct{})
	go func() { defer close(done); _ = p.Run(ctx) }()
	return &instance{p: p, cancel: cancel, done: done}
}

func TestInvariantsMultiInstance(t *testing.T) {
	ctx := context.Background()
	rounds := 300 * stressMul()
	clk := storetest.NewClock(storetest.Start)
	st := memory.New(memory.WithClock(clk.Now))
	orc := oracle.NewMock()
	pc := pool.DefaultPoolConfig("main", 50)
	pc.MinTTL = 5 * time.Second
	pc.DefaultTTL = 20 * time.Second
	pc.MaxTTL = time.Minute
	require.NoError(t, pc.Validate())

	// Seed 6 mined shards × 40 rows = 240 rows.
	total := 0
	for i := 0; i < 6; i++ {
		_, rows := storetest.MintShard(t, st, "main", 40, store.ShardMined)
		total += len(rows)
		sh := rows[0].TxID
		require.NoError(t, st.UpdateShard(ctx, "main", sh, store.ShardMined, nil))
		for _, r := range rows {
			orc.SetUnspent(r.Outpoint)
		}
	}

	rcfg := jobs.DefaultReaperConfig()
	rcfg.Grace = 2 * time.Second
	rcfg.QuarantineMinAge = 2 * time.Second
	rcfg.MinUnspentAge = 2 * time.Second
	rcfg.HintPatience = 10 * time.Second
	reaper := jobs.NewReaper(st, orc, []string{"main"}, rcfg, pool.ClockFunc(clk.Now), slog.New(slog.DiscardHandler))

	instances := []*instance{startInstance(t, st, pc, clk, "i0"), startInstance(t, st, pc, clk, "i1"), startInstance(t, st, pc, clk, "i2")}
	t.Cleanup(func() {
		for _, in := range instances {
			in.cancel()
			<-in.done
		}
	})
	// Let managers adopt before the storm.
	for _, in := range instances {
		in.p.EnsureRows(ctx)
	}

	lv := &live{held: map[store.Outpoint]liveRes{}}
	rng := rand.New(rand.NewSource(42))
	var mu sync.Mutex // guards instances slice and rng
	type held struct {
		tok  store.Token
		ops  []store.Outpoint
		inst int
	}
	var mine []held

	for round := 0; round < rounds; round++ {
		mu.Lock()
		action := rng.Intn(100)
		inst := rng.Intn(len(instances))
		in := instances[inst]
		mu.Unlock()
		switch {
		case action < 55: // reserve
			res, err := in.p.Reserve(ctx, pool.ReserveRequest{Count: 1 + rng.Intn(5), MinCount: 1})
			if err != nil {
				continue // exhausted / insufficient are legitimate under load
			}
			ops := outpointsOf(res.Outputs)
			lv.claim(t, clk.Now(), res.Token, res.ExpiresAt, ops)
			mine = append(mine, held{tok: res.Token, ops: ops, inst: inst})
		case action < 70 && len(mine) > 0: // release
			i := rng.Intn(len(mine))
			h := mine[i]
			n, err := in.p.Release(ctx, h.tok, h.ops)
			require.NoError(t, err)
			if n > 0 {
				lv.drop(h.ops)
			}
			mine = append(mine[:i], mine[i+1:]...)
		case action < 85 && len(mine) > 0: // spend on-chain then notify
			i := rng.Intn(len(mine))
			h := mine[i]
			spender := storetest.NewTxID()
			for _, op := range h.ops {
				orc.SetSpent(op, spender)
			}
			orc.SetTx(spender, oracle.TxStatus{Known: true})
			_, err := in.p.Spend(ctx, h.tok, h.ops, spender)
			require.NoError(t, err)
			lv.drop(h.ops)
			mine = append(mine[:i], mine[i+1:]...)
		case action < 92 && len(mine) > 0: // abandon: expiry will handle it (row stays unspent on-chain)
			i := rng.Intn(len(mine))
			mine = append(mine[:i], mine[i+1:]...)
		case action < 96: // time passes; reaper runs
			clk.Advance(3 * time.Second)
			// reservations whose expiry passed are no longer live
			lv.mu.Lock()
			for op, r := range lv.held {
				if !clk.Now().Before(r.exp) {
					delete(lv.held, op)
				}
			}
			lv.mu.Unlock()
			require.NoError(t, reaper.RunOnce(ctx))
		default: // crash an instance (no ReleaseAll) and start a replacement
			mu.Lock()
			victim := instances[inst]
			victim.cancel()
			<-victim.done
			instances[inst] = startInstance(t, st, pc, clk, "r"+string(rune('a'+round%26)))
			mu.Unlock()
		}
	}

	// Drain: release everything still held, advance past every TTL, reap until stable.
	for _, h := range mine {
		_, _ = instances[0].p.Release(ctx, h.tok, h.ops)
	}
	for i := 0; i < 10; i++ {
		clk.Advance(30 * time.Second)
		require.NoError(t, reaper.RunOnce(ctx))
	}

	// G2: every row is in exactly one state, nothing is lost, and after draining
	// every row is available or provably spent.
	stats, err := st.Stats(ctx, "main")
	require.NoError(t, err)
	sum := stats.Available + stats.Reserved + stats.Quarantined + stats.Spent + stats.Invalid
	require.Equal(t, int64(total), sum, "rows conserved")
	require.Equal(t, int64(0), stats.Reserved, "no reservation survives the drain")
	require.Equal(t, int64(0), stats.Quarantined, "every quarantined row resolved (oracle knew every outpoint)")
	require.Equal(t, int64(0), stats.Invalid)
	t.Logf("final: available=%d spent=%d", stats.Available, stats.Spent)
}

	return 1
}
