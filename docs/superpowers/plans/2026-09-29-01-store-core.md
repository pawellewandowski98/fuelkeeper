# FuelKeeper Plan 1: Bootstrap + Store Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the repository and deliver the fuel store contract (`pkg/store`), a conformance suite that pins the G1/G2 guarantees, and three passing backends: `memory`, `postgres`, `sqlite`.

**Architecture:** Every fuel output is a row with a small state machine (`available → reserved → quarantined → spent | available`, plus `invalid`), and every transition is a compare-and-set on the row's current state (and on the reservation token for release/spend). Shards (one per split transaction) carry a lease so instances can own them without contention; the lease is never trusted for correctness. The `storetest` suite is the specification: a backend is correct when it passes the suite with `-race`.

**Tech Stack:** Go 1.26, `github.com/bsv-blockchain/go-sdk/chainhash` (32-byte tx ids), `github.com/jackc/pgx/v5` + `pgxpool`, `modernc.org/sqlite` (CGO-free), `github.com/stretchr/testify`, golangci-lint v2, docker compose (Postgres 16 for integration tests).

**Spec:** `docs/superpowers/specs/2026-09-29-fuelkeeper-design.md` (sections "Domain model", "Store interface", "Conformance suite", "Per-backend essentials").

**Model tiering for subagents (user preference):** every task below carries complete code, so implementers run on cheap models: `haiku` for Tasks 1–3 and 9 (bootstrap, types, contract, factory), `sonnet` for Tasks 4–8 (suite, memory, concurrency tests, Postgres, SQLite). Spec-compliance reviews run on `sonnet`; code-quality reviews of Tasks 4–8 (the correctness-critical store code) run on the most capable model, the rest on `sonnet`.

Conventions used throughout:
- Tx ids are `chainhash.Hash` (32 bytes, internal byte order). Databases store `hash[:]` raw bytes; `hash.String()` is the display hex.
- Times are `time.Time` in Go. Postgres stores `TIMESTAMPTZ`; SQLite stores unix microseconds as `INTEGER` with `NULL` meaning "zero time".
- Each backend takes an injectable clock (`WithClock(func() time.Time)`) used for `created_at`, `reserved_at`, `quarantined_at`, `updated_at` and lease expiry. Callers pass `now`/`before`/`until` explicitly where the contract says so.
- Integer state codes are fixed and shared by every backend: states `0 available, 1 reserved, 2 quarantined, 3 spent, 4 invalid`; shard statuses `0 unmined, 1 mined, 2 retired, 3 invalid`; lock modes `0 puzzle, 1 p2pkh`.

```go
package store

import "errors"

var (
	// ErrNotFound is returned when a shard or lease the caller named does not exist.
	ErrNotFound = errors.New("store: not found")
	// ErrNotOwner is returned by RenewShard when the shard is owned by another holder (or by nobody).
	ErrNotOwner = errors.New("store: not the shard owner")
	// ErrShardExists is returned by PutShard when a shard with the same (pool, txid)
	// exists with different size, denomination or lock mode.
	ErrShardExists = errors.New("store: shard exists with different parameters")
	// ErrTerminalShard is returned by UpdateShard when the shard is retired or invalid.
	ErrTerminalShard = errors.New("store: shard is retired or invalid")
	// ErrInvalidArgument is returned for malformed input; it is never retryable.
	ErrInvalidArgument = errors.New("store: invalid argument")
	// ErrUnavailable wraps transient backend failures (lock timeouts, connection loss).
	// Callers may retry an ErrUnavailable operation; Reserve is idempotent per (outpoint, token).
	ErrUnavailable = errors.New("store: temporarily unavailable")
```

```go
package store

import (
	"crypto/rand"
	"encoding/base64"
	"fmt"
	"strconv"
	"strings"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"
)

// Outpoint identifies one transaction output.
type Outpoint struct {
	TxID chainhash.Hash
	Vout uint32
}

// String renders "<txid-hex>:<vout>" using the display (reversed) hex form.
func (o Outpoint) String() string {
	return o.TxID.String() + ":" + strconv.FormatUint(uint64(o.Vout), 10)
}

// ParseOutpoint accepts "<txid>:<vout>" and the BRC-100 form "<txid>.<vout>".
func ParseOutpoint(s string) (Outpoint, error) {
	i := strings.LastIndexAny(s, ":.")
	if i <= 0 || i == len(s)-1 {
		return Outpoint{}, fmt.Errorf("%w: outpoint %q", ErrInvalidArgument, s)
	}
	h, err := chainhash.NewHashFromHex(s[:i])
	if err != nil {
		return Outpoint{}, fmt.Errorf("%w: outpoint txid: %w", ErrInvalidArgument, err)
	}
	v, err := strconv.ParseUint(s[i+1:], 10, 32)
	if err != nil {
		return Outpoint{}, fmt.Errorf("%w: outpoint vout: %w", ErrInvalidArgument, err)
	}
	return Outpoint{TxID: *h, Vout: uint32(v)}, nil
}

// Token is a reservation token: 16 random bytes, base64url (no padding) on the wire.
type Token [16]byte

// NewToken returns a cryptographically random token.
func NewToken() Token {
	var t Token
	if _, err := rand.Read(t[:]); err != nil {
		panic("store: crypto/rand failed: " + err.Error())
	}
	return t
}

// String encodes the token as 22 base64url characters.
func (t Token) String() string { return base64.RawURLEncoding.EncodeToString(t[:]) }

// IsZero reports whether the token is the all-zero value (meaning "no reservation").
func (t Token) IsZero() bool { return t == Token{} }

// ParseToken decodes a token produced by Token.String.
func ParseToken(s string) (Token, error) {
	b, err := base64.RawURLEncoding.DecodeString(s)
	if err != nil || len(b) != len(Token{}) {
		return Token{}, fmt.Errorf("%w: token %q", ErrInvalidArgument, s)
	}
	var t Token
	copy(t[:], b)
	return t, nil
}

// State is the lifecycle state of one fuel row. The numeric codes are part of the
// storage format and must not change.
type State uint8

const (
	StateAvailable   State = 0
	StateReserved    State = 1
	StateQuarantined State = 2
	StateSpent       State = 3
	StateInvalid     State = 4
)

func (s State) String() string {
	switch s {
	case StateAvailable:
		return "available"
	case StateReserved:
		return "reserved"
	case StateQuarantined:
		return "quarantined"
	case StateSpent:
		return "spent"
	case StateInvalid:
		return "invalid"
	}
	return "state(" + strconv.Itoa(int(s)) + ")"
}

// Terminal reports whether no transition leaves this state.
func (s State) Terminal() bool { return s == StateSpent || s == StateInvalid }

// ShardStatus is the lifecycle status of a split transaction.
type ShardStatus uint8

const (
	ShardUnmined ShardStatus = 0
	ShardMined   ShardStatus = 1
	ShardRetired ShardStatus = 2 // all rows terminal; kept for history
	ShardInvalid ShardStatus = 3 // split tx rejected or double-spent; rows invalidated
)

func (s ShardStatus) String() string {
	switch s {
	case ShardUnmined:
		return "unmined"
	case ShardMined:
		return "mined"
	case ShardRetired:
		return "retired"
	case ShardInvalid:
		return "invalid"
	}
	return "status(" + strconv.Itoa(int(s)) + ")"
}

// Terminal reports whether the shard can no longer change status.
func (s ShardStatus) Terminal() bool { return s == ShardRetired || s == ShardInvalid }

// LockMode says how a fuel output is unlocked.
type LockMode uint8

const (
	LockPuzzle LockMode = 0 // OP_SHA256 <32B> OP_EQUAL; unlocking script pushes the preimage
	LockP2PKH  LockMode = 1 // P2PKH presigned with NONE|ANYONECANPAY|FORKID
)

func (m LockMode) String() string {
	switch m {
	case LockPuzzle:
		return "puzzle"
	case LockP2PKH:
		return "p2pkh"
	}
	return "lockmode(" + strconv.Itoa(int(m)) + ")"
}

// Row is one fuel output.
type Row struct {
	Outpoint
	Pool            string
	Satoshis        uint64
	LockingScript   []byte
	UnlockingScript []byte
	State           State
	ReservedBy      Token           // zero unless reserved (kept for provenance after quarantine/spend)
	ReservedAt      time.Time       // zero unless reserved
	ExpiresAt       time.Time       // zero unless reserved
	QuarantinedAt   time.Time       // zero unless quarantined
	SpentHint       *chainhash.Hash // txid the client claims spent this row; nil if none
	SpentBy         *chainhash.Hash // oracle-proven spending txid; set only in StateSpent
	UpdatedAt       time.Time
}

// Shard is one split transaction and the lease that lets an instance own its rows.
type Shard struct {
	Pool         string
	TxID         chainhash.Hash
	Size         int
	Denomination uint64
	LockMode     LockMode
	Status       ShardStatus
	Owner        string    // "" when unowned
	LeaseUntil   time.Time // zero when unowned
	Version      int64     // starts at 1; bumped by UpdateShard and RetireShard
	BEEF         []byte    // nil in ListShards results
	CreatedAt    time.Time
	UpdatedAt    time.Time
}

// ShardHead is what RenewShard returns: enough for an owner to notice status/BEEF changes.
type ShardHead struct {
	Status  ShardStatus
	Version int64
}

// ShardFilter narrows ListShards.
type ShardFilter struct {
	Statuses []ShardStatus // nil = all
	Owner    *string       // nil = any; pointer to "" = unowned only
	Limit    int           // 0 = no limit
}

// ReserveOp asks to move one row from available to reserved under Token.
type ReserveOp struct {
	Outpoint
	Token     Token
	ExpiresAt time.Time
}

// RowCursor pages ListQuarantined: rows strictly after (At, Outpoint) in
// (quarantined_at, txid internal bytes, vout) order.
type RowCursor struct {
	At       time.Time
	Outpoint Outpoint
}

// QuarantineCursor returns the cursor positioned at this row.
func (r Row) QuarantineCursor() RowCursor { return RowCursor{At: r.QuarantinedAt, Outpoint: r.Outpoint} }

// Stats are per-pool counts.
type Stats struct {
	Available, Reserved, Quarantined, Spent, Invalid          int64
	ShardsUnmined, ShardsMined, ShardsRetired, ShardsInvalid int64
}

// Caps describes what a backend can do; the coalescer and the suite adapt to it.
type Caps struct {
	MaxConcurrentFlushes int  // how many Reserve calls may usefully run in parallel
	MaxBatch             int  // maximum ops per Reserve/Release/Spend/Resolve/GetRows call
	AtomicBatch          bool // Reserve applies all-or-nothing on backend failure
	MultiProcess         bool // several processes may share this store
}
```

```go
func hashN(n byte) chainhash.Hash {
	var h chainhash.Hash

	return h

func rows(txid chainhash.Hash, pool string, n int) []store.Row {
	out := make([]store.Row, n)
	for i := range out {
		out[i] = store.Row{
			Outpoint:        store.Outpoint{TxID: txid, Vout: uint32(i)},
			Pool:            pool,
			Satoshis:        50,
			LockingScript:   []byte{0xa8},
			UnlockingScript: []byte{0x20},
		}
	}
	return out
```

```go
		return invalid("holder must be 1..128 bytes")
	}
	return nil

		return invalid("limit must be > 0")

	return nil

	}

	if len(statuses) == 0 {
		return invalid("at least one shard status is required")

			return invalid("unknown shard status %d", s)
		}
	}
	return nil

// reachable through RetireShard.

	if s == ShardInvalid || s > ShardInvalid {
		return invalid("UpdateShard cannot set status %s", s)
	}
	return nil
}

// ValidatePutShard checks a shard and its rows before insertion.
func ValidatePutShard(shard Shard, rows []Row) error {
	if err := ValidatePool(shard.Pool); err != nil {
		return err
	}
	if shard.Size <= 0 || shard.Size != len(rows) {
		return invalid("shard size %d must equal len(rows) %d and be > 0", shard.Size, len(rows))
	}
	if shard.Denomination == 0 {
		return invalid("shard denomination must be > 0")
	}
	if shard.Status != ShardUnmined && shard.Status != ShardMined {
		return invalid("new shard status must be unmined or mined, got %s", shard.Status)
	}
	if shard.LockMode > LockP2PKH {
		return invalid("unknown lock mode %d", shard.LockMode)
	}
	seen := make(map[uint32]struct{}, len(rows))
	for i := range rows {
		r := &rows[i]
		if r.Pool != shard.Pool {
			return invalid("row %d pool %q != shard pool %q", i, r.Pool, shard.Pool)
		}
		if r.TxID != shard.TxID {
			return invalid("row %d txid != shard txid", i)
		}
		if _, dup := seen[r.Vout]; dup {
			return invalid("duplicate vout %d", r.Vout)
		}
		seen[r.Vout] = struct{}{}
		if r.Satoshis == 0 {
			return invalid("row %d satoshis must be > 0", i)
		}
		if len(r.LockingScript) == 0 || len(r.UnlockingScript) == 0 {
			return invalid("row %d needs locking and unlocking scripts", i)
		}
	}
	return nil
}

// ValidateReserveOps checks a Reserve batch: 1..maxBatch ops, distinct outpoints,
// non-zero tokens, non-zero expiry.
func ValidateReserveOps(ops []ReserveOp, maxBatch int) error {
	if len(ops) == 0 || len(ops) > maxBatch {
		return invalid("reserve batch must have 1..%d ops, got %d", maxBatch, len(ops))
	}
	seen := make(map[Outpoint]struct{}, len(ops))
	for i, op := range ops {
		if op.Token.IsZero() {
			return invalid("op %d has a zero token", i)
		}
		if op.ExpiresAt.IsZero() {
			return invalid("op %d has no expiry", i)
		}
		if _, dup := seen[op.Outpoint]; dup {
			return invalid("duplicate outpoint %s in batch", op.Outpoint)
		}
		seen[op.Outpoint] = struct{}{}
	}
	return nil
}

// ValidateOutpoints checks a Release/Spend/Resolve/GetRows batch.
func ValidateOutpoints(ops []Outpoint, maxBatch int) error {
	if len(ops) == 0 || len(ops) > maxBatch {
		return invalid("batch must have 1..%d outpoints, got %d", maxBatch, len(ops))
	}
	seen := make(map[Outpoint]struct{}, len(ops))
	for _, op := range ops {
		if _, dup := seen[op]; dup {
			return invalid("duplicate outpoint %s in batch", op)
		}
		seen[op] = struct{}{}
	}
	return nil
}

// ValidateResolve checks the Resolve target: available (spentBy ignored) or spent (spentBy required).
func ValidateResolve(to State, spentBy *chainhash.Hash) error {
	switch to {
	case StateAvailable:
		return nil
	case StateSpent:
		if spentBy == nil {
			return invalid("resolve to spent requires spentBy")
		}
		return nil
	default:
		return invalid("resolve target must be available or spent, got %s", to)
	}
}
```

```go
package store

import (
	"context"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"
)

// Store is the fuel store contract. Every implementation must pass storetest.RunSuite.
//
// Correctness model
//
// Each Row has exactly one State. Every mutation is a compare-and-set on the row's
// current state (and on ReservedBy == token for Release and Spend). Backends must
// never overwrite a row without re-asserting that predicate in the same atomic
// write. Rows that fail the predicate are silently skipped and reported through
// the returned count / applied set; they are never errors.
//
// Shards carry a lease (Owner, LeaseUntil) that lets one instance serve a shard's
// rows without contention. The lease is an optimisation only: two instances may
// race Reserve on the same row and the row CAS decides. Adoption never writes rows.
//
// Idempotency
//
// Reserve applies to a row when state == available OR (state == reserved AND
// reserved_by == op.Token). A replay under the same token is therefore reported
// as applied without changing reserved_at / expires_at. Callers retry Reserve on
// ErrUnavailable instead of compensating.
//
// Clocks
//
// Backends stamp created_at / reserved_at / quarantined_at / updated_at and lease
// expiry from their injected clock (time.Now by default). Callers pass now, before
// and until explicitly where a comparison is made, so a single clock policy holds
// across processes (see the design spec, "Restart / recovery, identity, clocks").
type Store interface {
	// Ping reports connectivity. It must not allocate or mutate.
	Ping(ctx context.Context) error
	// Capabilities is constant for the life of the store.
	Capabilities() Caps

	// PutShard inserts a shard and all its rows (state available). A shard txid is
	// globally unique across pools. It is idempotent on (pool, txid): an identical
	// replay returns nil without touching existing rows, Version or BEEF; a replay
	// with different size/denomination/lock mode, or the same txid under another
	// pool, returns ErrShardExists. Backends without multi-record atomicity write
	// rows first and the shard record last, so a visible shard implies its rows exist.
	PutShard(ctx context.Context, shard Shard, rows []Row) error
	// GetShard returns the full record including BEEF, or ErrNotFound.
	GetShard(ctx context.Context, pool string, txid chainhash.Hash) (*Shard, error)
	// UpdateShard sets status (unmined, mined or retired) and, when beef != nil,
	// replaces BEEF; it bumps Version. Retired/invalid shards return ErrTerminalShard.
	UpdateShard(ctx context.Context, pool string, txid chainhash.Hash, status ShardStatus, beef []byte) error
	// RetireShard marks the shard invalid, clears its lease, bumps Version, and moves
	// every non-terminal row (available, reserved, quarantined) to invalid, keeping
	// provenance fields (ReservedBy, SpentHint). It returns the number of rows changed.
	// A second call is a no-op (0, no Version bump). Retired shards may be retired.
	// Missing shard: ErrNotFound.
	RetireShard(ctx context.Context, pool string, txid chainhash.Hash) (int, error)
	// AcquireShard atomically claims the oldest shard (by created_at, then txid
	// compared as its 32 internal bytes) whose status is in statuses, whose lease is
	// free (owner == "" or lease_until <= now), and whose txid is not in exclude. It
	// sets Owner = holder, LeaseUntil = until (until must be after now) and returns
	// the full record. (nil, nil) means nothing was adoptable. Invariant: Owner == ""
	// iff LeaseUntil is zero.
	AcquireShard(ctx context.Context, pool, holder string, now, until time.Time, statuses []ShardStatus, exclude []chainhash.Hash) (*Shard, error)
	// RenewShard extends the lease if Owner == holder (even if it had expired and
	// nobody took it) and returns the current status and version. until must be
	// non-zero. Otherwise ErrNotOwner; missing: ErrNotFound.
	RenewShard(ctx context.Context, pool string, txid chainhash.Hash, holder string, until time.Time) (ShardHead, error)
	// ReleaseShard clears the lease if Owner == holder; otherwise it is a no-op.
	// Missing shard: ErrNotFound.
	ReleaseShard(ctx context.Context, pool string, txid chainhash.Hash, holder string) error
	// ListShards returns shards ordered by (created_at, txid internal bytes) with BEEF
	// omitted. ShardFilter.Owner matches the stored owner regardless of lease expiry.
	// Every method taking pool validates it (ErrInvalidArgument on "").
	ListShards(ctx context.Context, pool string, f ShardFilter) ([]Shard, error)

	// ListAvailable returns available rows of one shard with vout >= fromVout,
	// ordered by vout, at most limit.
	ListAvailable(ctx context.Context, pool string, txid chainhash.Hash, fromVout uint32, limit int) ([]Row, error)
	// GetRows returns the rows that exist in pool among ops, in any order. Unknown
	// outpoints are omitted.
	GetRows(ctx context.Context, pool string, ops []Outpoint) ([]Row, error)
	// Reserve performs the reservation CAS for each op and returns the applied ops
	// (ExpiresAt reflects the stored value, which on a replay is the original). Ops
	// whose row is missing, in another state, or reserved under another token are
	// omitted. Batch-level ErrInvalidArgument for duplicates, zero tokens or size.
	Reserve(ctx context.Context, pool string, ops []ReserveOp) ([]ReserveOp, error)
	// Release moves rows from reserved(token) back to available, clearing the
	// reservation fields. Returns the number of rows changed.
	Release(ctx context.Context, pool string, token Token, ops []Outpoint) (int, error)
	// Spend moves rows from reserved(token) to quarantined with SpentHint = txid, or,
	// when terminal is true (spend_policy=trust), directly to spent with SpentBy = txid.
	// Provenance rule (all backends): ReservedBy, ReservedAt, ExpiresAt, QuarantinedAt
	// and SpentHint are set when a row enters the corresponding state and kept
	// afterwards; only Release and Resolve→available clear all five. UpdatedAt is
	// stamped by every state change and not by a same-token Reserve replay.
	// Returns the number of rows changed.
	Spend(ctx context.Context, pool string, token Token, ops []Outpoint, txid chainhash.Hash, terminal bool) (int, error)
	// Quarantine moves up to limit reserved rows whose expires_at < before to
	// quarantined (oldest expiry first) and returns how many it moved.
	Quarantine(ctx context.Context, pool string, before time.Time, limit int) (int, error)
	// ListQuarantined returns quarantined rows with quarantined_at < before, ordered
	// by (quarantined_at, txid internal bytes, vout), at most limit; when after is
	// non-nil only rows strictly greater than that tuple are returned, so a stuck
	// head never starves the reaper.
	ListQuarantined(ctx context.Context, pool string, before time.Time, after *RowCursor, limit int) ([]Row, error)
	// Resolve moves quarantined rows to available (clearing reservation fields and
	// SpentHint) or to spent (setting SpentBy). Returns the number of rows changed.
	Resolve(ctx context.Context, pool string, ops []Outpoint, to State, spentBy *chainhash.Hash) (int, error)
	// Stats counts rows by state and shards by status for one pool.
	Stats(ctx context.Context, pool string) (Stats, error)

	Leaser
	// Close releases resources. It is idempotent.
	Close() error
}

// Leaser is single-runner election for background jobs. It has no fencing token:
// only gate work that is safe if two holders briefly overlap (every job write in
// FuelKeeper is a CAS, so that holds).
type Leaser interface {
	// TryAcquireOrRenew grants or extends the named lease for holder until now+ttl
	// when the lease is free, expired, or already held by holder. It returns the new
	// expiry, or the zero time when another holder owns an unexpired lease. Errors
	// are for infrastructure failures only.
	TryAcquireOrRenew(ctx context.Context, name, holder string, ttl time.Duration) (time.Time, error)
	// ReleaseLease drops the lease if holder owns it; otherwise it is a no-op.
	// (Named ReleaseLease because Store.Release is the row operation.)
	ReleaseLease(ctx context.Context, name, holder string) error
}
```

```go
package storetest

import (
	"sync"
	"time"
)

// Clock is a manual clock. Backends under test are constructed with clk.Now as
// their clock so expiry logic is deterministic.
type Clock struct {
	mu sync.Mutex
	t  time.Time
}

// NewClock starts at t.
func NewClock(t time.Time) *Clock { return &Clock{t: t} }

// Now returns the current manual time.
func (c *Clock) Now() time.Time {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.t
}

// Advance moves the clock forward by d and returns the new time.
func (c *Clock) Advance(d time.Duration) time.Time {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.t = c.t.Add(d)
	return c.t
}
```

```go
	require.NoError(t, err)
	require.NotNil(t, got)
	assert.Equal(t, s2.TxID, got.TxID)

	require.ErrorIs(t, err, store.ErrNotFound)

	require.NoError(t, s.ReleaseShard(ctx, "main", s1.TxID, "b"), "non-owner release is a no-op")
```

```go
// Package memory is the in-memory reference store used by tests and benchmarks.
// One mutex guards everything; correctness over speed.
package memory

import (
	"bytes"
	"context"
	"errors"
	"sort"
	"sync"
	"time"

	"github.com/bsv-blockchain/go-sdk/chainhash"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

var errNotImplemented = errors.New("memory: not implemented")

type shardKey struct {
	pool string
	txid chainhash.Hash
}

type lease struct {
	holder  string
	expires time.Time
}

// Store is an in-memory store.Store.
type Store struct {
	now    func() time.Time
	mu     sync.Mutex
	shards map[shardKey]*store.Shard
	rows   map[store.Outpoint]*store.Row
	leases map[string]lease
}

// Option configures New.
type Option func(*Store)

// WithClock injects the clock used for timestamps and lease expiry.
func WithClock(now func() time.Time) Option { return func(s *Store) { s.now = now } }

// New returns an empty store.
func New(opts ...Option) *Store {
	s := &Store{
		now:    time.Now,
		shards: map[shardKey]*store.Shard{},
		rows:   map[store.Outpoint]*store.Row{},
		leases: map[string]lease{},
	}
	for _, o := range opts {
		o(s)
	}
	return s
}

var _ store.Store = (*Store)(nil)

func (s *Store) Ping(context.Context) error { return nil }

func (s *Store) Capabilities() store.Caps {
	return store.Caps{MaxConcurrentFlushes: 1, MaxBatch: 1000, AtomicBatch: true, MultiProcess: false}
}

func (s *Store) Close() error { return nil }

func cloneShard(sh *store.Shard, withBEEF bool) store.Shard {
	c := *sh
	if withBEEF {
		c.BEEF = bytes.Clone(sh.BEEF)
	} else {
		c.BEEF = nil
	}
	return c
}

func cloneRow(r *store.Row) store.Row {
	c := *r
	c.LockingScript = bytes.Clone(r.LockingScript)
	c.UnlockingScript = bytes.Clone(r.UnlockingScript)
	if r.SpentHint != nil {
		h := *r.SpentHint
		c.SpentHint = &h
	}
	if r.SpentBy != nil {
		h := *r.SpentBy
		c.SpentBy = &h
	}
	return c
}

func lessShard(a, b *store.Shard) bool {
	if !a.CreatedAt.Equal(b.CreatedAt) {
		return a.CreatedAt.Before(b.CreatedAt)
	}
	return bytes.Compare(a.TxID[:], b.TxID[:]) < 0
}

// ---- shards ----

func (s *Store) PutShard(_ context.Context, shard store.Shard, rows []store.Row) error {
	if err := store.ValidatePutShard(shard, rows); err != nil {
		return err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	k := shardKey{shard.Pool, shard.TxID}
	if ex, ok := s.shards[k]; ok {
		if ex.Size != shard.Size || ex.Denomination != shard.Denomination || ex.LockMode != shard.LockMode {
			return store.ErrShardExists
		}
		return nil
	}
	now := s.now()
	for i := range rows {
		r := cloneRow(&rows[i])
		r.State = store.StateAvailable
		r.ReservedBy = store.Token{}
		r.ReservedAt, r.ExpiresAt, r.QuarantinedAt = time.Time{}, time.Time{}, time.Time{}
		r.SpentHint, r.SpentBy = nil, nil
		r.UpdatedAt = now
		if _, exists := s.rows[r.Outpoint]; !exists {
			s.rows[r.Outpoint] = &r
		}
	}
	sh := cloneShard(&shard, true)
	sh.Owner, sh.LeaseUntil, sh.Version = "", time.Time{}, 1
	sh.CreatedAt, sh.UpdatedAt = now, now
	s.shards[k] = &sh
	return nil
}

func (s *Store) GetShard(_ context.Context, pool string, txid chainhash.Hash) (*store.Shard, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	sh, ok := s.shards[shardKey{pool, txid}]
	if !ok {
		return nil, store.ErrNotFound
	}
	c := cloneShard(sh, true)
	return &c, nil
}

func (s *Store) UpdateShard(_ context.Context, pool string, txid chainhash.Hash, status store.ShardStatus, beef []byte) error {
	if err := store.ValidateUpdateStatus(status); err != nil {
		return err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	sh, ok := s.shards[shardKey{pool, txid}]
	if !ok {
		return store.ErrNotFound
	}
	if sh.Status.Terminal() {
		return store.ErrTerminalShard
	}
	sh.Status = status
	if beef != nil {
		sh.BEEF = bytes.Clone(beef)
	}
	sh.Version++
	sh.UpdatedAt = s.now()
	return nil
}

func (s *Store) RetireShard(_ context.Context, pool string, txid chainhash.Hash) (int, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	sh, ok := s.shards[shardKey{pool, txid}]
	if !ok {
		return 0, store.ErrNotFound
	}
	now := s.now()
	sh.Status = store.ShardInvalid
	sh.Owner, sh.LeaseUntil = "", time.Time{}
	sh.Version++
	sh.UpdatedAt = now
	n := 0
	for _, r := range s.rows {
		if r.Pool == pool && r.TxID == txid && !r.State.Terminal() {
			r.State = store.StateInvalid
			r.UpdatedAt = now
			n++
		}
	}
	return n, nil
}

func (s *Store) AcquireShard(_ context.Context, pool, holder string, now, until time.Time, statuses []store.ShardStatus, exclude []chainhash.Hash) (*store.Shard, error) {
	if err := store.ValidatePool(pool); err != nil {
		return nil, err
	}
	if err := store.ValidateHolder(holder); err != nil {
		return nil, err
	}
	if err := store.ValidateShardStatuses(statuses); err != nil {
		return nil, err
	}
	want := map[store.ShardStatus]bool{}
	for _, st := range statuses {
		want[st] = true
	}
	skip := map[chainhash.Hash]bool{}
	for _, h := range exclude {
		skip[h] = true
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	var best *store.Shard
	for k, sh := range s.shards {
		if k.pool != pool || !want[sh.Status] || skip[sh.TxID] {
			continue
		}
		if sh.Owner != "" && !sh.LeaseUntil.IsZero() && !sh.LeaseUntil.Before(now) {
			continue
		}
		if best == nil || lessShard(sh, best) {
			best = sh
		}
	}
	if best == nil {
		return nil, nil //nolint:nilnil // contract: (nil, nil) means nothing adoptable
	}
	best.Owner = holder
	best.LeaseUntil = until
	best.UpdatedAt = now
	c := cloneShard(best, true)
	return &c, nil
}

func (s *Store) RenewShard(_ context.Context, pool string, txid chainhash.Hash, holder string, until time.Time) (store.ShardHead, error) {
	if err := store.ValidateHolder(holder); err != nil {
		return store.ShardHead{}, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	sh, ok := s.shards[shardKey{pool, txid}]
	if !ok {
		return store.ShardHead{}, store.ErrNotFound
	}
	if sh.Owner != holder {
		return store.ShardHead{}, store.ErrNotOwner
	}
	sh.LeaseUntil = until
	return store.ShardHead{Status: sh.Status, Version: sh.Version}, nil
}

func (s *Store) ReleaseShard(_ context.Context, pool string, txid chainhash.Hash, holder string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	sh, ok := s.shards[shardKey{pool, txid}]
	if !ok {
		return store.ErrNotFound
	}
	if sh.Owner == holder {
		sh.Owner, sh.LeaseUntil = "", time.Time{}
		sh.UpdatedAt = s.now()
	}
	return nil
}

func (s *Store) ListShards(_ context.Context, pool string, f store.ShardFilter) ([]store.Shard, error) {
	want := map[store.ShardStatus]bool{}
	for _, st := range f.Statuses {
		want[st] = true
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	var out []*store.Shard
	for k, sh := range s.shards {
		if k.pool != pool {
			continue
		}
		if len(want) > 0 && !want[sh.Status] {
			continue
		}
		if f.Owner != nil && sh.Owner != *f.Owner {
			continue
		}
		out = append(out, sh)
	}
	sort.Slice(out, func(i, j int) bool { return lessShard(out[i], out[j]) })
	if f.Limit > 0 && len(out) > f.Limit {
		out = out[:f.Limit]
	}
	res := make([]store.Shard, len(out))
	for i, sh := range out {
		res[i] = cloneShard(sh, false)
	}
	return res, nil
}

func (s *Store) TryAcquireOrRenew(_ context.Context, name, holder string, ttl time.Duration) (time.Time, error) {
	if err := store.ValidateLease(name, holder, ttl); err != nil {
		return time.Time{}, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	now := s.now()
	l, ok := s.leases[name]
	if ok && l.holder != holder && l.expires.After(now) {
		return time.Time{}, nil
	}
	exp := now.Add(ttl)
	s.leases[name] = lease{holder: holder, expires: exp}
	return exp, nil
}

func (s *Store) ReleaseLease(_ context.Context, name, holder string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	if l, ok := s.leases[name]; ok && l.holder == holder {
		delete(s.leases, name)
	}
	return nil
}
```

```go
	s.mu.Lock()
	defer s.mu.Unlock()
	now := s.now()
	applied := make([]store.ReserveOp, 0, len(ops))
	for _, op := range ops {
		r, ok := s.rows[op.Outpoint]
		if !ok || r.Pool != pool {
			continue
		}
		switch {
		case r.State == store.StateAvailable:
			r.State = store.StateReserved
			r.ReservedBy = op.Token
			r.ReservedAt = now
			r.ExpiresAt = op.ExpiresAt
			r.UpdatedAt = now
			applied = append(applied, op)
		case r.State == store.StateReserved && r.ReservedBy == op.Token:
			op.ExpiresAt = r.ExpiresAt // report the stored expiry on replay
			applied = append(applied, op)
		}
	}
	return applied, nil
}

// mutateReserved applies fn to every op row in state reserved under token.
func (s *Store) mutateReserved(pool string, token store.Token, ops []store.Outpoint, fn func(r *store.Row, now time.Time)) (int, error) {
	if err := store.ValidateOutpoints(ops, s.Capabilities().MaxBatch); err != nil {
		return 0, err
	}
	if token.IsZero() {
		return 0, store.ErrInvalidArgument
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	now := s.now()
	n := 0
	for _, op := range ops {
		r, ok := s.rows[op]
		if !ok || r.Pool != pool || r.State != store.StateReserved || r.ReservedBy != token {
			continue
		}
		fn(r, now)
		r.UpdatedAt = now
		n++
	}
	return n, nil
}

func clearReservation(r *store.Row) {
	r.ReservedBy = store.Token{}
	r.ReservedAt, r.ExpiresAt, r.QuarantinedAt = time.Time{}, time.Time{}, time.Time{}
	r.SpentHint = nil
}

func (s *Store) Release(_ context.Context, pool string, token store.Token, ops []store.Outpoint) (int, error) {
	return s.mutateReserved(pool, token, ops, func(r *store.Row, _ time.Time) {
		r.State = store.StateAvailable
		clearReservation(r)
	})
}

func (s *Store) Spend(_ context.Context, pool string, token store.Token, ops []store.Outpoint, txid chainhash.Hash, terminal bool) (int, error) {
	return s.mutateReserved(pool, token, ops, func(r *store.Row, now time.Time) {
		h := txid
		if terminal {
			r.State = store.StateSpent
			r.SpentBy = &h
			return
		}
		r.State = store.StateQuarantined
		r.QuarantinedAt = now
		r.SpentHint = &h
	})
}

func (s *Store) Quarantine(_ context.Context, pool string, before time.Time, limit int) (int, error) {
	if err := store.ValidateLimit(limit); err != nil {
		return 0, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	var due []*store.Row
	for _, r := range s.rows {
		if r.Pool == pool && r.State == store.StateReserved && r.ExpiresAt.Before(before) {
			due = append(due, r)
		}
	}
	sort.Slice(due, func(i, j int) bool {
		if !due[i].ExpiresAt.Equal(due[j].ExpiresAt) {
			return due[i].ExpiresAt.Before(due[j].ExpiresAt)
		}
		if c := bytes.Compare(due[i].TxID[:], due[j].TxID[:]); c != 0 {
			return c < 0
		}
		return due[i].Vout < due[j].Vout
	})
	if len(due) > limit {
		due = due[:limit]
	}
	now := s.now()
	for _, r := range due {
		r.State = store.StateQuarantined
		r.QuarantinedAt = now
		r.UpdatedAt = now
	}
	return len(due), nil
}

func (s *Store) ListQuarantined(_ context.Context, pool string, before time.Time, limit int) ([]store.Row, error) {
	if err := store.ValidateLimit(limit); err != nil {
		return nil, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	var out []store.Row
	for _, r := range s.rows {
		if r.Pool == pool && r.State == store.StateQuarantined && r.QuarantinedAt.Before(before) {
			out = append(out, cloneRow(r))
		}
	}
	sort.Slice(out, func(i, j int) bool {
		if !out[i].QuarantinedAt.Equal(out[j].QuarantinedAt) {
			return out[i].QuarantinedAt.Before(out[j].QuarantinedAt)
		}
		if c := bytes.Compare(out[i].TxID[:], out[j].TxID[:]); c != 0 {
			return c < 0
		}
		return out[i].Vout < out[j].Vout
	})
	if len(out) > limit {
		out = out[:limit]
	}
	return out, nil
}

func (s *Store) Resolve(_ context.Context, pool string, ops []store.Outpoint, to store.State, spentBy *chainhash.Hash) (int, error) {
	if err := store.ValidateOutpoints(ops, s.Capabilities().MaxBatch); err != nil {
		return 0, err
	}
	if err := store.ValidateResolve(to, spentBy); err != nil {
		return 0, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	now := s.now()
	n := 0
	for _, op := range ops {
		r, ok := s.rows[op]
		if !ok || r.Pool != pool || r.State != store.StateQuarantined {
			continue
		}
		if to == store.StateAvailable {
			r.State = store.StateAvailable
			clearReservation(r)
		} else {
			h := *spentBy
			r.State = store.StateSpent
			r.SpentBy = &h
		}
		r.UpdatedAt = now
		n++
	}
	return n, nil
}

func (s *Store) Stats(_ context.Context, pool string) (store.Stats, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	var st store.Stats
	for _, r := range s.rows {
		if r.Pool != pool {
			continue
		}
		switch r.State {
		case store.StateAvailable:
			st.Available++
		case store.StateReserved:
			st.Reserved++
		case store.StateQuarantined:
			st.Quarantined++
		case store.StateSpent:
			st.Spent++
		case store.StateInvalid:
			st.Invalid++
		}

			continue
		}

		case store.ShardMined:
			st.ShardsMined++
		case store.ShardRetired:
			st.ShardsRetired++
		case store.ShardInvalid:
			st.ShardsInvalid++
		}
	}
	return st, nil
}
```

```go
func pick(rng *rand.Rand, rows []store.Row, n int) []store.Row {
	perm := rng.Perm(len(rows))

// testReserveExclusivityConcurrent: W workers with distinct tokens reserve random
// overlapping subsets and never release. The applied sets must partition the
// reserved rows exactly (G1), and the store must agree with the workers' view.
func testReserveExclusivityConcurrent(t *testing.T, s store.Store, clk *Clock) {
	ctx := context.Background()
	_, rows := MintShard(t, s, "main", 200, store.ShardMined)
	const workers = 8
	iters := 30 * stress()
	exp := clk.Now().Add(time.Hour)

	var mu sync.Mutex
	ownerOf := map[store.Outpoint]store.Token{}
	var wg sync.WaitGroup
	for w := 0; w < workers; w++ {
		wg.Add(1)
		go func(seed int64) {
			defer wg.Done()
			rng := rand.New(rand.NewSource(seed))
			tok := store.NewToken()
			for i := 0; i < iters; i++ {
				batch := pick(rng, rows, 10)
				applied, err := s.Reserve(ctx, "main", ReserveOps(batch, tok, exp))
				if !assert.NoError(t, err) {
					return
				}
				mu.Lock()
				for _, op := range applied {
					if prev, ok := ownerOf[op.Outpoint]; ok {
						assert.Equal(t, tok, prev, "outpoint %s applied to two tokens", op.Outpoint)
					}
					ownerOf[op.Outpoint] = tok
				}
				mu.Unlock()
			}
		}(int64(w) + 1)

	wg.Wait()

	got := RowsByOutpoint(t, s, "main", Ops(rows))

			t.Errorf("unexpected state %s for %s", r.State, op)
		}
	}

	assert.Positive(t, reserved)
}

// testReserveReleaseCycleConcurrent: workers loop reserve→release. A second
// reserve of a row may only ever follow an acknowledged release of the previous
// holder. Workers clear their claim BEFORE calling Release so the check cannot
// race the store.
func testReserveReleaseCycleConcurrent(t *testing.T, s store.Store, clk *Clock) {
	ctx := context.Background()
	_, rows := MintShard(t, s, "main", 50, store.ShardMined)
	const workers = 8
	iters := 100 * stress()
	exp := clk.Now().Add(time.Hour)

	var mu sync.Mutex
	holder := map[store.Outpoint]store.Token{}
	var wg sync.WaitGroup
	for w := 0; w < workers; w++ {
		wg.Add(1)
		go func(seed int64) {
			defer wg.Done()
			rng := rand.New(rand.NewSource(seed))
			tok := store.NewToken()
			for i := 0; i < iters; i++ {
				batch := pick(rng, rows, 5)
				applied, err := s.Reserve(ctx, "main", ReserveOps(batch, tok, exp))
				if !assert.NoError(t, err) {
					return
				}

					continue
				}
				ops := make([]store.Outpoint, len(applied))

				for j, op := range applied {
					ops[j] = op.Outpoint
					prev, held := holder[op.Outpoint]
					assert.False(t, held, "%s applied to %s while still held by %s", op.Outpoint, tok, prev)
					holder[op.Outpoint] = tok
				}
				mu.Unlock()

				mu.Lock()

					delete(holder, op)
				}
				mu.Unlock()

				if !assert.NoError(t, err) {
					return
				}
				assert.Equal(t, len(ops), n, "every applied row must release")
			}
		}(int64(w) + 100)
	}
	wg.Wait()

	st, err := s.Stats(ctx, "main")
	require.NoError(t, err)
	assert.Equal(t, int64(50), st.Available)
	assert.Equal(t, int64(0), st.Reserved)
}

// testAcquireShardExclusiveConcurrent: N goroutines adopt and release shards in
// a loop; at every instant each shard has at most one holder.
func testAcquireShardExclusiveConcurrent(t *testing.T, s store.Store, clk *Clock) {
	ctx := context.Background()
	const shards, workers = 4, 8
	iters := 50 * stress()
	ids := make([]chainhash.Hash, shards)
	for i := range ids {
		sh, _ := MintShard(t, s, "main", 1, store.ShardMined)
		ids[i] = sh.TxID
	}
	mined := []store.ShardStatus{store.ShardMined}

	var mu sync.Mutex
	owned := map[chainhash.Hash]string{}
	var wg sync.WaitGroup
	for w := 0; w < workers; w++ {
		wg.Add(1)
		go func(holder string) {
			defer wg.Done()
			for i := 0; i < iters; i++ {
				now := clk.Now()
				sh, err := s.AcquireShard(ctx, "main", holder, now, now.Add(time.Hour), mined, nil)
				if !assert.NoError(t, err) {
					return
				}
				if sh == nil {
					continue
				}
				mu.Lock()
				prev, taken := owned[sh.TxID]
				assert.False(t, taken, "shard %s acquired by %s while owned by %s", sh.TxID, holder, prev)
				owned[sh.TxID] = holder
				mu.Unlock()

				head, err := s.RenewShard(ctx, "main", sh.TxID, holder, now.Add(2*time.Hour))
				assert.NoError(t, err)
				assert.Equal(t, store.ShardMined, head.Status)

				mu.Lock()
				delete(owned, sh.TxID)
				mu.Unlock()
				assert.NoError(t, s.ReleaseShard(ctx, "main", sh.TxID, holder))
			}
		}("h" + strconv.Itoa(w))
	}
	wg.Wait()

	// Everything must be adoptable again, and each acquisition distinct.
	seen := map[chainhash.Hash]bool{}
	for i := 0; i < shards; i++ {
		sh, err := s.AcquireShard(ctx, "main", "final", clk.Now(), clk.Now().Add(time.Hour), mined, nil)
		require.NoError(t, err)
		require.NotNil(t, sh)
		assert.False(t, seen[sh.TxID])
		seen[sh.TxID] = true
	}
	sh, err := s.AcquireShard(ctx, "main", "final2", clk.Now(), clk.Now().Add(time.Hour), mined, nil)
	require.NoError(t, err)
	assert.Nil(t, sh)
}

// testReserveVsQuarantineRace: rows whose reservation has expired are released
// by the client while the reaper quarantines them. Each row ends in exactly one
// of the two states and the two counts add up to the row count.
func testReserveVsQuarantineRace(t *testing.T, s store.Store, clk *Clock) {
	ctx := context.Background()
	rounds := 5 * stress()
	for round := 0; round < rounds; round++ {
		_, rows := MintShard(t, s, "main", 40, store.ShardMined)
		tok := store.NewToken()
		MustReserve(t, s, "main", rows, tok, clk.Now().Add(time.Second))
		clk.Advance(2 * time.Second)

		var released, quarantined int
		var wg sync.WaitGroup
		wg.Add(2)
		go func() {
			defer wg.Done()
			for i := 0; i < len(rows); i += 4 {
				n, err := s.Release(ctx, "main", tok, Ops(rows[i:i+4]))
				assert.NoError(t, err)
				released += n
			}
		}()
		go func() {
			defer wg.Done()
			for {
				n, err := s.Quarantine(ctx, "main", clk.Now(), 7)
				assert.NoError(t, err)
				quarantined += n
				if n == 0 {
					return
				}
			}
		}()
		wg.Wait()

		// the reaper may have stopped early if the client released everything first;
		// drain once more so the invariant check below is exact.
		n, err := s.Quarantine(ctx, "main", clk.Now(), 100)
		require.NoError(t, err)
		quarantined += n

		got := RowsByOutpoint(t, s, "main", Ops(rows))
		var a, q int
		for _, r := range got {
			switch r.State {
			case store.StateAvailable:
				a++
			case store.StateQuarantined:
				q++
			default:
				t.Fatalf("round %d: row %s in state %s", round, r.Outpoint, r.State)
			}
		}
		assert.Equal(t, released, a, "released count matches available rows")
		assert.Equal(t, quarantined, q, "quarantined count matches quarantined rows")
		assert.Equal(t, len(rows), a+q)
	}
}
```

```sql
-- FuelKeeper fuel store schema (Postgres). Idempotent; applied under an advisory lock.
-- State codes: rows 0 available 1 reserved 2 quarantined 3 spent 4 invalid;
-- shards 0 unmined 1 mined 2 retired 3 invalid.

CREATE TABLE IF NOT EXISTS fuel_shards (
  pool         TEXT        NOT NULL,
  txid         BYTEA       NOT NULL,
  size         INTEGER     NOT NULL,
  denomination BIGINT      NOT NULL,
  lock_mode    SMALLINT    NOT NULL,
  status       SMALLINT    NOT NULL,
  owner        TEXT        NOT NULL DEFAULT '',
  lease_until  TIMESTAMPTZ NULL,
  version      BIGINT      NOT NULL DEFAULT 1,
  beef         BYTEA       NOT NULL,
  created_at   TIMESTAMPTZ NOT NULL,
  updated_at   TIMESTAMPTZ NOT NULL,
  PRIMARY KEY (pool, txid)
);
CREATE UNIQUE INDEX IF NOT EXISTS fuel_shards_txid ON fuel_shards (txid); -- a split tx belongs to one pool
CREATE INDEX IF NOT EXISTS fuel_shards_adopt ON fuel_shards (pool, status, created_at, txid);

CREATE TABLE IF NOT EXISTS fuel_rows (
  txid             BYTEA       NOT NULL,
  vout             INTEGER     NOT NULL,
  pool             TEXT        NOT NULL,
  satoshis         BIGINT      NOT NULL,
  locking_script   BYTEA       NOT NULL,
  unlocking_script BYTEA       NOT NULL,
  state            SMALLINT    NOT NULL,
  reserved_by      BYTEA       NULL,
  reserved_at      TIMESTAMPTZ NULL,
  expires_at       TIMESTAMPTZ NULL,
  quarantined_at   TIMESTAMPTZ NULL,
  spent_hint       BYTEA       NULL,
  spent_by         BYTEA       NULL,
  updated_at       TIMESTAMPTZ NOT NULL,
  PRIMARY KEY (txid, vout)
);
-- Partial indexes shrink with the live set (see the reservation review, idx_utxos_reserved lesson).
CREATE INDEX IF NOT EXISTS fuel_rows_avail      ON fuel_rows (pool, txid, vout)      WHERE state = 0;
```

```go
// Package postgres implements store.Store on PostgreSQL using pgx.
//
// Every mutation is a single statement whose WHERE clause re-asserts the row's
// current state, so concurrent writers serialise on the row and the loser simply
// matches zero rows. Batches use unnest() to pair per-op values positionally.
package postgres

import (
	"context"
	_ "embed"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgconn"
	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

//go:embed schema.sql
var schemaSQL string

// advisoryLockKey serialises schema application across replicas.
const advisoryLockKey int64 = 0x6675656c6b656570 // "fuelkeep"

const maxBatch = 1000

// Config configures New.
type Config struct {
	DSN      string
	MaxConns int32 // 0 = pgxpool default
}

// Option configures New.
type Option func(*Store)

// WithClock injects the clock used for timestamps and lease expiry.
func WithClock(now func() time.Time) Option { return func(s *Store) { s.now = now } }

// Store is a PostgreSQL store.Store.
type Store struct {
	pool *pgxpool.Pool
	now  func() time.Time
}

var _ store.Store = (*Store)(nil)

// New opens a pool, applies the schema (idempotently, under an advisory lock) and pings.
func New(ctx context.Context, cfg Config, opts ...Option) (*Store, error) {
	pc, err := pgxpool.ParseConfig(cfg.DSN)
	if err != nil {
		return nil, fmt.Errorf("postgres: parse dsn: %w", err)
	}
	if cfg.MaxConns > 0 {
		pc.MaxConns = cfg.MaxConns
	}
	pool, err := pgxpool.NewWithConfig(ctx, pc)
	if err != nil {
		return nil, fmt.Errorf("postgres: open: %w", err)
	}
	s := &Store{pool: pool, now: time.Now}
	for _, o := range opts {
		o(s)
	}
	if err := s.migrate(ctx); err != nil {
		pool.Close()
		return nil, err
	}
	return s, nil
}

func (s *Store) migrate(ctx context.Context) error {
	conn, err := s.pool.Acquire(ctx)
	if err != nil {
		return fmt.Errorf("postgres: acquire for migrate: %w", err)
	}
	defer conn.Release()
	if _, err := conn.Exec(ctx, "SELECT pg_advisory_lock($1)", advisoryLockKey); err != nil {
		return fmt.Errorf("postgres: advisory lock: %w", err)
	}
	defer func() { _, _ = conn.Exec(ctx, "SELECT pg_advisory_unlock($1)", advisoryLockKey) }()
	if _, err := conn.Exec(ctx, schemaSQL); err != nil {
		return fmt.Errorf("postgres: apply schema: %w", err)
	}
	return nil
}

// Ping checks connectivity.
func (s *Store) Ping(ctx context.Context) error { return wrap(s.pool.Ping(ctx)) }

// Capabilities reports Postgres limits.
func (s *Store) Capabilities() store.Caps {
	return store.Caps{MaxConcurrentFlushes: 8, MaxBatch: maxBatch, AtomicBatch: true, MultiProcess: true}
}

// Close closes the pool. Idempotent.
func (s *Store) Close() error {
	if s.pool != nil {
		s.pool.Close()
		s.pool = nil
	}
	return nil
}

// wrap maps transient database errors to store.ErrUnavailable and passes the rest through.
func wrap(err error) error {
	if err == nil {
		return nil
	}
	var pgErr *pgconn.PgError
	if errors.As(err, &pgErr) {
		switch pgErr.Code {
		case "40001", "40P01", "55P03", "57014", "08000", "08003", "08006", "53300":
			return fmt.Errorf("%w: %w", store.ErrUnavailable, err)
		}
		return err
	}
	if pgconn.SafeToRetry(err) || errors.Is(err, context.DeadlineExceeded) {
		return fmt.Errorf("%w: %w", store.ErrUnavailable, err)
	}
	return err
}

// tx runs fn in a transaction.
func (s *Store) tx(ctx context.Context, fn func(pgx.Tx) error) error {
	tx, err := s.pool.Begin(ctx)
	if err != nil {
		return wrap(err)
	}
	if err := fn(tx); err != nil {
		_ = tx.Rollback(ctx)
		return wrap(err)
	}
	return wrap(tx.Commit(ctx))
}

// ---- helpers shared by shards.go / rows.go ----

func utc(t *time.Time) time.Time {
	if t == nil {
		return time.Time{}
	}
	return t.UTC()
}

func nullTime(t time.Time) *time.Time {
	if t.IsZero() {
		return nil
	}
	return &t
}

func hashBytes(hs []store.Outpoint) ([][]byte, []int32) {
	txids := make([][]byte, len(hs))
	vouts := make([]int32, len(hs))
	for i, op := range hs {
		txids[i] = op.TxID[:]
		vouts[i] = int32(op.Vout) //nolint:gosec // vout is bounded by tx output count
	}
	return txids, vouts
}
```

```go
const shardCols = `pool, txid, size, denomination, lock_mode, status, owner, lease_until, version, beef, created_at, updated_at`

type rowScanner interface{ Scan(dest ...any) error }

func scanShard(sc rowScanner, withBEEF bool) (store.Shard, error) {
	var (
		sh          store.Shard
		txid        []byte
		size        int32
		denom       int64
		mode, stat  int16
		leaseUntil  *time.Time
		beef        []byte
		created, up time.Time
	)
	dest := []any{&sh.Pool, &txid, &size, &denom, &mode, &stat, &sh.Owner, &leaseUntil, &sh.Version, &beef, &created, &up}
	if err := sc.Scan(dest...); err != nil {
		return store.Shard{}, err
	}
	copy(sh.TxID[:], txid)
	sh.Size = int(size)
	sh.Denomination = uint64(denom) //nolint:gosec // stored from uint64
	sh.LockMode = store.LockMode(mode)
	sh.Status = store.ShardStatus(stat)
	sh.LeaseUntil = utc(leaseUntil)
	if withBEEF {
		sh.BEEF = beef
	}
	sh.CreatedAt, sh.UpdatedAt = created.UTC(), up.UTC()
	return sh, nil
}

func (s *Store) PutShard(ctx context.Context, shard store.Shard, rows []store.Row) error {
	if err := store.ValidatePutShard(shard, rows); err != nil {
		return err
	}
	now := s.now()
	return s.tx(ctx, func(tx pgx.Tx) error {
		tag, err := tx.Exec(ctx, `
			INSERT INTO fuel_shards (`+shardCols+`)
			VALUES ($1, $2, $3, $4, $5, $6, '', NULL, 1, $7, $8, $8)
			ON CONFLICT (pool, txid) DO NOTHING`,
			shard.Pool, shard.TxID[:], int32(shard.Size), int64(shard.Denomination), int16(shard.LockMode), int16(shard.Status), shard.BEEF, now) //nolint:gosec // validated ranges
		if err != nil {
			var pgErr *pgconn.PgError
			if errors.As(err, &pgErr) && pgErr.Code == "23505" { // fuel_shards_txid: same txid under another pool
				return store.ErrShardExists
			}
			return err
		}
		if tag.RowsAffected() == 0 {
			var size int32
			var denom int64
			var mode int16
			err := tx.QueryRow(ctx, `SELECT size, denomination, lock_mode FROM fuel_shards WHERE pool=$1 AND txid=$2`, shard.Pool, shard.TxID[:]).Scan(&size, &denom, &mode)
			if err != nil {
				return err
			}
			if int(size) != shard.Size || uint64(denom) != shard.Denomination || store.LockMode(mode) != shard.LockMode { //nolint:gosec // stored from uint64
				return store.ErrShardExists
			}
			return nil // idempotent replay
		}
		vouts := make([]int32, len(rows))
		sats := make([]int64, len(rows))
		ls := make([][]byte, len(rows))
		us := make([][]byte, len(rows))
		for i := range rows {
			vouts[i] = int32(rows[i].Vout)    //nolint:gosec // bounded
			sats[i] = int64(rows[i].Satoshis) //nolint:gosec // bounded
			ls[i] = rows[i].LockingScript
			us[i] = rows[i].UnlockingScript
		}
		_, err = tx.Exec(ctx, `
			INSERT INTO fuel_rows (txid, vout, pool, satoshis, locking_script, unlocking_script, state, updated_at)
			SELECT $1::bytea, v.vout, $2::text, v.sats, v.ls, v.us, 0, $3::timestamptz
			  FROM unnest($4::int4[], $5::int8[], $6::bytea[], $7::bytea[]) AS v(vout, sats, ls, us)
			ON CONFLICT (txid, vout) DO NOTHING`,
			shard.TxID[:], shard.Pool, now, vouts, sats, ls, us)
		return err
	})
}

func (s *Store) UpdateShard(ctx context.Context, pool string, txid chainhash.Hash, status store.ShardStatus, beef []byte) error {

		return err
	}
	tag, err := s.pool.Exec(ctx, `
		UPDATE fuel_shards
		   SET status = $3,
		       beef = COALESCE($4::bytea, beef),
		       version = version + 1,
		       updated_at = $5
		 WHERE pool = $1 AND txid = $2 AND status IN (0, 1)`, pool, txid[:], int16(status), beef, s.now())

		return wrap(err)

		return nil
	}

		return wrap(err)
	}

func (s *Store) RetireShard(ctx context.Context, pool string, txid chainhash.Hash) (int, error) {
	if err := store.ValidatePool(pool); err != nil {
		return 0, err
	}
	now := s.now()
	n := 0
	err := s.tx(ctx, func(tx pgx.Tx) error {
		// status <> 3 keeps a second retire from bumping the version; the row update
		// below still runs (and changes nothing) so the call stays idempotent.
		tag, err := tx.Exec(ctx, `
			UPDATE fuel_shards SET status=3, owner='', lease_until=NULL, version=version+1, updated_at=$3
			 WHERE pool=$1 AND txid=$2 AND status <> 3`, pool, txid[:], now)
		if err != nil {
			return err
		}
		if tag.RowsAffected() == 0 {
			var exists bool
			if err := tx.QueryRow(ctx, `SELECT EXISTS (SELECT 1 FROM fuel_shards WHERE pool=$1 AND txid=$2)`, pool, txid[:]).Scan(&exists); err != nil {
				return err
			}
			if !exists {
				return store.ErrNotFound
			}
		}
		tag, err = tx.Exec(ctx, `
			UPDATE fuel_rows SET state=4, updated_at=$3
			 WHERE pool=$1 AND txid=$2 AND state IN (0,1,2)`, pool, txid[:], now)
		if err != nil {
			return err
		}
		n = int(tag.RowsAffected())
		return nil
	})
	if errors.Is(err, store.ErrNotFound) {
		return 0, store.ErrNotFound
	}
	return n, err
}

func (s *Store) AcquireShard(ctx context.Context, pool, holder string, now, until time.Time, statuses []store.ShardStatus, exclude []chainhash.Hash) (*store.Shard, error) {
	if err := store.ValidatePool(pool); err != nil {
		return nil, err
	}
	if err := store.ValidateHolder(holder); err != nil {
		return nil, err
	}
	if err := store.ValidateShardStatuses(statuses); err != nil {
		return nil, err
	}
	st := make([]int16, len(statuses))
	for i, x := range statuses {
		st[i] = int16(x)
	}
	ex := make([][]byte, len(exclude))
	for i := range exclude {
		ex[i] = exclude[i][:]
	}
	row := s.pool.QueryRow(ctx, `
		WITH c AS (
		  SELECT pool, txid FROM fuel_shards
		   WHERE pool = $1 AND status = ANY($2::int2[])
		     AND (owner = '' OR lease_until IS NULL OR lease_until <= $3)
		     AND NOT (txid = ANY($6::bytea[]))
		   ORDER BY created_at, txid
		   LIMIT 1
		   FOR UPDATE SKIP LOCKED)
		UPDATE fuel_shards s SET owner = $4, lease_until = $5, updated_at = $7
		  FROM c
		 WHERE s.pool = c.pool AND s.txid = c.txid
		   AND (s.owner = '' OR s.lease_until IS NULL OR s.lease_until <= $3)
		RETURNING s.pool, s.txid, s.size, s.denomination, s.lock_mode, s.status, s.owner, s.lease_until, s.version, s.beef, s.created_at, s.updated_at`,
		pool, st, now, holder, until, ex, s.now())

	if errors.Is(err, pgx.ErrNoRows) {
		return nil, nil //nolint:nilnil // contract: nothing adoptable
	}
	if err != nil {
		return nil, wrap(err)
	}
	return &sh, nil
}

func (s *Store) RenewShard(ctx context.Context, pool string, txid chainhash.Hash, holder string, until time.Time) (store.ShardHead, error) {

		return store.ShardHead{}, err

	var stat int16
	var version int64
	err := s.pool.QueryRow(ctx, `

		RETURNING owner, status, version`, pool, txid[:], holder, until).Scan(&owner, &stat, &version)

		return store.ShardHead{}, store.ErrNotFound
	}
	if err != nil {
		return store.ShardHead{}, wrap(err)
	}
	if owner != holder {
		return store.ShardHead{}, store.ErrNotOwner

	}
	if tag.RowsAffected() == 0 {
```

```go
func scanRow(sc rowScanner) (store.Row, error) {

func (s *Store) Reserve(ctx context.Context, pool string, ops []store.ReserveOp) ([]store.ReserveOp, error) {
	if err := store.ValidateReserveOps(ops, maxBatch); err != nil {
		return nil, err
	}
	now := s.now()
	txids := make([][]byte, len(ops))
	vouts := make([]int32, len(ops))
	tokens := make([][]byte, len(ops))
	exps := make([]time.Time, len(ops))
	byOutpoint := make(map[store.Outpoint]int, len(ops))
	for i, op := range ops {
		txids[i] = op.TxID[:]
		vouts[i] = int32(op.Vout) //nolint:gosec // bounded
		tokens[i] = op.Token[:]
		exps[i] = op.ExpiresAt
		byOutpoint[op.Outpoint] = i
	}
	rs, err := s.pool.Query(ctx, `
		UPDATE fuel_rows r
		   SET state = 1,
		       reserved_by = k.token,
		       reserved_at = CASE WHEN r.state = 0 THEN $2 ELSE r.reserved_at END,
		       expires_at  = CASE WHEN r.state = 0 THEN k.exp ELSE r.expires_at END,
		       updated_at  = $2
		  FROM unnest($3::bytea[], $4::int4[], $5::bytea[], $6::timestamptz[]) AS k(txid, vout, token, exp)
		 WHERE r.pool = $1 AND r.txid = k.txid AND r.vout = k.vout
		   AND (r.state = 0 OR (r.state = 1 AND r.reserved_by = k.token))
		RETURNING r.txid, r.vout, r.expires_at`, pool, now, txids, vouts, tokens, exps)
	if err != nil {
		return nil, wrap(err)
	}
	defer rs.Close()
	applied := make([]store.ReserveOp, 0, len(ops))
	for rs.Next() {
		var txid []byte
		var vout int32
		var exp time.Time
		if err := rs.Scan(&txid, &vout, &exp); err != nil {
			return nil, wrap(err)
		}
		var op store.Outpoint
		copy(op.TxID[:], txid)
		op.Vout = uint32(vout) //nolint:gosec // bounded
		src := ops[byOutpoint[op]]
		src.ExpiresAt = exp.UTC()
		applied = append(applied, src)
	}
	return applied, wrap(rs.Err())
}

// updateReserved runs setClause against rows in state reserved under token.
func (s *Store) updateReserved(ctx context.Context, pool string, token store.Token, ops []store.Outpoint, setClause string, extra ...any) (int, error) {
	if err := store.ValidateOutpoints(ops, maxBatch); err != nil {
		return 0, err
	}
	if token.IsZero() {
		return 0, store.ErrInvalidArgument
	}
	txids, vouts := hashBytes(ops)
	args := append([]any{pool, token[:], txids, vouts, s.now()}, extra...)
	tag, err := s.pool.Exec(ctx, `
		UPDATE fuel_rows r SET `+setClause+`
		  FROM unnest($3::bytea[], $4::int4[]) AS k(txid, vout)
		 WHERE r.pool = $1 AND r.txid = k.txid AND r.vout = k.vout
		   AND r.state = 1 AND r.reserved_by = $2`, args...)
	if err != nil {
		return 0, wrap(err)
	}
	return int(tag.RowsAffected()), nil

func (s *Store) Release(ctx context.Context, pool string, token store.Token, ops []store.Outpoint) (int, error) {

func (s *Store) Resolve(ctx context.Context, pool string, ops []store.Outpoint, to store.State, spentBy *chainhash.Hash) (int, error) {
	if err := store.ValidateOutpoints(ops, maxBatch); err != nil {
		return 0, err
	}
	if err := store.ValidateResolve(to, spentBy); err != nil {
		return 0, err
	}
	txids, vouts := hashBytes(ops)
	var set string
	args := []any{pool, txids, vouts, s.now()}
	if to == store.StateAvailable {
		set = `state = 0, reserved_by = NULL, reserved_at = NULL, expires_at = NULL, quarantined_at = NULL, spent_hint = NULL, updated_at = $4`
	} else {
		set = `state = 3, spent_by = $5, updated_at = $4`
		args = append(args, spentBy[:])
	}
	tag, err := s.pool.Exec(ctx, `
		UPDATE fuel_rows r SET `+set+`
		  FROM unnest($2::bytea[], $3::int4[]) AS k(txid, vout)
		 WHERE r.pool = $1 AND r.txid = k.txid AND r.vout = k.vout AND r.state = 2`, args...)
	if err != nil {
		return 0, wrap(err)
	}
	return int(tag.RowsAffected()), nil
}
```

### Task 8: SQLite backend (modernc, single writer, pragma read-back)

```go
// Package sqlite implements store.Store on SQLite via modernc.org/sqlite (no CGO).
//
// One process, one writer: all mutations run on a dedicated connection inside
// BEGIN IMMEDIATE, so they serialise without SQLITE_BUSY storms. Reads use a
// separate pool; WAL lets them proceed while a write is open.
package sqlite

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"strings"
	"time"

	sqlite "modernc.org/sqlite"
	sqlite3 "modernc.org/sqlite/lib"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
)

const maxBatch = 1000

// Sync is the synchronous pragma level.
type Sync string

const (
	// SyncFull fsyncs every commit; a committed reservation survives power loss. Default.
	SyncFull Sync = "FULL"
	// SyncNormal is faster but may lose the last commits on power loss (WAL mode).
	// Opt in only when the deployment accepts that a committed reservation can revert.
	SyncNormal Sync = "NORMAL"
)

// Config configures Open.
type Config struct {
	Path string
	Sync Sync // "" = SyncFull
}

// Option configures Open.
type Option func(*Store)

// WithClock injects the clock used for timestamps and lease expiry.
func WithClock(now func() time.Time) Option { return func(s *Store) { s.now = now } }

// DSN builds the modernc DSN. Only the _pragma=name(value) form is honoured by
// modernc; mattn-style keys (_journal_mode=WAL) are silently ignored, which is
// why Open reads the pragmas back.
func DSN(path string, sync Sync) string {
	if sync == "" {
		sync = SyncFull
	}
	return "file:" + path +
		"?_pragma=busy_timeout(5000)" +
		"&_pragma=journal_mode(WAL)" +
		"&_pragma=synchronous(" + string(sync) + ")" +
		"&_pragma=foreign_keys(ON)"
}

// Store is a SQLite store.Store.
type Store struct {
	writer *sql.DB // MaxOpenConns(1)
	reader *sql.DB
	now    func() time.Time
}

var _ store.Store = (*Store)(nil)

// Pragmas is what Open verified.
type Pragmas struct {
	JournalMode string
	BusyTimeout int
	Synchronous int // 1 NORMAL, 2 FULL
}

// Open opens the database, applies the schema and verifies the pragmas.
func Open(ctx context.Context, cfg Config, opts ...Option) (*Store, error) {
	if cfg.Path == "" {
		return nil, fmt.Errorf("%w: sqlite path is required", store.ErrInvalidArgument)
	}
	dsn := DSN(cfg.Path, cfg.Sync)
	writer, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, fmt.Errorf("sqlite: open writer: %w", err)
	}
	writer.SetMaxOpenConns(1)
	writer.SetConnMaxLifetime(0)
	reader, err := sql.Open("sqlite", dsn)
	if err != nil {
		_ = writer.Close()
		return nil, fmt.Errorf("sqlite: open reader: %w", err)
	}
	reader.SetMaxOpenConns(8)
	s := &Store{writer: writer, reader: reader, now: time.Now}
	for _, o := range opts {
		o(s)
	}
	if _, err := writer.ExecContext(ctx, schemaSQL); err != nil {
		_ = s.Close()
		return nil, fmt.Errorf("sqlite: apply schema: %w", err)
	}
	p, err := s.Pragmas(ctx)
	if err != nil {
		_ = s.Close()
		return nil, err
	}
	wantSync := 2
	if cfg.Sync == SyncNormal {
		wantSync = 1
	}
	if p.JournalMode != "wal" || p.BusyTimeout != 5000 || p.Synchronous != wantSync {
		_ = s.Close()
		return nil, fmt.Errorf("sqlite: pragmas not applied (journal_mode=%q busy_timeout=%d synchronous=%d)", p.JournalMode, p.BusyTimeout, p.Synchronous)
	}
	return s, nil
}

// Pragmas reads the effective pragmas on the writer connection.
func (s *Store) Pragmas(ctx context.Context) (Pragmas, error) {
	var p Pragmas
	conn, err := s.writer.Conn(ctx)
	if err != nil {
		return p, wrap(err)
	}
	defer conn.Close()
	if err := conn.QueryRowContext(ctx, `PRAGMA journal_mode`).Scan(&p.JournalMode); err != nil {
		return p, wrap(err)
	}
	if err := conn.QueryRowContext(ctx, `PRAGMA busy_timeout`).Scan(&p.BusyTimeout); err != nil {
		return p, wrap(err)
	}
	if err := conn.QueryRowContext(ctx, `PRAGMA synchronous`).Scan(&p.Synchronous); err != nil {
		return p, wrap(err)
	}
	p.JournalMode = strings.ToLower(p.JournalMode)
	return p, nil
}

// Ping checks the reader handle.
func (s *Store) Ping(ctx context.Context) error { return wrap(s.reader.PingContext(ctx)) }

// Capabilities reports SQLite limits: one writer, batches bounded, single process.
func (s *Store) Capabilities() store.Caps {
	return store.Caps{MaxConcurrentFlushes: 1, MaxBatch: maxBatch, AtomicBatch: true, MultiProcess: false}
}

// Close closes both handles. Idempotent.
func (s *Store) Close() error {
	var errs []error
	if s.writer != nil {
		errs = append(errs, s.writer.Close())
		s.writer = nil
	}
	if s.reader != nil {
		errs = append(errs, s.reader.Close())
		s.reader = nil
	}
	return errors.Join(errs...)
}

// wrap maps SQLITE_BUSY / SQLITE_LOCKED to store.ErrUnavailable.
func wrap(err error) error {
	if err == nil {
		return nil
	}
	var se *sqlite.Error
	if errors.As(err, &se) {
		switch se.Code() & 0xff {
		case sqlite3.SQLITE_BUSY, sqlite3.SQLITE_LOCKED:
			return fmt.Errorf("%w: %w", store.ErrUnavailable, err)
		}
	}
	return err
}

// write runs fn inside BEGIN IMMEDIATE on the single writer connection, retrying
// a busy/locked begin a few times.
func (s *Store) write(ctx context.Context, fn func(*sql.Conn) error) error {
	conn, err := s.writer.Conn(ctx)
	if err != nil {
		return wrap(err)
	}
	defer conn.Close()
	var begun bool
	for attempt, backoff := 0, 20*time.Millisecond; attempt < 4; attempt, backoff = attempt+1, backoff*2 {
		if _, err = conn.ExecContext(ctx, "BEGIN IMMEDIATE"); err == nil {
			begun = true
			break
		}
		if !errors.Is(wrap(err), store.ErrUnavailable) {
			return err
		}
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(backoff):
		}
	}
	if !begun {
		return wrap(err)
	}
	if err := fn(conn); err != nil {
		_, _ = conn.ExecContext(ctx, "ROLLBACK")
		return wrap(err)
	}
	if _, err := conn.ExecContext(ctx, "COMMIT"); err != nil {
		_, _ = conn.ExecContext(ctx, "ROLLBACK")
		return wrap(err)
	}
	return nil
}

// ---- time helpers ----

func fromUS(v sql.NullInt64) time.Time {
	if !v.Valid {
		return time.Time{}
	}
	return time.UnixMicro(v.Int64).UTC()
}

func placeholders(n int) string {
	return strings.TrimSuffix(strings.Repeat("?,", n), ",")
}
```

```go
		leaseUntil            sql.NullInt64
		created, updated      int64
	)
	if err := sc.Scan(&sh.Pool, &txid, &sh.Size, &denom, &mode, &stat, &sh.Owner, &leaseUntil, &sh.Version, &beef, &created, &updated); err != nil {
		return store.Shard{}, err
	}
	copy(sh.TxID[:], txid)
	sh.Denomination = uint64(denom) //nolint:gosec // stored from uint64
	sh.LockMode = store.LockMode(mode)   //nolint:gosec // small enum
	sh.Status = store.ShardStatus(stat)  //nolint:gosec // small enum
	sh.LeaseUntil = fromUS(leaseUntil)
	if withBEEF {
		sh.BEEF = beef
	}
	sh.CreatedAt = time.UnixMicro(created).UTC()
	sh.UpdatedAt = time.UnixMicro(updated).UTC()
	return sh, nil
}

func (s *Store) PutShard(ctx context.Context, shard store.Shard, rows []store.Row) error {
	if err := store.ValidatePutShard(shard, rows); err != nil {
		return err
	}
	now := s.now().UnixMicro()
	return s.write(ctx, func(c *sql.Conn) error {
		var pool string
		var size, mode int
		var denom int64
		// look up by txid alone: a split tx belongs to exactly one pool
		err := c.QueryRowContext(ctx, `SELECT pool, size, denomination, lock_mode FROM fuel_shards WHERE txid=?`, shard.TxID[:]).Scan(&pool, &size, &denom, &mode)
		switch {
		case err == nil:
			if pool != shard.Pool || size != shard.Size || uint64(denom) != shard.Denomination || store.LockMode(mode) != shard.LockMode { //nolint:gosec // stored from uint64
				return store.ErrShardExists
			}
			return nil
		case !errors.Is(err, sql.ErrNoRows):
			return err
		}
		if _, err := c.ExecContext(ctx, `INSERT INTO fuel_shards (`+shardCols+`) VALUES (?,?,?,?,?,?,'',NULL,1,?,?,?)`,
			shard.Pool, shard.TxID[:], shard.Size, int64(shard.Denomination), int(shard.LockMode), int(shard.Status), shard.BEEF, now, now); err != nil { //nolint:gosec // validated
			return err
		}
		stmt, err := c.PrepareContext(ctx, `INSERT OR IGNORE INTO fuel_rows (txid, vout, pool, satoshis, locking_script, unlocking_script, state, updated_at) VALUES (?,?,?,?,?,?,0,?)`)
		if err != nil {
			return err
		}
		defer stmt.Close()
		for i := range rows {
			r := &rows[i]
			if _, err := stmt.ExecContext(ctx, r.TxID[:], r.Vout, r.Pool, int64(r.Satoshis), r.LockingScript, r.UnlockingScript, now); err != nil { //nolint:gosec // validated
				return err
			}
		}
		return nil
	})
}

func (s *Store) AcquireShard(ctx context.Context, pool, holder string, now, until time.Time, statuses []store.ShardStatus, exclude []chainhash.Hash) (*store.Shard, error) {

		return nil, err

		return nil, err
	}
	args := []any{holder, until.UnixMicro(), s.now().UnixMicro(), pool} // updated_at from the store clock
	for _, st := range statuses {
		args = append(args, int(st))
	}
	q := `UPDATE fuel_shards SET owner=?, lease_until=?, updated_at=? WHERE rowid = (
	        SELECT rowid FROM fuel_shards
	         WHERE pool=? AND status IN (` + placeholders(len(statuses)) + `)
	           AND (owner='' OR lease_until IS NULL OR lease_until <= ?)`
	args = append(args, now.UnixMicro())
	if len(exclude) > 0 {
		q += ` AND txid NOT IN (` + placeholders(len(exclude)) + `)`
		for i := range exclude {
			args = append(args, exclude[i][:])
		}
	}
	q += ` ORDER BY created_at, txid LIMIT 1) RETURNING ` + shardCols
	var out *store.Shard
	err := s.write(ctx, func(c *sql.Conn) error {
		sh, err := scanShard(c.QueryRowContext(ctx, q, args...), true)
		if errors.Is(err, sql.ErrNoRows) {
			return nil
		}
		if err != nil {
			return err
		}
		out = &sh
		return nil
	})
	return out, err
```

```go
func (s *Store) TryAcquireOrRenew(ctx context.Context, name, holder string, ttl time.Duration) (time.Time, error) {
	if err := store.ValidateLease(name, holder, ttl); err != nil {
		return time.Time{}, err
	}

	exp := now.Add(ttl)
	var got time.Time

		var curHolder string
		var curExp int64
		err := c.QueryRowContext(ctx, `SELECT holder, expires_at FROM fuel_leases WHERE name = ?`, name).Scan(&curHolder, &curExp)
		switch {
		case errors.Is(err, sql.ErrNoRows):
			_, err = c.ExecContext(ctx, `INSERT INTO fuel_leases (name, holder, expires_at) VALUES (?, ?, ?)`, name, holder, exp.UnixMicro())
		case err != nil:
			return err
		case curHolder == holder || curExp <= now.UnixMicro():
			_, err = c.ExecContext(ctx, `UPDATE fuel_leases SET holder = ?, expires_at = ? WHERE name = ?`, holder, exp.UnixMicro(), name)
		default:
			return nil // lost
		}
		if err != nil {
			return err
		}
		got = exp
		return nil
	})
	return got, err
}

	return s.write(ctx, func(c *sql.Conn) error {
		_, err := c.ExecContext(ctx, `DELETE FROM fuel_leases WHERE name = ? AND holder = ?`, name, holder)
```

```go
func (s *Store) Analyze(ctx context.Context) error {

// ExplainQueryPlan returns the detail lines of EXPLAIN QUERY PLAN for q.
func (s *Store) ExplainQueryPlan(ctx context.Context, q string, args ...any) ([]string, error) {
	rs, err := s.reader.QueryContext(ctx, "EXPLAIN QUERY PLAN "+q, args...)
	if err != nil {
		return nil, wrap(err)
	}
	defer rs.Close()
	var out []string
	for rs.Next() {
		var id, parent, notused int
		var detail string
		if err := rs.Scan(&id, &parent, &notused, &detail); err != nil {
			return nil, wrap(err)
		}
		out = append(out, detail)
	}
	return out, wrap(rs.Err())
}
```

```go
// Package factory builds a store.Store from configuration.
package factory

import (
	"context"
	"fmt"
	"time"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"
	"github.com/bsv-blockchain/fuelkeeper/pkg/store/memory"
	"github.com/bsv-blockchain/fuelkeeper/pkg/store/postgres"
	"github.com/bsv-blockchain/fuelkeeper/pkg/store/sqlite"
)

// Backend names a store implementation.
type Backend string

const (
	BackendMemory   Backend = "memory"
	BackendSQLite   Backend = "sqlite"
	BackendPostgres Backend = "postgres"
	// Plan 4 adds: pebble, mongodb, aerospike.
)

// Config selects and configures a backend.
type Config struct {
	Backend  Backend         `yaml:"backend"`
	SQLite   sqlite.Config   `yaml:"sqlite"`
	Postgres postgres.Config `yaml:"postgres"`
}

// New opens the configured backend with the given clock.
func New(ctx context.Context, cfg Config, now func() time.Time) (store.Store, error) {
	if now == nil {
		now = time.Now
	}
	switch cfg.Backend {
	case BackendMemory:
		return memory.New(memory.WithClock(now)), nil
	case BackendSQLite:
		return sqlite.Open(ctx, cfg.SQLite, sqlite.WithClock(now))
	case BackendPostgres:
		if cfg.Postgres.DSN == "" {
			return nil, fmt.Errorf("%w: postgres.dsn is required", store.ErrInvalidArgument)
		}
		return postgres.New(ctx, cfg.Postgres, postgres.WithClock(now))
	default:
		return nil, fmt.Errorf("%w: unknown store backend %q", store.ErrInvalidArgument, cfg.Backend)
	}
}
```
