# FuelKeeper Plan 5: Treasury, funding and chain oracles Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the mocks with the real world: a go-wallet-toolbox treasury that receives BRC-29 payments from any BRC-100 wallet and mints split transactions (hash-puzzle or presigned P2PKH), arcade and teranode oracles, the funding endpoints, and a `fuelkeeper fund` command that drives the user's local wallet. End state: FuelKeeper runs against teratestnet with real fuel.

**Architecture:** The treasury is split into a small pure core (`Build` from a createAction result, preimage/presign generation, chunk sizing) that is unit-tested against a fake wallet interface, and a thin wiring layer that constructs the toolbox wallet, storage, services and monitor. Splits use the toolbox's throughput UTXO strategy for two-level chunking, so a leaf split is a plain createAction funded by one exact-match claim. Broadcast is separated from build (`NoSend` + `SendWith`) so the minter can record the shard before anything reaches the network. Oracles are thin HTTP clients over documented arcade and teranode endpoints.

**Tech Stack:** `github.com/bsv-blockchain/go-wallet-toolbox v0.184.16` (go-sdk v1.4.1), go-sdk `script/interpreter` (presign verification), `transaction/template/p2pkh`, `wallet/substrates` (HTTP wallet JSON client), `auth/utils` (nonces), stdlib `net/http`, testify.

**Spec:** `docs/superpowers/specs/2026-09-29-fuelkeeper-design.md` — "Treasury", "Chain oracle", "CLI (fund)", the protocol facts at the top. Verified toolbox API facts are cited inline with file paths under `~/go/pkg/mod/github.com/bsv-blockchain/go-wallet-toolbox@v0.184.16` (abbreviated `TB`).

**Model tiering (user preference):** Tasks 1, 2, 3 (presign, arcade client, oracles) on `sonnet`; Task 4 (treasury core) and Task 5 (toolbox wiring + funding) on the most capable model, because they encode protocol behaviour and toolbox integration facts; Tasks 6–7 (API/CLI/config wiring, docs) on `sonnet`. ### Task 1: Presigned P2PKH fuel (`pkg/fuel/p2pkh.go`)

A fuel output presigned with `SIGHASH_NONE|ANYONECANPAY|FORKID` (0xC2) is valid in any transaction that uses version 1, lockTime 0 and sequence 0xffffffff on that input. The test proves both halves with go-sdk's script interpreter: a foreign transaction with different inputs/outputs verifies; changing version, lockTime or sequence breaks it.

```go
	tx.LockTime = lockTime
	// an unrelated first input so the fuel input is not at index 0

	prev := &transaction.TransactionOutput{Satoshis: sats, LockingScript: script.NewFromBytes(lock)}
	verify := func(tx *transaction.Transaction, idx int) error {
		return interpreter.NewEngine().Execute(interpreter.WithTx(tx, idx, prev), interpreter.WithForkID(), interpreter.WithAfterGenesis())
	}

	tx, idx := consumer(t, fuel.P2PKHTxVersion, fuel.P2PKHLockTime, fuel.P2PKHSequence, [32]byte(txid), vout, unlock, 2)
	require.NoError(t, verify(tx, idx), "different inputs and outputs than the presign tx still verify")
	tx, idx = consumer(t, fuel.P2PKHTxVersion, fuel.P2PKHLockTime, fuel.P2PKHSequence, [32]byte(txid), vout, unlock, 5)
	require.NoError(t, verify(tx, idx))

	tx, idx = consumer(t, 2, fuel.P2PKHLockTime, fuel.P2PKHSequence, [32]byte(txid), vout, unlock, 2)
	require.Error(t, verify(tx, idx), "version is committed")
	tx, idx = consumer(t, fuel.P2PKHTxVersion, 1, fuel.P2PKHSequence, [32]byte(txid), vout, unlock, 2)
	require.Error(t, verify(tx, idx), "lockTime is committed")
	tx, idx = consumer(t, fuel.P2PKHTxVersion, fuel.P2PKHLockTime, 0xfffffffe, [32]byte(txid), vout, unlock, 2)
	require.Error(t, verify(tx, idx), "this input's sequence is committed")

	wrongPrev := &transaction.TransactionOutput{Satoshis: sats + 1, LockingScript: script.NewFromBytes(lock)}
	tx, idx = consumer(t, fuel.P2PKHTxVersion, fuel.P2PKHLockTime, fuel.P2PKHSequence, [32]byte(txid), vout, unlock, 2)
	require.Error(t, interpreter.NewEngine().Execute(interpreter.WithTx(tx, idx, wrongPrev), interpreter.WithForkID(), interpreter.WithAfterGenesis()), "value is committed")
```

```go
package fuel

import (
	"fmt"

	"github.com/bsv-blockchain/go-sdk/chainhash"
	ec "github.com/bsv-blockchain/go-sdk/primitives/ec"
	"github.com/bsv-blockchain/go-sdk/script"
	"github.com/bsv-blockchain/go-sdk/transaction"
	"github.com/bsv-blockchain/go-sdk/transaction/sighash"
	"github.com/bsv-blockchain/go-sdk/transaction/template/p2pkh"
)

// Constraints a consuming transaction must satisfy for a presigned P2PKH fuel
// input: the FORKID sighash always commits to nVersion, this input's nSequence
// and nLockTime, even with NONE|ANYONECANPAY.
const (
	P2PKHTxVersion uint32 = 1
	P2PKHLockTime  uint32 = 0
	P2PKHSequence  uint32 = 0xffffffff
)

// P2PKHSigHash is NONE|ANYONECANPAY|FORKID (0xC2).
var P2PKHSigHash = sighash.NoneForkID | sighash.AnyOneCanPay

// P2PKHLock returns the P2PKH locking script for pub (network-independent bytes).
func P2PKHLock(pub *ec.PublicKey) ([]byte, error) {
	addr, err := script.NewAddressFromPublicKey(pub, true)
	if err != nil {
		return nil, fmt.Errorf("fuel: address: %w", err)
	}
	s, err := p2pkh.Lock(addr)
	if err != nil {
		return nil, fmt.Errorf("fuel: p2pkh lock: %w", err)
	}
	return *s, nil
}

// P2PKHPresign produces the unlocking script for a fuel output so that any
// transaction honouring P2PKHTxVersion/P2PKHLockTime/P2PKHSequence can spend it.
func P2PKHPresign(priv *ec.PrivateKey, txid chainhash.Hash, vout uint32, lockingScript []byte, satoshis uint64) ([]byte, error) {
	tx := transaction.NewTransaction()
	tx.Version = P2PKHTxVersion
	tx.LockTime = P2PKHLockTime
	src := txid
	in := &transaction.TransactionInput{SourceTXID: &src, SourceTxOutIndex: vout, SequenceNumber: P2PKHSequence}
	in.SetSourceTxOutput(&transaction.TransactionOutput{Satoshis: satoshis, LockingScript: script.NewFromBytes(lockingScript)})
	tx.AddInput(in)
	flag := P2PKHSigHash
	tmpl, err := p2pkh.Unlock(priv, &flag)
	if err != nil {
		return nil, fmt.Errorf("fuel: unlock template: %w", err)
	}
	unlock, err := tmpl.Sign(tx, 0)
	if err != nil {
		return nil, fmt.Errorf("fuel: presign: %w", err)
	}
	return *unlock, nil
}
```

```go
// Package arcadeclient is a minimal client for the arcade transaction gateway.
package arcadeclient

import (
	"bytes"
	"context"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"strconv"
	"strings"
	"time"
)

// ErrNotFound: arcade has no verdict for this txid (HTTP 404).
var ErrNotFound = errors.New("arcade: transaction not found")

// RejectError is a 4xx validation response for POST /tx.
type RejectError struct {
	Status int    `json:"status"`
	Title  string `json:"title"`
	Detail string `json:"detail"`
	TxID   string `json:"txid"`
}

func (e *RejectError) Error() string {
	return fmt.Sprintf("arcade rejected tx %s: %d %s: %s", e.TxID, e.Status, e.Title, e.Detail)
}

// BackpressureError is a 503 with Retry-After.
type BackpressureError struct{ RetryAfter int }

func (e *BackpressureError) Error() string { return fmt.Sprintf("arcade busy; retry after %ds", e.RetryAfter) }

// TxStatus is GET /tx/{txid}.
type TxStatus struct {
	TxID         string   `json:"txid"`
	TxStatus     string   `json:"txStatus"`
	Status       int      `json:"status"`
	BlockHash    string   `json:"blockHash"`
	BlockHeight  uint32   `json:"blockHeight"`
	MerklePath   string   `json:"merklePath"`
	ExtraInfo    string   `json:"extraInfo"`
	CompetingTxs []string `json:"competingTxs"`
}

// Mined reports MINED or IMMUTABLE.
func (s *TxStatus) Mined() bool { return s.TxStatus == "MINED" || s.TxStatus == "IMMUTABLE" }

// Rejected reports a terminal REJECTED verdict.
func (s *TxStatus) Rejected() bool { return s.TxStatus == "REJECTED" }

// Conflicting reports DOUBLE_SPEND_ATTEMPTED.
func (s *TxStatus) Conflicting() bool { return s.TxStatus == "DOUBLE_SPEND_ATTEMPTED" }

// MerklePathBytes decodes the BUMP hex (nil when absent or malformed).
func (s *TxStatus) MerklePathBytes() []byte {
	if s.MerklePath == "" {
		return nil
	}
	b, err := hex.DecodeString(s.MerklePath)
	if err != nil {
		return nil
	}
	return b
}

// SubmitResponse is POST /tx's 202 body.
type SubmitResponse struct {
	TxID     string `json:"txid"`
	Status   int    `json:"status"`
	TxStatus string `json:"txStatus"`
}

// BatchSubmitResponse is POST /txs's body.
type BatchSubmitResponse struct {
	Submitted  int `json:"submitted"`
	Duplicates int `json:"duplicates"`
	Total      int `json:"total"`
}

// Policy is GET /policy.
type Policy struct {
	MiningFee struct {
		Satoshis uint64 `json:"satoshis"`
		Bytes    uint64 `json:"bytes"`
	} `json:"miningFee"`
	MaxTxSizePolicy uint64 `json:"maxtxsizepolicy"`
}

// SatPerKB converts the fee quote to sat/kB (ceil).
func (p *Policy) SatPerKB() uint64 {
	if p.MiningFee.Bytes == 0 {
		return 0
	}
	return (p.MiningFee.Satoshis*1000 + p.MiningFee.Bytes - 1) / p.MiningFee.Bytes
}

// Option configures New.
type Option func(*Client)

// WithCallbackToken sets X-CallbackToken on submissions.
func WithCallbackToken(tok string) Option { return func(c *Client) { c.callbackToken = tok } }

// WithHTTPClient replaces the HTTP client.
func WithHTTPClient(h *http.Client) Option { return func(c *Client) { c.http = h } }

// Client talks to one arcade base URL (e.g. https://arcade.example/v1).
type Client struct {
	base          string
	http          *http.Client
	callbackToken string
}

// New validates the base URL.
func New(base string, opts ...Option) (*Client, error) {
	u, err := url.Parse(base)
	if err != nil || u.Scheme == "" || u.Host == "" {
		return nil, fmt.Errorf("arcade: invalid base url %q", base)
	}
	c := &Client{base: strings.TrimRight(base, "/"), http: &http.Client{Timeout: 15 * time.Second}}
	for _, o := range opts {
		o(c)
	}
	return c, nil
}

func (c *Client) do(ctx context.Context, method, path string, contentType string, body []byte) (*http.Response, error) {
	var rd io.Reader
	if body != nil {
		rd = bytes.NewReader(body)
	}
	req, err := http.NewRequestWithContext(ctx, method, c.base+path, rd)
	if err != nil {
		return nil, err
	}
	if contentType != "" {
		req.Header.Set("Content-Type", contentType)
	}
	if c.callbackToken != "" {
		req.Header.Set("X-CallbackToken", c.callbackToken)
	}
	return c.http.Do(req)
}

// GetTx returns arcade's verdict, or ErrNotFound.
func (c *Client) GetTx(ctx context.Context, txid string) (*TxStatus, error) {
	resp, err := c.do(ctx, http.MethodGet, "/tx/"+txid, "", nil)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode == http.StatusNotFound {
		return nil, ErrNotFound
	}
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("arcade: GET /tx/%s: %s", txid, resp.Status)
	}
	var st TxStatus
	if err := json.NewDecoder(resp.Body).Decode(&st); err != nil {
		return nil, fmt.Errorf("arcade: decode status: %w", err)
	}
	return &st, nil
}

// PostTx submits one transaction (raw or extended format bytes).
func (c *Client) PostTx(ctx context.Context, tx []byte) (*SubmitResponse, error) {
	resp, err := c.do(ctx, http.MethodPost, "/tx", "application/octet-stream", tx)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	switch {
	case resp.StatusCode == http.StatusOK || resp.StatusCode == http.StatusAccepted:
		var sr SubmitResponse
		if err := json.NewDecoder(resp.Body).Decode(&sr); err != nil {
			return nil, fmt.Errorf("arcade: decode submit: %w", err)
		}
		return &sr, nil
	case resp.StatusCode == http.StatusServiceUnavailable || resp.StatusCode == http.StatusTooManyRequests:
		ra, _ := strconv.Atoi(resp.Header.Get("Retry-After"))
		if ra <= 0 {
			ra = 1
		}
		return nil, &BackpressureError{RetryAfter: ra}
	case resp.StatusCode >= 400 && resp.StatusCode < 500:
		var re RejectError
		_ = json.NewDecoder(io.LimitReader(resp.Body, 64<<10)).Decode(&re)
		if re.Status == 0 {
			re.Status = resp.StatusCode
		}
		return nil, &re
	default:
		return nil, fmt.Errorf("arcade: POST /tx: %s", resp.Status)
	}
}

// PostTxs submits independent raw transactions in one request (all-or-nothing).
func (c *Client) PostTxs(ctx context.Context, rawTxs [][]byte) (*BatchSubmitResponse, error) {
	body := bytes.Join(rawTxs, nil)
	resp, err := c.do(ctx, http.MethodPost, "/txs", "application/octet-stream", body)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK && resp.StatusCode != http.StatusAccepted {
		if resp.StatusCode == http.StatusServiceUnavailable {
			ra, _ := strconv.Atoi(resp.Header.Get("Retry-After"))
			return nil, &BackpressureError{RetryAfter: max(ra, 1)}
		}
		var re RejectError
		_ = json.NewDecoder(io.LimitReader(resp.Body, 64<<10)).Decode(&re)
		if re.Status == 0 {
			re.Status = resp.StatusCode
		}
		return nil, &re
	}
	var br BatchSubmitResponse
	if err := json.NewDecoder(resp.Body).Decode(&br); err != nil {
		return nil, fmt.Errorf("arcade: decode batch submit: %w", err)
	}
	return &br, nil
}

// Policy fetches the fee and size policy.
```

```go
package teranode

import (
	"context"

	"github.com/bsv-blockchain/fuelkeeper/pkg/store"

// Oracle queries GET {base}/utxo/{txid}/json?vout=N.
type Oracle struct {
	base        string
	http        *http.Client
	concurrency int
}

var _ oracle.Oracle = (*Oracle)(nil)

// New validates base (e.g. http://teranode:8090/api/v1).
func New(base string) (*Oracle, error) {
	u, err := url.Parse(base)
	if err != nil || u.Scheme == "" || u.Host == "" {
		return nil, fmt.Errorf("teranode: invalid base url %q", base)
	}
	return &Oracle{base: strings.TrimRight(base, "/"), http: &http.Client{Timeout: 10 * time.Second}, concurrency: 16}, nil
}

type spendResponse struct {
	Status       int `json:"status"`
	SpendingData *struct {
		TxID string `json:"txId"`
		Vin  int    `json:"vin"`
	} `json:"spendingData"`
}

func (o *Oracle) one(ctx context.Context, op store.Outpoint) (oracle.Result, error) {
	res := oracle.Result{Outpoint: op, Status: oracle.StatusUnknown}
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, o.base+"/utxo/"+op.TxID.String()+"/json?vout="+strconv.FormatUint(uint64(op.Vout), 10), nil)
	if err != nil {
		return res, err
	}
	resp, err := o.http.Do(req)
	if err != nil {
		return res, fmt.Errorf("%w: %w", oracle.ErrUnavailable, err)
	}
	defer resp.Body.Close()
	switch {
	case resp.StatusCode == http.StatusNotFound:
		return res, nil
	case resp.StatusCode >= 500:
		return res, fmt.Errorf("%w: teranode %s", oracle.ErrUnavailable, resp.Status)
	case resp.StatusCode != http.StatusOK:
		return res, fmt.Errorf("teranode: %s", resp.Status)
	}
	var sr spendResponse
	if err := json.NewDecoder(resp.Body).Decode(&sr); err != nil {
		return res, fmt.Errorf("%w: decode: %w", oracle.ErrUnavailable, err)
	}
	switch sr.Status {
	case 0, 2: // OK, IMMATURE
		res.Status = oracle.StatusUnspent
	case 1: // SPENT
		res.Status = oracle.StatusSpent
		if sr.SpendingData != nil {
			if h, err := chainhash.NewHashFromHex(sr.SpendingData.TxID); err == nil {
				res.SpendingTxID = h
			}
		}
	case 5, 6: // FROZEN, CONFLICTING
		res.Status = oracle.StatusConflicting
	default: // NOT_FOUND, LOCKED, unknown codes
		res.Status = oracle.StatusUnknown
	}
	return res, nil
}

// Outpoints resolves each outpoint with bounded concurrency, preserving order.
func (o *Oracle) Outpoints(ctx context.Context, ops []store.Outpoint) ([]oracle.Result, error) {
	out := make([]oracle.Result, len(ops))
	g, gctx := errgroup.WithContext(ctx)
	g.SetLimit(o.concurrency)
	for i, op := range ops {
		g.Go(func() error {
			r, err := o.one(gctx, op)
			out[i] = r
			return err
		})
	}
	if err := g.Wait(); err != nil {
		if errors.Is(err, oracle.ErrUnavailable) {
			return nil, err
		}
		return nil, fmt.Errorf("%w: %w", oracle.ErrUnavailable, err)
	}
	return out, nil
}

// Tx is not served by this oracle (use the composite with arcade).
func (o *Oracle) Tx(context.Context, chainhash.Hash) (oracle.TxStatus, error) { return oracle.TxStatus{}, nil }
```

```go
// Validate checks the config against the pools it must fund.
func (c *Config) Validate(pools []pool.PoolConfig) error {
	switch c.Network {
	case "main", "test", "ttn", "tstn":
	default:
		return fmt.Errorf("treasury.toolbox.network %q must be main|test|ttn|tstn", c.Network)
	}
	if len(c.RootKeyHex) != 64 {
		return errors.New("treasury.toolbox.root_key_hex must be 32 bytes hex")
	}
	if c.ArcadeURL == "" || c.ChaintracksURL == "" {
		return errors.New("treasury.toolbox.arcade_url and chaintracks_url are required")
	}
	switch c.Storage.Engine {
	case "sqlite":
		if c.Storage.SQLite == "" {
			return errors.New("treasury.toolbox.storage.sqlite_path is required")
		}
	case "postgres":
		if c.Storage.Postgres.Host == "" || c.Storage.Postgres.DBName == "" {
			return errors.New("treasury.toolbox.storage.postgres host and dbname are required")
		}
	default:
		return fmt.Errorf("treasury.toolbox.storage.engine %q must be sqlite|postgres", c.Storage.Engine)
	}
	if c.Chunking.FanoutOutputsPerTx <= 0 || c.Chunking.TargetChunks <= 0 || c.Chunking.LowWaterPercent <= 0 || c.Chunking.LowWaterPercent > 100 {
		return errors.New("treasury.toolbox.chunking values must be positive (low_water_percent 1..100)")
	}
	if c.Chunking.ReserveBasket == "" || c.Chunking.ChunkBasket == "" || c.Chunking.ReserveBasket == c.Chunking.ChunkBasket || c.Chunking.ReserveBasket == "default" || c.Chunking.ChunkBasket == "default" {
		return errors.New("treasury.toolbox.chunking baskets must be distinct and not 'default'")
	}
	if len(pools) == 0 {
		return errors.New("treasury needs at least one pool")
	}
	first := LeafCost(pools[0], c.FeeSatPerKB)
	for _, p := range pools[1:] {
		if LeafCost(p, c.FeeSatPerKB) != first {
			return fmt.Errorf("treasury.toolbox: pools %s and %s have different leaf costs; one treasury funds one leaf cost class (run one FuelKeeper per class)", pools[0].Name, p.Name)
		}
	}
	return nil
}

// LeafSizeBytes estimates a leaf split: one P2PKH input, n outputs, one change output.
func LeafSizeBytes(p pool.PoolConfig) int {
	outSize := 9 + 35 // varint+satoshis+script for a 35-byte puzzle lock
	if p.LockMode == 1 { // store.LockP2PKH
		outSize = 9 + 25
	}
	return 10 + fuel.P2PKHInputSize + p.SplitSize*outSize + 34
}

// LeafCost is what one leaf split needs from its funding chunk (outputs + fee).
func LeafCost(p pool.PoolConfig, satPerKB uint64) uint64 {
	if satPerKB == 0 {
		satPerKB = 100
	}
	return uint64(p.SplitSize)*p.Denomination + fuel.FeeSats(LeafSizeBytes(p), satPerKB) //nolint:gosec // bounded
}

// ChunkValue is the exact-denomination coin the toolbox keeps ready for leaves.
func (c *Config) ChunkValue(p pool.PoolConfig, satPerKB uint64) uint64 {
	return LeafCost(p, satPerKB) + c.Chunking.HeadroomSats
}
```

```go
	sent bool

type Core struct {
	w      WalletAPI
	bc     Broadcaster

	nonces Nonces

// NewCore wires the core over an already-constructed wallet.
func NewCore(w WalletAPI, bc Broadcaster, keeper ChunkKeeper, nonces Nonces, kd *sdk.KeyDeriver, cfg Config, feeSatPerKB uint64, log *slog.Logger) *Core {
	return &Core{w: w, bc: bc, keeper: keeper, nonces: nonces, kd: kd, cfg: cfg, feeKB: feeSatPerKB, log: log.With("component", "treasury"), built: map[chainhash.Hash]*builtInfo{}}

func ptr[T any](v T) *T { return &v }

// Build creates and signs a split without broadcasting (NoSend).
func (c *Core) Build(ctx context.Context, req treasury.MintRequest) (*treasury.BuiltShard, error) {
	if err := req.Validate(); err != nil {
		return nil, err
	}
	outputs := make([]sdk.CreateActionOutput, req.Count)
	var seed fuel.Seed
	var p2pkhLock []byte
	switch req.LockMode {
	case store.LockPuzzle:
		seed = fuel.NewSeed()
		for i := range outputs {
			pre := fuel.Preimage(seed, uint32(i)) //nolint:gosec // count is bounded
			outputs[i] = sdk.CreateActionOutput{LockingScript: fuel.PuzzleLock(fuel.PuzzleHash(pre)), Satoshis: req.Denomination, OutputDescription: "fuel " + req.Pool}
		}
	case store.LockP2PKH:
		key, err := poolKey(c.kd, req.Pool)
		if err != nil {
			return nil, err
		}
		lock, err := fuel.P2PKHLock(key.PubKey())
		if err != nil {
			return nil, err
		}
		p2pkhLock = lock
		for i := range outputs {
			outputs[i] = sdk.CreateActionOutput{LockingScript: lock, Satoshis: req.Denomination, OutputDescription: "fuel " + req.Pool}
		}
	default:
		return nil, treasury.ErrUnsupportedLockMode
	}
	res, err := c.w.CreateAction(ctx, sdk.CreateActionArgs{
		Description: "fuel split " + req.Pool,
		Outputs:     outputs,
		Labels:      []string{"fuelkeeper", "fuel-split", "pool:" + req.Pool},
		Options:     &sdk.CreateActionOptions{NoSend: ptr(true), RandomizeOutputs: ptr(false), AcceptDelayedBroadcast: ptr(false)},
	}, c.cfg.Originator)
	if err != nil {
		if isInsufficient(err) {
			return nil, fmt.Errorf("%w: %w", treasury.ErrInsufficientFunds, err)
		}
		return nil, fmt.Errorf("treasury: create split: %w", err)
	}
	if res == nil || res.Txid == nil || len(res.Tx) == 0 {
		return nil, errors.New("treasury: create split returned no txid/beef")
	}
	txid := *res.Txid
	raw, err := fuel.RawTxFromBEEF(res.Tx, txid)
	if err != nil {
		return nil, fmt.Errorf("treasury: split beef: %w", err)
	}
	rows := make([]store.Row, req.Count)
	for i := range rows {
		rows[i] = store.Row{Outpoint: store.Outpoint{TxID: txid, Vout: uint32(i)}, Pool: req.Pool, Satoshis: req.Denomination, LockingScript: outputs[i].LockingScript} //nolint:gosec // bounded
		switch req.LockMode {
		case store.LockPuzzle:
			rows[i].UnlockingScript = fuel.PuzzleUnlock(fuel.Preimage(seed, uint32(i))) //nolint:gosec // bounded
		case store.LockP2PKH:
			key, _ := poolKey(c.kd, req.Pool)
			unl, err := fuel.P2PKHPresign(key, txid, uint32(i), p2pkhLock, req.Denomination) //nolint:gosec // bounded
			if err != nil {
				return nil, err
			}
			rows[i].UnlockingScript = unl
		}
	}
	c.mu.Lock()
	c.built[txid] = &builtInfo{pool: req.Pool, beef: res.Tx}
	c.mu.Unlock()
	return &treasury.BuiltShard{
		Shard: store.Shard{Pool: req.Pool, TxID: txid, Size: req.Count, Denomination: req.Denomination, LockMode: req.LockMode, Status: store.ShardUnmined, BEEF: res.Tx},
		Rows:  rows,
		RawTx: raw,
	}, nil
}

func isInsufficient(err error) bool {
	msg := strings.ToLower(err.Error())
	return strings.Contains(msg, "not enough funds") || strings.Contains(msg, "insufficient") || strings.Contains(msg, "contention")
}

// Send broadcasts a built split: the first time through the wallet (SendWith),
// afterwards by re-posting the stored BEEF.
func (c *Core) Send(ctx context.Context, txid chainhash.Hash) error {
	c.mu.Lock()
	info, ok := c.built[txid]
	c.mu.Unlock()
	if !ok {
		return treasury.ErrUnknownTx
	}
	if info.sent {
		if c.bc == nil {
			return nil
		}
		return c.bc.PostBEEF(ctx, info.beef, txid.String())
	}
	_, err := c.w.CreateAction(ctx, sdk.CreateActionArgs{
		Description: "send fuel split",
		Options:     &sdk.CreateActionOptions{SendWith: []chainhash.Hash{txid}, AcceptDelayedBroadcast: ptr(false)},
	}, c.cfg.Originator)
	if err != nil {
		if isRejected(err) {
			return fmt.Errorf("%w: %w", treasury.ErrRejected, err)
		}
		return fmt.Errorf("treasury: send split: %w", err)
	}
	c.mu.Lock()
	info.sent = true
	c.mu.Unlock()
	return nil
}

func isRejected(err error) bool {
	msg := strings.ToLower(err.Error())
	return strings.Contains(msg, "double") || strings.Contains(msg, "rejected") || strings.Contains(msg, "invalid")
}

// MinedBEEF converts the unmined BEEF plus BUMP into the compact mined BEEF.
func (c *Core) MinedBEEF(_ context.Context, txid chainhash.Hash, unminedBEEF, merklePath []byte) ([]byte, error) {
	return fuel.MinedBEEF(unminedBEEF, txid, merklePath)
}

// EnsureChunks runs one chunk-maintenance round (no-op without a keeper).
func (c *Core) EnsureChunks(ctx context.Context) error {
	if c.keeper == nil {
		return nil
	}
	return c.keeper.RunOnce(ctx)
}

// FundingInfo returns the identity key and a fresh derivation-prefix nonce.
func (c *Core) FundingInfo(ctx context.Context) (treasury.FundingInfo, error) {
	pk, err := c.w.GetPublicKey(ctx, sdk.GetPublicKeyArgs{IdentityKey: true}, c.cfg.Originator)
	if err != nil {
		return treasury.FundingInfo{}, fmt.Errorf("treasury: identity key: %w", err)
	}
	nonce, err := c.nonces.Create(ctx)
	if err != nil {
		return treasury.FundingInfo{}, fmt.Errorf("treasury: nonce: %w", err)
	}
	return treasury.FundingInfo{IdentityKeyHex: pk.PublicKey.ToDERHex(), DerivationPrefixB64: nonce, Network: c.cfg.Network}, nil
}

// Internalize verifies the prefix nonce and internalizes the BRC-29 payment.
func (c *Core) Internalize(ctx context.Context, req treasury.FundRequest) (treasury.FundResult, error) {
	ok, err := c.nonces.Verify(ctx, req.DerivationPrefixB64)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("treasury: verify nonce: %w", err)
	}
	if !ok {
		return treasury.FundResult{}, fmt.Errorf("%w: unknown derivation prefix", treasury.ErrBadFunding)
	}
	prefix, err := base64.StdEncoding.DecodeString(req.DerivationPrefixB64)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("%w: prefix: %w", treasury.ErrBadFunding, err)
	}
	suffix, err := base64.StdEncoding.DecodeString(req.DerivationSuffixB64)
	if err != nil || len(suffix) == 0 {
		return treasury.FundResult{}, fmt.Errorf("%w: suffix", treasury.ErrBadFunding)
	}
	senderBytes, err := hex.DecodeString(req.SenderIdentityKeyHex)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("%w: sender key: %w", treasury.ErrBadFunding, err)
	}
	sender, err := ec.ParsePubKey(senderBytes)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("%w: sender key: %w", treasury.ErrBadFunding, err)
	}
	_, txid, err := transaction.NewBeefFromAtomicBytes(req.AtomicBEEF)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("%w: atomic beef: %w", treasury.ErrBadFunding, err)
	}
	raw, err := fuel.RawTxFromBEEF(req.AtomicBEEF, *txid)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("%w: %w", treasury.ErrBadFunding, err)
	}
	tx, err := transaction.NewTransactionFromBytes(raw)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("%w: tx: %w", treasury.ErrBadFunding, err)
	}
	idx := req.OutputIndex
	if idx < 0 || idx >= len(tx.Outputs) {
		return treasury.FundResult{}, fmt.Errorf("%w: output index %d out of range", treasury.ErrBadFunding, idx)
	}
	_, err = c.w.InternalizeAction(ctx, sdk.InternalizeActionArgs{
		Tx:          req.AtomicBEEF,
		Description: "fuelkeeper funding",
		Labels:      []string{"fuelkeeper", "funding"},
		Outputs: []sdk.InternalizeOutput{{
			OutputIndex:       uint32(idx), //nolint:gosec // bounded by len(tx.Outputs)
			Protocol:          sdk.InternalizeProtocolWalletPayment,
			PaymentRemittance: &sdk.Payment{DerivationPrefix: prefix, DerivationSuffix: suffix, SenderIdentityKey: sender},
		}},
	}, c.cfg.Originator)
	if err != nil {
		return treasury.FundResult{}, fmt.Errorf("treasury: internalize: %w", err)
	}
	return treasury.FundResult{TxID: txid.String(), Satoshis: tx.Outputs[idx].Satoshis}, nil
}
```

```go
// fundCmd: fuelkeeper fund --url http://fk:8080 --sats 100000 [--wallet http://localhost:3321] [--key K]
func fundCmd(ctx context.Context, args []string) error {
	fs := flag.NewFlagSet("fund", flag.ContinueOnError)
	url := fs.String("url", "http://127.0.0.1:8080", "FuelKeeper base URL")
	key := fs.String("key", os.Getenv("FUELKEEPER_API_KEY"), "FuelKeeper API key")
	walletURL := fs.String("wallet", "http://localhost:3321", "local BRC-100 wallet (BSV Desktop / Metanet Desktop)")
	sats := fs.Uint64("sats", 0, "amount to send")
	if err := fs.Parse(args); err != nil {
		return err
	}
	if *sats == 0 {
		return fmt.Errorf("--sats is required")
	}
	fk, err := client.New(*url, client.WithAPIKey(*key))
	if err != nil {
		return err
	}
	return fund(ctx, fk, substrates.NewHTTPWalletJSON("fuelkeeper.cli", *walletURL, nil), *sats, os.Stdout)
}

// payer is the slice of the BRC-100 wallet we use (substrates.HTTPWalletJSON satisfies it).
type payer interface {
	GetPublicKey(ctx context.Context, args sdk.GetPublicKeyArgs) (*sdk.GetPublicKeyResult, error)
	CreateAction(ctx context.Context, args sdk.CreateActionArgs) (*sdk.CreateActionResult, error)
}

func fund(ctx context.Context, fk *client.Client, w payer, sats uint64, out io.Writer) error {
	info, err := fk.FundingInfo(ctx)
	if err != nil {
		return fmt.Errorf("funding info: %w", err)
	}
	fkKey, err := ec.PublicKeyFromString(info.IdentityKey)
	if err != nil {
		return fmt.Errorf("treasury identity key: %w", err)
	}
	var suffix [8]byte
	_, _ = rand.Read(suffix[:])
	suffixB64 := base64.StdEncoding.EncodeToString(suffix[:])

	me, err := w.GetPublicKey(ctx, sdk.GetPublicKeyArgs{IdentityKey: true})
	if err != nil {
		return fmt.Errorf("wallet identity key (is the wallet running at the --wallet URL?): %w", err)
	}
	derived, err := w.GetPublicKey(ctx, sdk.GetPublicKeyArgs{EncryptionArgs: sdk.EncryptionArgs{
		ProtocolID:   sdk.Protocol{SecurityLevel: sdk.SecurityLevelEveryAppAndCounterparty, Protocol: "3241645161d8"},
		KeyID:        info.DerivationPrefix + " " + suffixB64,
		Counterparty: sdk.Counterparty{Type: sdk.CounterpartyTypeOther, Counterparty: fkKey},
	}})
	if err != nil {
		return fmt.Errorf("derive payment key: %w", err)
	}
	addr, err := script.NewAddressFromPublicKey(derived.PublicKey, info.Network == "main")
	if err != nil {
		return err
	}
	lock, err := p2pkh.Lock(addr)
	if err != nil {
		return err
	}
	f := false
	res, err := w.CreateAction(ctx, sdk.CreateActionArgs{
		Description: "Fund FuelKeeper",
		Outputs:     []sdk.CreateActionOutput{{LockingScript: *lock, Satoshis: sats, OutputDescription: "FuelKeeper treasury funding"}},
		Options:     &sdk.CreateActionOptions{RandomizeOutputs: &f},
	})
	if err != nil {
		return fmt.Errorf("wallet createAction: %w", err)
	}
	if len(res.Tx) == 0 {
		return fmt.Errorf("wallet returned no transaction (is signAndProcess enabled?)")
	}
	fr, err := fk.Fund(ctx, wire.FundRequest{
		AtomicBEEF:        base64.StdEncoding.EncodeToString(res.Tx),
		OutputIndex:       intPtr(0),
		DerivationPrefix:  info.DerivationPrefix,
		DerivationSuffix:  suffixB64,
		SenderIdentityKey: me.PublicKey.ToDERHex(),
	})
	if err != nil {
		return fmt.Errorf("fuelkeeper rejected the payment (%v); the transaction %s was already broadcast by your wallet — keep the prefix/suffix and retry POST /v1/funding", err, hex.EncodeToString(res.Txid[:]))
	}
	fmt.Fprintf(out, "funded %d sats in %s\n", fr.Satoshis, fr.TxID)
	return nil
}

func intPtr(i int) *int { return &i }
```
